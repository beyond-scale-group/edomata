34async fn journal_of(s: &EsBackend, address: &str) -> Vec<i32> {
35    s.journal()
36        .read_stream(address)
37        .map_ok(|e| e.payload)
38        .try_collect()
39        .await
40        .unwrap()
41}

43async fn outbox_of(s: &EsBackend, address: &str) -> Vec<i32> {
44    s.outbox()
45        .read()
46        .try_filter(|i| std::future::ready(i.stream_id == address))
47        .map_ok(|i| i.data)
48        .try_collect()
49        .await
50        .unwrap()
51}

53/// Waits for one update signal, failing after a short timeout.
54async fn assert_notified(mut stream: futures::stream::BoxStream<'static, ()>, what: &str) {
55    tokio::time::timeout(Duration::from_secs(5), stream.next())
56        .await
57        .unwrap_or_else(|_| panic!("{what} listeners were not notified"))
58        .unwrap_or_else(|| panic!("{what} signal ended"));
59}
60
61async fn assert_notified_journal(s: &EsBackend) {
62    assert_notified(s.updates().journal(), "journal").await;
63}
64
65async fn assert_notified_outbox(s: &EsBackend) {
66    assert_notified(s.updates().outbox(), "outbox").await;
67}

69fn accept_and_publish() -> App<()> {
70    Edomaton::lift(ResponseD::accept(NonEmpty::of(1, [2, 3])).publish([4, 5, 6]))

75/// "Must append correctly"
76pub async fn must_append_correctly(s: &EsBackend) {
77    let cmd = some_cmd();
78    s.compile(accept_and_publish())(cmd.clone())
79        .await
80        .unwrap()
81        .unwrap();
82    assert_eq!(journal_of(s, &cmd.address).await, vec![1, 2, 3]);
83    assert_eq!(outbox_of(s, &cmd.address).await, vec![4, 5, 6]);
84    assert_notified_journal(s).await;
85    assert_notified_outbox(s).await;
86}

88/// "Appending must be idempotent"
89pub async fn appending_must_be_idempotent(s: &EsBackend) {
90    let cmd = some_cmd();
91    let service = s.compile(accept_and_publish());
92    let attempts = futures::future::join_all((0..20).map(|_| service(cmd.clone()))).await;
93    for outcome in attempts {
94        match outcome {
95            Ok(_) | Err(BackendError::VersionConflict) => {}
96            Err(other) => panic!("Invalid implementation: {other}"),
97        }
98    }
99    assert_eq!(journal_of(s, &cmd.address).await, vec![1, 2, 3]);
100    assert_eq!(outbox_of(s, &cmd.address).await, vec![4, 5, 6]);
101    assert_notified_journal(s).await;
102    assert_notified_outbox(s).await;
103}

115/// "Must consume outbox correctly"
116pub async fn must_consume_outbox_correctly(s: &EsBackend) {
117    let cmd = some_cmd();
118    let app: App<()> = Edomaton::lift(ResponseD::publish_only([4, 5, 6]));
119    s.compile(app)(cmd.clone()).await.unwrap().unwrap();
120    let items: Vec<OutboxItem<i32>> = s.outbox().read().try_collect().await.unwrap();
121    assert!(items.len() >= 3);
122    let consumed = items[0].clone();
123    s.outbox().mark_as_sent(&consumed).await.unwrap();
124    let items2: Vec<OutboxItem<i32>> = s.outbox().read().try_collect().await.unwrap();
125    assert!(items2.len() >= 2);
126    assert_eq!(items2, items[1..].to_vec());
127    let consumed_all = NonEmpty::from_vec(items2.clone()).unwrap();
128    s.outbox().mark_all_as_sent(&consumed_all).await.unwrap();
129    let items3: Vec<OutboxItem<i32>> = s.outbox().read().try_collect().await.unwrap();
130    assert!(
131        !items3.windows(items2.len()).any(|w| w == items2.as_slice()),
132        "marked items must not be read again"
133    );
134}

136/// "Must read all journal"
137pub async fn must_read_all_journal(s: &EsBackend) {
138    let cmd = some_cmd();
139    let app: App<()> = Edomaton::lift(ResponseD::accept(NonEmpty::of(1, [2, 3])));
140    s.compile(app)(cmd).await.unwrap().unwrap();
141    let events: Vec<EventMessage<i32>> =
142        s.journal().read_all().take(10).try_collect().await.unwrap();
143    let ev_size = events.len();
144    assert!(ev_size >= 3);
145    let mut sorted = events.clone();
146    sorted.sort_by_key(|e| e.metadata.seq_nr);
147    assert_eq!(events, sorted);
148    let pivot_idx = ev_size / 2;
149    let pivot = events[pivot_idx].metadata.seq_nr;
150    let before = &events[..pivot_idx];
151    let after = &events[pivot_idx..];
152    let read_before: Vec<EventMessage<i32>> = s
153        .journal()
154        .read_all_before(pivot)
155        .take(before.len())
156        .try_collect()
157        .await
158        .unwrap();
159    assert_eq!(read_before, before);
160    let read_after: Vec<EventMessage<i32>> = s
161        .journal()
162        .read_all_after(pivot - 1)
163        .take(after.len())
164        .try_collect()
165        .await
166        .unwrap();
167    assert_eq!(read_after, after);
168}

170/// "Must read single stream from journal"
171pub async fn must_read_single_stream_from_journal(s: &EsBackend) {
172    let cmd = some_cmd();
173    let app: App<()> = Edomaton::lift(ResponseD::accept(NonEmpty::of(1, [2, 3])));
174    s.compile(app)(cmd.clone()).await.unwrap().unwrap();
175    let events: Vec<EventMessage<i32>> = s
176        .journal()
177        .read_stream(&cmd.address)
178        .take(10)
179        .try_collect()
180        .await
181        .unwrap();
182    assert_eq!(events.len(), 3);
183    let mut by_seq = events.clone();
184    by_seq.sort_by_key(|e| e.metadata.seq_nr);
185    let mut by_version = events.clone();
186    by_version.sort_by_key(|e| e.metadata.version);
187    assert_eq!(events, by_seq);
188    assert_eq!(events, by_version);
189    assert!(events.iter().all(|e| e.metadata.stream == cmd.address));
190    let pivot_idx = 1;
191    let pivot = events[pivot_idx].metadata.version;
192    let before = &events[..pivot_idx];
193    let after = &events[pivot_idx..];
194    let read_before: Vec<EventMessage<i32>> = s
195        .journal()
196        .read_stream_before(&cmd.address, pivot)
197        .take(before.len())
198        .try_collect()
199        .await
200        .unwrap();
201    assert_eq!(read_before, before);
202    let read_after: Vec<EventMessage<i32>> = s
203        .journal()
204        .read_stream_after(&cmd.address, pivot - 1)
205        .take(after.len())
206        .try_collect()
207        .await
208        .unwrap();
209    assert_eq!(read_after, after);
210}

231                version: 0,

234            payload: 1234,
235        }]
236    }

238    /// The one outbox item.
239    pub fn outbox() -> Vec<OutboxItem<i32>> {
240        vec![OutboxItem {
241            seq_nr: 1,
242            stream_id: STREAM_ID.to_string(),
243            time: DateTime::UNIX_EPOCH,
244            data: 123456,
245            metadata: MessageMetadata::new("correlation", "causation"),
246        }]
247    }

249    /// The persisted snapshot of the aggregate.
250    pub fn aggregate() -> ValidState<i32> {
251        ValidState::new(12, 1)
252    }

254    fn redundant_command() -> CommandMessage<()> {
255        CommandMessage::new(REDUNDANT_CMD, DateTime::<Utc>::MIN_UTC, STREAM_ID, ())
256    }

258    /// "Must read all journal"
259    pub async fn must_read_all_journal(s: &EsBackend) {

261        assert_eq!(all, journal());

277    /// "Must load for non existing command id"
278    pub async fn must_load_for_non_existing_command_id(s: &EsBackend) {
279        let mut new_command = redundant_command();
280        new_command.id = format!("new-{REDUNDANT_CMD}");
281        let seen: Arc<Mutex<Option<RequestContext<(), i32>>>> = Arc::new(Mutex::new(None));
282        let sink = Arc::clone(&seen);
283        let app: Edomaton<RequestContext<(), i32>, String, i32, i32, ()> =
284            Edomaton::run_with(move |ctx: RequestContext<(), i32>| {
285                let sink = Arc::clone(&sink);
286                async move {
287                    *sink.lock().unwrap() = Some(ctx);
288                }
289            });
290        s.compile(app)(new_command.clone()).await.unwrap().unwrap();
291        let ctx = seen.lock().unwrap().clone();
292        assert_eq!(ctx, Some(RequestContext::new(new_command, 12)));
293    }
294
295    /// "Must skip loading for existing command id"
296    pub async fn must_skip_loading_for_existing_command_id(s: &EsBackend) {
297        let counter = Arc::new(Mutex::new(0));
298        let c = Arc::clone(&counter);
299        let app: Edomaton<RequestContext<(), i32>, String, i32, i32, ()> =
300            Edomaton::eval(move || {
301                let c = Arc::clone(&c);
302                async move {
303                    *c.lock().unwrap() += 1;
304                }
305            });
306        s.compile(app)(redundant_command()).await.unwrap().unwrap();
307        assert_eq!(*counter.lock().unwrap(), 0);
308    }
309}
