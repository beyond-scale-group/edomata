1//! Shared behavioural suites for Edomata storages: the port of
2//! `modules/backend-tests`.
3//!
4//! Every check is an `async fn` taking a backend (or a snapshot
5//! persistence) so that the same checks run against the in-memory driver
6//! and against PostgreSQL. Test crates call them from `#[tokio::test]`
7//! functions, which replaces Scala's `StorageSuite.check`.
8//!
9//! ```
10//! use edomata_backend::cqrs::Backend as CqrsBackend;
11//! use edomata_backend::eventsourcing::Backend;
12//! use edomata_backend::inmemory::InMemoryDriver;
13//! use edomata_backend_tests::{cqrs, eventsourcing as es};
14//! use edomata_backend_tests::{TestCqrsModel, TestDomain, test_cqrs_dsl, test_domain_dsl};
15//!
16//! # tokio::runtime::Builder::new_current_thread().enable_time().build().unwrap().block_on(async {
17//! // In a test crate, each check is usually its own `#[tokio::test]`.
18//! let backend = Backend::builder(TestDomain, test_domain_dsl())
19//!     .driver(InMemoryDriver::new())
20//!     .build_default()
21//!     .await
22//!     .unwrap();
23//! es::must_append_correctly(&backend).await;
24//! es::appending_must_be_idempotent(&backend).await;
25//!
26//! let backend = CqrsBackend::builder(TestCqrsModel, test_cqrs_dsl())
27//!     .driver(InMemoryDriver::new())
28//!     .build_default()
29//!     .await
30//!     .unwrap();
31//! cqrs::inserts_state(&backend).await;
32//! # });
33//! ```
34//!
35//! To check a new storage driver, build its backends with [`TestDomain`] /
36//! [`TestCqrsModel`] and run every check of [`eventsourcing`] and [`cqrs`];
37//! for the [`eventsourcing::prepared_data`] checks, seed the storage with
38//! the rows that module describes first. `edomata-sqlx` does this in
39//! `tests/shared_suites.rs`.
40//!
41//! ## Panics
42//!
43//! Every check panics (through `assert!` and `unwrap`) when the storage does
44//! not behave as expected: that is how it fails the calling test.
45//!
46//! ## Where it fits
47//!
48//! A test-support crate: it depends on `edomata-core` and `edomata-backend`
49//! and is a dev-dependency of the storage drivers (`edomata-sqlx`). It is
50//! the port of the Scala `backend-tests` module. Its checks run against the
51//! in-memory driver in this crate's own tests.
52//!
53//! ## Feature flags
54//!
55//! This crate has no Cargo feature flags.

67use edomata_core::{CqrsDomainDsl, CqrsModel, DomainDsl, DomainModel, NonEmpty};

69/// Port of `TestDomain`: an event-sourced counter whose events are added to
70/// the state.
71#[derive(Clone, Copy, Debug, Default)]
72pub struct TestDomain;
73
74impl DomainModel for TestDomain {
75    type State = i32;
76    type Event = i32;
77    type Rejection = String;
78
79    fn initial(&self) -> i32 {
80        0
81    }
82
83    fn transition(&self, event: &i32, state: i32) -> Result<i32, NonEmpty<String>> {
84        Ok(event + state)
85    }
86}
87
88/// The DSL of [`TestDomain`] with `i32` commands and notifications.
89pub fn test_domain_dsl() -> DomainDsl<i32, i32, i32, String, i32> {
90    TestDomain.dsl()
91}
