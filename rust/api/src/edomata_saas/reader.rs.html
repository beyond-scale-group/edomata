10/// Reads a single entity on behalf of a caller; implementations filter by
11/// the caller's tenant, so an entity of another tenant reads as missing.
12///
13/// ```
14/// use std::collections::HashMap;
15/// use edomata_saas::{AuthPolicy, CallerIdentity, CrudState, PermissivePolicy, TenantAwareReader};
16///
17/// struct Todos(HashMap<String, CrudState<String>>);
18///
19/// #[async_trait::async_trait]
20/// impl TenantAwareReader<CallerIdentity, String> for Todos {
21///     async fn get(&self, auth: &CallerIdentity, id: &str) -> Option<String> {
22///         let tenant = PermissivePolicy.tenant_id(auth);
23///         match self.0.get(id)? {
24///             CrudState::Active { tenant_id, data, .. } if *tenant_id == tenant => Some(data.clone()),
25///             _ => None,
26///         }
27///     }
28/// }
29///
30/// let todos = Todos(HashMap::from([("t1".to_string(), CrudState::active("acme", "alice", "milk".to_string()))]));
31/// let alice = CallerIdentity::new("acme", "alice", Vec::<String>::new());
32/// let mallory = CallerIdentity::new("evil", "mallory", Vec::<String>::new());
33/// futures::executor::block_on(async {
34///     assert_eq!(todos.get(&alice, "t1").await.as_deref(), Some("milk"));
35///     assert_eq!(todos.get(&mallory, "t1").await, None);
36/// });
37/// ```
38#[async_trait]
39pub trait TenantAwareReader<Auth, A>: Send + Sync {
40    /// The entity, if it exists and belongs to the caller's tenant.
41    async fn get(&self, auth: &Auth, entity_id: &str) -> Option<A>;
42}
43
44/// A query whose results are scoped to the caller's tenant. The `Auth`
45/// parameter is structurally required, so tenant filtering cannot be
46/// forgotten.
47#[async_trait]
48pub trait TenantScopedQuery<Auth, A, Q>: Send + Sync {
49    /// Runs the query for the caller's tenant.
50    async fn query(&self, auth: &Auth, q: Q) -> Vec<A>;
51}
52
53/// A query across all tenants. The `unsafe` prefix flags usage in code
54/// reviews; use it only for administrative endpoints.
55#[async_trait]
56pub trait UnsafeCrossTenantQuery<A, Q>: Send + Sync {
57    /// Runs the query.
58    async fn query(&self, q: Q) -> Vec<A>;
59}
60
61type ScopedRun<A, Q> = dyn Fn(TenantId, Q) -> BoxFuture<'static, Vec<A>> + Send + Sync;
62type CrossRun<A, Q> = dyn Fn(Q) -> BoxFuture<'static, Vec<A>> + Send + Sync;
63
64/// A [`TenantScopedQuery`] built from a policy and a function of the
65/// tenant (Scala's `TenantScopedQuery.apply`): the function receives the
66/// tenant extracted by the policy, never the raw auth context.
67///
68/// ```
69/// use edomata_saas::{CallerIdentity, PermissivePolicy, ScopedQueryFn, TenantScopedQuery};
70///
71/// let query = ScopedQueryFn::new(PermissivePolicy, |tenant, prefix: String| async move {
72///     vec![format!("{prefix} of {tenant}")]
73/// });
74/// let caller = CallerIdentity::new("acme", "alice", Vec::<String>::new());
75/// let rows = futures::executor::block_on(query.query(&caller, "todos".to_string()));
76/// assert_eq!(rows, ["todos of acme"]);
77/// ```
78pub struct ScopedQueryFn<Auth, A, Q> {
79    policy: Arc<dyn AuthPolicy<Auth>>,
80    run: Arc<ScopedRun<A, Q>>,
81}
82
83impl<Auth, A, Q> ScopedQueryFn<Auth, A, Q> {
84    /// Builds the query.
85    pub fn new<P, F, Fut>(policy: P, run: F) -> Self
86    where
87        P: AuthPolicy<Auth> + 'static,
88        F: Fn(TenantId, Q) -> Fut + Send + Sync + 'static,
89        Fut: std::future::Future<Output = Vec<A>> + Send + 'static,
90    {
91        Self {
92            policy: Arc::new(policy),
93            run: Arc::new(move |tid, q| Box::pin(run(tid, q))),
94        }
95    }
96}
97
98#[async_trait]
99impl<Auth, A, Q> TenantScopedQuery<Auth, A, Q> for ScopedQueryFn<Auth, A, Q>
100where
101    Auth: Sync,
102    A: Send,
103    Q: Send + 'static,
104{
105    async fn query(&self, auth: &Auth, q: Q) -> Vec<A> {
106        (self.run)(self.policy.tenant_id(auth), q).await
107    }
108}

110/// An [`UnsafeCrossTenantQuery`] built from a function (Scala's
111/// `UnsafeCrossTenantQuery.apply`).
112pub struct CrossTenantQueryFn<A, Q> {
113    run: Arc<CrossRun<A, Q>>,
114}
115
116impl<A, Q> CrossTenantQueryFn<A, Q> {
117    /// Builds the query.
118    pub fn new<F, Fut>(run: F) -> Self
119    where
120        F: Fn(Q) -> Fut + Send + Sync + 'static,
121        Fut: std::future::Future<Output = Vec<A>> + Send + 'static,
122    {
123        Self {
124            run: Arc::new(move |q| Box::pin(run(q))),
125        }
126    }
127}
128
129#[async_trait]
130impl<A, Q> UnsafeCrossTenantQuery<A, Q> for CrossTenantQueryFn<A, Q>
131where
132    A: Send,
133    Q: Send + 'static,
134{
135    async fn query(&self, q: Q) -> Vec<A> {
136        (self.run)(q).await
137    }
