5/// PostgreSQL Row-Level Security configuration for
6/// [`SaaSPGSchema::cqrs_with`].
7#[derive(Clone, Debug, PartialEq, Eq)]
8pub struct RlsConfig {
9    /// The PostgreSQL role to grant access to (e.g. `"app_user"`).
10    pub pg_role: String,
11    /// The session variable holding the current tenant id (e.g.
12    /// `"app.tenant_id"`).
13    pub tenant_session_var: String,
14}
15
16impl RlsConfig {
17    /// Builds a configuration.
18    pub fn new(pg_role: impl Into<String>, tenant_session_var: impl Into<String>) -> Self {
19        Self {
20            pg_role: pg_role.into(),
21            tenant_session_var: tenant_session_var.into(),
22        }
23    }
24}

26/// DDL for SaaS-aware CQRS tables. Unlike `PGSchema`, the states and outbox
27/// tables carry `tenant_id` (and `owner_id`) columns, enabling Row-Level
28/// Security, tenant-scoped indexes and cross-tenant audit queries. The
29/// output is byte-for-byte identical to Scala's `SaaSPGSchema`.
30///
31/// ```
32/// use edomata_saas::{PGNaming, RlsConfig, SaaSPGSchema};
33///
34/// let naming = PGNaming::prefixed_str("catalog").unwrap();
35/// let ddl = SaaSPGSchema::cqrs_with(&naming, "jsonb", "jsonb", Some(&RlsConfig::new("app_user", "app.tenant_id")));
36/// assert!(ddl.iter().any(|s| s.contains("ENABLE ROW LEVEL SECURITY")));
37/// assert!(ddl.iter().any(|s| s.contains("USING (tenant_id = current_setting('app.tenant_id'))")));
38/// assert!(ddl.iter().any(|s| s.starts_with("GRANT SELECT, INSERT, UPDATE ON catalog_states TO app_user")));
39/// ```
40///
41/// With RLS, the policies compare `tenant_id` with the session variable, so
42/// connections using `pg_role` must set it (`SET app.tenant_id = 'acme'`,
43/// or `set_config` per transaction) before touching the tables. Unlike the
44/// table statements, `CREATE POLICY` has no `IF NOT EXISTS`: run the RLS
45/// statements once, from a migration tool.
46#[derive(Clone, Copy, Debug, Default)]
47pub struct SaaSPGSchema;
48
49impl SaaSPGSchema {
50    /// DDL for states (with `tenant_id`, `owner_id`), outbox (with
51    /// `tenant_id`) and commands, `jsonb` payloads, no RLS. In schema naming
52    /// mode the first statement is `CREATE SCHEMA IF NOT EXISTS`.
53    pub fn cqrs(naming: &PGNaming) -> Vec<String> {
54        Self::cqrs_with(naming, "jsonb", "jsonb", None)
55    }
56
57    /// DDL with explicit payload types and optional RLS statements
58    /// (appended last). Payload types are spliced into the DDL without
59    /// validation; use `"jsonb"`, `"json"` or `"bytea"`.
60    pub fn cqrs_with(
61        naming: &PGNaming,
62        state_type: &str,
63        notification_type: &str,
64        rls: Option<&RlsConfig>,
65    ) -> Vec<String> {
66        let mut out = edomata_postgres::ddl::schema_statement(naming);
67        out.extend(ddl::states_statements(naming, state_type));
68        out.extend(ddl::outbox_statements(naming, notification_type));
69        out.extend(edomata_postgres::ddl::commands_statements(naming));
70        out.extend(ddl::rls_statements(naming, rls));
71        out
72    }
73}

75/// The individual tenant-aware DDL statements, reused by the SaaS driver.
76pub mod ddl {
77    use edomata_postgres::PGNaming;
78
79    use super::RlsConfig;
80
81    /// The states table with tenant and owner columns and indexes.
82    pub fn states_statements(naming: &PGNaming, payload_type: &str) -> Vec<String> {
83        let t = naming.table("states");
84        let pk = naming.constraint("states_pk");
85        let tenant_idx = naming.index("states_tenant_idx");
86        let tenant_owner_idx = naming.index("states_tenant_owner_idx");
87        vec![
88            format!(
89                "CREATE TABLE IF NOT EXISTS {t} (\n  id text NOT NULL,\n  \"version\" int8 NOT NULL,\n  state {payload_type} NOT NULL,\n  tenant_id text NOT NULL,\n  owner_id text NOT NULL,\n  CONSTRAINT {pk} PRIMARY KEY (id)\n);"
90            ),
91            format!("CREATE INDEX IF NOT EXISTS {tenant_idx} ON {t} (tenant_id);"),
92            format!("CREATE INDEX IF NOT EXISTS {tenant_owner_idx} ON {t} (tenant_id, owner_id);"),
93        ]
94    }
95
96    /// The outbox table with a tenant column and index.
97    pub fn outbox_statements(naming: &PGNaming, payload_type: &str) -> Vec<String> {
98        let t = naming.table("outbox");
99        let pk = naming.constraint("outbox_pk");
100        let tenant_idx = naming.index("outbox_tenant_idx");
101        vec![
102            format!(
103                "CREATE TABLE IF NOT EXISTS {t} (\n  seqnr bigserial NOT NULL,\n  stream text NOT NULL,\n  correlation text NULL,\n  causation text NULL,\n  payload {payload_type} NOT NULL,\n  created timestamptz NOT NULL,\n  published timestamptz NULL,\n  tenant_id text NOT NULL,\n  CONSTRAINT {pk} PRIMARY KEY (seqnr)\n);"
104            ),
105            format!("CREATE INDEX IF NOT EXISTS {tenant_idx} ON {t} (tenant_id);"),
106        ]
107    }
108
109    /// Row-Level Security policies and grants, when configured (empty for
110    /// `None`).
111    pub fn rls_statements(naming: &PGNaming, rls: Option<&RlsConfig>) -> Vec<String> {
112        let Some(config) = rls else {
113            return Vec::new();
114        };
115        let states_t = naming.table("states");
116        let outbox_t = naming.table("outbox");
117        let states_policy = naming.constraint("states_tenant_policy");
118        let outbox_policy = naming.constraint("outbox_tenant_policy");
119        let var = &config.tenant_session_var;
120        let role = &config.pg_role;
121        vec![
122            format!("ALTER TABLE {states_t} ENABLE ROW LEVEL SECURITY;"),
123            format!(
124                "CREATE POLICY {states_policy} ON {states_t}\n  USING (tenant_id = current_setting('{var}'));"
125            ),
126            format!("GRANT SELECT, INSERT, UPDATE ON {states_t} TO {role};"),
127            format!("ALTER TABLE {outbox_t} ENABLE ROW LEVEL SECURITY;"),
128            format!(
129                "CREATE POLICY {outbox_policy} ON {outbox_t}\n  USING (tenant_id = current_setting('{var}'));"
130            ),
131            format!("GRANT SELECT, INSERT, UPDATE ON {outbox_t} TO {role};"),
132        ]
133    }
