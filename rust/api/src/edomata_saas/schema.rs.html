8/// PostgreSQL Row-Level Security configuration for
9/// [`SaaSPGSchema::cqrs_with`].
10#[derive(Clone, Debug, PartialEq, Eq)]
11pub struct RlsConfig {
12    /// The PostgreSQL role to grant access to (e.g. `"app_user"`).
13    pub pg_role: String,
14    /// The session variable holding the current tenant id (e.g.
15    /// `"app.tenant_id"`).
16    pub tenant_session_var: String,
17}
18
19impl RlsConfig {
20    /// Builds a configuration.
21    pub fn new(pg_role: impl Into<String>, tenant_session_var: impl Into<String>) -> Self {
22        Self {
23            pg_role: pg_role.into(),
24            tenant_session_var: tenant_session_var.into(),
25        }
26    }
27}

29/// DDL for SaaS-aware CQRS tables. Unlike `PGSchema`, the states and outbox
30/// tables carry `tenant_id` (and `owner_id`) columns, enabling Row-Level
31/// Security, tenant-scoped indexes and cross-tenant audit queries. The
32/// output is byte-for-byte identical to Scala's `SaaSPGSchema`.
33///
34/// ```
35/// use edomata_saas::{PGNaming, RlsConfig, SaaSPGSchema};
36///
37/// let naming = PGNaming::prefixed_str("catalog").unwrap();
38/// let ddl = SaaSPGSchema::cqrs_with(&naming, "jsonb", "jsonb", Some(&RlsConfig::new("app_user", "app.tenant_id")));
39/// assert!(ddl.iter().any(|s| s.contains("ENABLE ROW LEVEL SECURITY")));
40/// assert!(ddl.iter().any(|s| s.contains("USING (tenant_id = current_setting('app.tenant_id'))")));
41/// assert!(ddl.iter().any(|s| s.starts_with("GRANT SELECT, INSERT, UPDATE ON catalog_states TO app_user")));
42/// ```
43///
44/// With RLS, the policies compare `tenant_id` with the session variable, so
45/// connections using `pg_role` must set it (`SET app.tenant_id = 'acme'`,
46/// or `set_config` per transaction) before touching the tables. Unlike the
47/// table statements, `CREATE POLICY` has no `IF NOT EXISTS`: run the RLS
48/// statements once, from a migration tool.
49#[derive(Clone, Copy, Debug, Default)]
50pub struct SaaSPGSchema;
51
52impl SaaSPGSchema {
53    /// DDL for states (with `tenant_id`, `owner_id`), outbox (with
54    /// `tenant_id`) and commands, `jsonb` payloads, no RLS. In schema naming
55    /// mode the first statement is `CREATE SCHEMA IF NOT EXISTS`.
56    pub fn cqrs(naming: &PGNaming) -> Vec<String> {
57        Self::cqrs_with(naming, "jsonb", "jsonb", None)
58    }
59
60    /// DDL with explicit payload types and optional RLS statements
61    /// (appended last). Payload types are spliced into the DDL without
62    /// validation; use `"jsonb"`, `"json"` or `"bytea"`.
63    pub fn cqrs_with(
64        naming: &PGNaming,
65        state_type: &str,
66        notification_type: &str,
67        rls: Option<&RlsConfig>,
68    ) -> Vec<String> {
69        let mut out = edomata_postgres::ddl::schema_statement(naming);
70        out.extend(ddl::states_statements(naming, state_type));
71        out.extend(ddl::outbox_statements(naming, notification_type));
72        out.extend(edomata_postgres::ddl::commands_statements(naming));
73        out.extend(ddl::rls_statements(naming, rls));
74        out
75    }
76
77    /// DDL of the **event-sourced** tables (`journal`, `outbox`,
78    /// `commands`, `snapshots`) under mandatory Row-Level Security: the
79    /// tables with their generated `tenant_id` column
80    /// ([`eventsourcing_rls_tables`](Self::eventsourcing_rls_tables)) followed
81    /// by the RLS statements (`ENABLE` and `FORCE`, the three policies per
82    /// table, the grants). Payload types are spliced as is (`jsonb`, `json`
83    /// or `bytea`).
84    ///
85    /// Run it **once, from a migration tool, as the owner of the tables**
86    /// (a role the application does not use): `CREATE POLICY` has no `IF
87    /// NOT EXISTS`, so applying it twice fails; the table statements are
88    /// idempotent. Create the roles before
89    /// ([`rls_roles`](Self::rls_roles)) and grant the login roles
90    /// afterwards ([`rls_memberships`](Self::rls_memberships)).
91    ///
92    /// ```
93    /// use edomata_saas::{PGNaming, RlsAccessPolicy, SaaSPGSchema, TenantId};
94    ///
95    /// let naming = PGNaming::prefixed_str("shop").unwrap();
96    /// let policy: RlsAccessPolicy<TenantId> = RlsAccessPolicy::with_prefix("shop").unwrap();
97    /// let ddl = SaaSPGSchema::eventsourcing_rls(&naming, "jsonb", "jsonb", "jsonb", &policy);
98    /// assert!(ddl.iter().any(|s| s == "ALTER TABLE shop_journal FORCE ROW LEVEL SECURITY;"));
99    /// assert!(ddl.iter().any(|s| s.contains("CREATE POLICY shop_journal_tenant_policy")));
100    /// ```
101    pub fn eventsourcing_rls<T>(
102        naming: &PGNaming,
103        event_type: &str,
104        notification_type: &str,
105        snapshot_type: &str,
106        policy: &RlsAccessPolicy<T>,
107    ) -> Vec<String> {
108        let mut out =
109            Self::eventsourcing_rls_tables(naming, event_type, notification_type, snapshot_type);
110        out.extend(ddl::es_rls_statements(naming, policy));
111        out
112    }
113
114    /// Only the (idempotent) table statements of
115    /// [`eventsourcing_rls`](Self::eventsourcing_rls): `journal`,
116    /// `outbox`, `commands` and `snapshots`, each with a `tenant_id` column
117    /// **generated** from the tenant-prefixed stream or command id, and the
118    /// indexes. The `migrations` and `relay_checkpoints` tables hold no
119    /// tenant data and are not part of it (append
120    /// [`edomata_postgres::PGSchema::relay_checkpoints`] or
121    /// `ddl::migrations_statements` when you use them).
122    pub fn eventsourcing_rls_tables(
123        naming: &PGNaming,
124        event_type: &str,
125        notification_type: &str,
126        snapshot_type: &str,
127    ) -> Vec<String> {
128        let mut out = edomata_postgres::ddl::schema_statement(naming);
129        out.extend(ddl::es_journal_statements(naming, event_type));
130        out.extend(ddl::es_outbox_statements(naming, notification_type));
131        out.extend(ddl::es_commands_statements(naming));
132        out.extend(ddl::es_snapshots_statements(naming, snapshot_type));
133        out
134    }
135
136    /// Statements creating the three `NOLOGIN` roles of `policy` unless they
137    /// exist (idempotent; roles are cluster-wide, so a DBA usually runs
138    /// them, not the migration of one service).
139    pub fn rls_roles<T>(policy: &RlsAccessPolicy<T>) -> Vec<String> {
140        [
141            policy.tenant_role(),
142            policy.system_role(),
143            policy.inspection_role(),
144        ]
145        .into_iter()
146        .map(|role| {
147            format!(
148                "DO $$ BEGIN IF NOT EXISTS (SELECT FROM pg_roles WHERE rolname = '{role}') THEN CREATE ROLE {role} NOLOGIN NOSUPERUSER NOBYPASSRLS; END IF; END $$;"
149            )
150        })
151        .collect()
152    }
153
154    /// `GRANT role TO login` for each of `modes`: the **membership** that
155    /// lets connections of `login` `SET LOCAL ROLE` to those roles. Grant
156    /// each login only the modes it needs: the web tier's login gets
157    /// [`AccessMode::Tenant`], a relay's login [`AccessMode::System`], an
158    /// inspection agent's login [`AccessMode::Inspection`]. Create the
159    /// login with `NOINHERIT`, so that it has no access without the explicit
160    /// `SET LOCAL ROLE` of the entry point.
161    ///
162    /// # Errors
163    ///
164    /// [`RlsPolicyError::InvalidIdentifier`] if `login` is not a lowercase
165    /// identifier.
166    ///
167    /// ```
168    /// use edomata_saas::{AccessMode, RlsAccessPolicy, SaaSPGSchema, TenantId};
169    ///
170    /// let policy: RlsAccessPolicy<TenantId> = RlsAccessPolicy::with_prefix("shop").unwrap();
171    /// let sql = SaaSPGSchema::rls_memberships(&policy, "shop_web", &[AccessMode::Tenant]).unwrap();
172    /// assert_eq!(sql, vec!["GRANT shop_tenant TO shop_web;".to_string()]);
173    /// ```
174    pub fn rls_memberships<T>(
175        policy: &RlsAccessPolicy<T>,
176        login: &str,
177        modes: &[AccessMode],
178    ) -> Result<Vec<String>, RlsPolicyError> {
179        check_identifier(login)?;
180        Ok(modes
181            .iter()
182            .map(|mode| format!("GRANT {} TO {login};", policy.role(*mode)))
183            .collect())
184    }
185}

193    /// The states table with tenant and owner columns and indexes.
194    pub fn states_statements(naming: &PGNaming, payload_type: &str) -> Vec<String> {
195        let t = naming.table("states");
196        let pk = naming.constraint("states_pk");
197        let tenant_idx = naming.index("states_tenant_idx");
198        let tenant_owner_idx = naming.index("states_tenant_owner_idx");
199        vec![
200            format!(
201                "CREATE TABLE IF NOT EXISTS {t} (\n  id text NOT NULL,\n  \"version\" int8 NOT NULL,\n  state {payload_type} NOT NULL,\n  tenant_id text NOT NULL,\n  owner_id text NOT NULL,\n  CONSTRAINT {pk} PRIMARY KEY (id)\n);"
202            ),
203            format!("CREATE INDEX IF NOT EXISTS {tenant_idx} ON {t} (tenant_id);"),
204            format!("CREATE INDEX IF NOT EXISTS {tenant_owner_idx} ON {t} (tenant_id, owner_id);"),
205        ]
206    }

208    /// The outbox table with a tenant column and index.
209    pub fn outbox_statements(naming: &PGNaming, payload_type: &str) -> Vec<String> {
210        let t = naming.table("outbox");
211        let pk = naming.constraint("outbox_pk");
212        let tenant_idx = naming.index("outbox_tenant_idx");
213        vec![
214            format!(
215                "CREATE TABLE IF NOT EXISTS {t} (\n  seqnr bigserial NOT NULL,\n  stream text NOT NULL,\n  correlation text NULL,\n  causation text NULL,\n  payload {payload_type} NOT NULL,\n  created timestamptz NOT NULL,\n  published timestamptz NULL,\n  tenant_id text NOT NULL,\n  CONSTRAINT {pk} PRIMARY KEY (seqnr)\n);"
216            ),
217            format!("CREATE INDEX IF NOT EXISTS {tenant_idx} ON {t} (tenant_id);"),
218        ]

221    /// Row-Level Security policies and grants, when configured (empty for
222    /// `None`).
223    pub fn rls_statements(naming: &PGNaming, rls: Option<&RlsConfig>) -> Vec<String> {
224        let Some(config) = rls else {
225            return Vec::new();
226        };
227        let states_t = naming.table("states");
228        let outbox_t = naming.table("outbox");
229        let states_policy = naming.constraint("states_tenant_policy");
230        let outbox_policy = naming.constraint("outbox_tenant_policy");
231        let var = &config.tenant_session_var;
232        let role = &config.pg_role;
233        vec![
234            format!("ALTER TABLE {states_t} ENABLE ROW LEVEL SECURITY;"),
235            format!(
236                "CREATE POLICY {states_policy} ON {states_t}\n  USING (tenant_id = current_setting('{var}'));"
237            ),
238            format!("GRANT SELECT, INSERT, UPDATE ON {states_t} TO {role};"),
239            format!("ALTER TABLE {outbox_t} ENABLE ROW LEVEL SECURITY;"),
240            format!(
241                "CREATE POLICY {outbox_policy} ON {outbox_t}\n  USING (tenant_id = current_setting('{var}'));"
242            ),
243            format!("GRANT SELECT, INSERT, UPDATE ON {outbox_t} TO {role};"),
244        ]
245    }
246
247    /// The generated `tenant_id` expression over a tenant-prefixed id
248    /// column: the text before the first `/`, and NULL (so the NOT NULL
249    /// constraint refuses the row) when there is no `/` or the tenant is
250    /// empty.
251    fn tenant_of(column: &str) -> String {
252        format!("CASE WHEN position('/' in {column}) > 1 THEN split_part({column}, '/', 1) END")
253    }
254
255    fn tenant_column(column: &str) -> String {
256        format!(
257            "  tenant_id text GENERATED ALWAYS AS ({}) STORED NOT NULL",
258            tenant_of(column)
259        )
260    }

262    /// The event-sourced journal with a `tenant_id` column generated from
263    /// `stream`.
264    pub fn es_journal_statements(naming: &PGNaming, payload_type: &str) -> Vec<String> {
265        let t = naming.table("journal");
266        let pk = naming.constraint("journal_pk");
267        let un = naming.constraint("journal_un");
268        let seqnr_idx = naming.index("journal_seqnr_idx");
269        let stream_idx = naming.index("journal_stream_idx");
270        let tenant_idx = naming.index("journal_tenant_idx");
271        let tenant = tenant_column("stream");
272        vec![
273            format!(
274                "CREATE TABLE IF NOT EXISTS {t} (\n  id uuid NOT NULL,\n  \"time\" timestamptz NOT NULL,\n  seqnr bigserial NOT NULL,\n  \"version\" int8 NOT NULL,\n  stream text NOT NULL,\n  payload {payload_type} NOT NULL,\n{tenant},\n  CONSTRAINT {pk} PRIMARY KEY (id),\n  CONSTRAINT {un} UNIQUE (stream, version)\n);"
275            ),
276            format!("CREATE INDEX IF NOT EXISTS {seqnr_idx} ON {t} USING btree (seqnr);"),
277            format!(
278                "CREATE INDEX IF NOT EXISTS {stream_idx} ON {t} USING btree (stream, version);"
279            ),
280            format!(
281                "CREATE INDEX IF NOT EXISTS {tenant_idx} ON {t} USING btree (tenant_id, seqnr);"
282            ),
283        ]
284    }

331    /// RLS enabled **and forced**, the three policies and the grants of
332    /// each of the four event-sourced tables. Not idempotent (`CREATE
333    /// POLICY`): run once from a migration tool, as the table owner.
334    ///
335    /// Per table: `<table>_tenant_policy` (the tenant role, `USING` and
336    /// `WITH CHECK` on `tenant_id = NULLIF(current_setting(setting, true),
337    /// '')`, so a missing or empty tenant matches nothing),
338    /// `<table>_system_policy` (the system role, everything) and
339    /// `<table>_inspection_policy` (the inspection role, `SELECT` only).
340    /// Privileges are the minimum each role needs: the tenant role cannot
341    /// update or delete journal, outbox or command rows, and the
342    /// inspection role has `SELECT` only, so its writes fail on privileges
343    /// before RLS.
344    pub fn es_rls_statements<T>(naming: &PGNaming, policy: &RlsAccessPolicy<T>) -> Vec<String> {
345        let tenant_role = policy.tenant_role();
346        let system_role = policy.system_role();
347        let inspection_role = policy.inspection_role();
348        let tenant_check = format!(
349            "tenant_id = NULLIF(current_setting('{}', true), '')",
350            policy.tenant_setting()
351        );
352        // (table, tenant privileges, sequence)
353        let tables: [(&str, &str, Option<&str>); 4] = [
354            ("journal", "SELECT, INSERT", Some("journal_seqnr_seq")),
355            ("outbox", "SELECT, INSERT", Some("outbox_seqnr_seq")),
356            ("commands", "SELECT, INSERT", None),
357            ("snapshots", "SELECT, INSERT, UPDATE", None),
358        ];
359        let mut out = Vec::new();
360        if naming.needs_schema_setup() {
361            // Table privileges are useless without the right to look names
362            // up in the schema.
363            out.push(format!(
364                "GRANT USAGE ON SCHEMA \"{}\" TO {tenant_role}, {system_role}, {inspection_role};",
365                naming.namespace()
366            ));
367        }
368        for (name, tenant_privileges, sequence) in tables {
369            let t = naming.table(name);
370            let tenant_policy = naming.constraint(&format!("{name}_tenant_policy"));
371            let system_policy = naming.constraint(&format!("{name}_system_policy"));
372            let inspection_policy = naming.constraint(&format!("{name}_inspection_policy"));
373            out.push(format!("ALTER TABLE {t} ENABLE ROW LEVEL SECURITY;"));
374            out.push(format!("ALTER TABLE {t} FORCE ROW LEVEL SECURITY;"));
375            out.push(format!(
376                "CREATE POLICY {tenant_policy} ON {t} FOR ALL TO {tenant_role}\n  USING ({tenant_check})\n  WITH CHECK ({tenant_check});"
377            ));
378            out.push(format!(
379                "CREATE POLICY {system_policy} ON {t} FOR ALL TO {system_role}\n  USING (true)\n  WITH CHECK (true);"
380            ));
381            out.push(format!(
382                "CREATE POLICY {inspection_policy} ON {t} FOR SELECT TO {inspection_role}\n  USING (true);"
383            ));
384            out.push(format!(
385                "GRANT {tenant_privileges} ON {t} TO {tenant_role};"
386            ));
387            out.push(format!(
388                "GRANT SELECT, INSERT, UPDATE, DELETE ON {t} TO {system_role};"
389            ));
390            out.push(format!("GRANT SELECT ON {t} TO {inspection_role};"));
391            if let Some(seq) = sequence {
392                let seq = naming.table(seq);
393                out.push(format!(
394                    "GRANT USAGE ON SEQUENCE {seq} TO {tenant_role}, {system_role};"
395                ));
396            }
397        }
398        out
399    }
