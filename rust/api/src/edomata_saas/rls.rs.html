10/// The separator between the tenant and the rest of a tenant-scoped
11/// stream or command id: `{tenant}/{id}`. The generated `tenant_id` column
12/// of the event-sourced tables is the part before the first separator, so
13/// a tenant key can never contain it.
14pub const TENANT_SEPARATOR: char = '/';
15
16/// A tenant identifier as the database sees it. The tenant id **type** is
17/// the one thing a project chooses (a string, a UUID...): implement this
18/// trait for it. [`TenantId`], `String` and `&str` already implement it.
19///
20/// The key is validated by [`RlsAccessPolicy::tenant_access`] (not
21/// empty, no `/`, no NUL), so an implementation need not check it.
22///
23/// ```
24/// use edomata_saas::TenantKey;
25///
26/// struct OrgNumber(u64);
27///
28/// impl TenantKey for OrgNumber {
29///     fn tenant_key(&self) -> String {
30///         format!("org{}", self.0)
31///     }
32/// }
33/// assert_eq!(OrgNumber(7).tenant_key(), "org7");
34/// ```
35pub trait TenantKey {
36    /// The text stored in the `tenant_id` column and set in the tenant
37    /// setting of the transaction.
38    fn tenant_key(&self) -> String;
39}
40
41impl TenantKey for TenantId {
42    fn tenant_key(&self) -> String {
43        self.value().to_owned()
44    }
45}
46
47impl TenantKey for String {
48    fn tenant_key(&self) -> String {
49        self.clone()
50    }
51}
52
53impl TenantKey for &str {
54    fn tenant_key(&self) -> String {
55        (*self).to_owned()
56    }
57}

59/// Why a tenant key is refused.
60#[derive(Clone, Copy, Debug, PartialEq, Eq)]
61pub enum TenantKeyError {
62    /// The key is empty: an empty tenant denies everything, so it is never
63    /// a valid tenant.
64    Empty,
65    /// The key contains `/`, which separates the tenant from the id in
66    /// `{tenant}/{id}`. A tenant `a/b` would be read by the database as
67    /// tenant `a`: refused.
68    ContainsSeparator,
69    /// The key contains a NUL character, which PostgreSQL text cannot hold.
70    ContainsNul,
71}
72
73impl fmt::Display for TenantKeyError {
74    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
75        match self {
76            TenantKeyError::Empty => f.write_str("the tenant id is empty"),
77            TenantKeyError::ContainsSeparator => {
78                f.write_str("the tenant id contains '/', the separator of `{tenant}/{id}`")
79            }
80            TenantKeyError::ContainsNul => f.write_str("the tenant id contains a NUL character"),
81        }
82    }
83}
84
85impl std::error::Error for TenantKeyError {}

87/// Checks a tenant key: not empty, no `/`, no NUL.
88///
89/// # Errors
90///
91/// A [`TenantKeyError`] naming the first problem.
92///
93/// ```
94/// use edomata_saas::{TenantKeyError, validate_tenant_key};
95///
96/// assert_eq!(validate_tenant_key("acme"), Ok(()));
97/// assert_eq!(validate_tenant_key(""), Err(TenantKeyError::Empty));
98/// assert_eq!(validate_tenant_key("a/b"), Err(TenantKeyError::ContainsSeparator));
99/// ```
100pub fn validate_tenant_key(key: &str) -> Result<(), TenantKeyError> {
101    if key.is_empty() {
102        Err(TenantKeyError::Empty)
103    } else if key.contains(TENANT_SEPARATOR) {
104        Err(TenantKeyError::ContainsSeparator)
105    } else if key.contains('\0') {
106        Err(TenantKeyError::ContainsNul)
107    } else {
108        Ok(())
109    }
110}

112/// Why an [`RlsAccessPolicy`] is refused.
113#[derive(Clone, Debug, PartialEq, Eq)]
114pub enum RlsPolicyError {
115    /// A role, prefix or login name is not a plain lowercase PostgreSQL
116    /// identifier (`[a-z_][a-z0-9_]*`, at most 63 characters).
117    InvalidIdentifier(String),
118    /// The tenant setting is not `namespace.name` of two such identifiers
119    /// (a custom setting needs the dot).
120    InvalidSetting(String),
121    /// Two of the three roles have the same name.
122    DuplicateRole(String),
123}
124
125impl fmt::Display for RlsPolicyError {
126    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
127        match self {
128            RlsPolicyError::InvalidIdentifier(s) => write!(
129                f,
130                "`{s}` is not a lowercase PostgreSQL identifier ([a-z_][a-z0-9_]*, at most 63 characters)"
131            ),
132            RlsPolicyError::InvalidSetting(s) => write!(
133                f,
134                "`{s}` is not a `namespace.name` setting of two lowercase identifiers"
135            ),
136            RlsPolicyError::DuplicateRole(s) => {
137                write!(f, "the role `{s}` is used for two access modes")
138            }
139        }
140    }
141}

143impl std::error::Error for RlsPolicyError {}

145fn is_identifier(s: &str) -> bool {
146    let mut chars = s.chars();
147    matches!(chars.next(), Some(c) if c.is_ascii_lowercase() || c == '_')
148        && s.len() <= 63
149        && chars.all(|c| c.is_ascii_lowercase() || c.is_ascii_digit() || c == '_')
150}

152/// Validates a PostgreSQL role name for use in generated SQL.
153///
154/// # Errors
155///
156/// [`RlsPolicyError::InvalidIdentifier`].
157pub(crate) fn check_identifier(s: &str) -> Result<(), RlsPolicyError> {
158    if is_identifier(s) {
159        Ok(())
160    } else {
161        Err(RlsPolicyError::InvalidIdentifier(s.to_owned()))
162    }
163}

165/// The three ways to reach the event-sourced tables, each a PostgreSQL
166/// role.
167#[derive(Clone, Copy, Debug, PartialEq, Eq, Hash)]
168pub enum AccessMode {
169    /// Reads and writes the rows of one tenant (`USING` and `WITH CHECK`).
170    Tenant,
171    /// Reads and writes every row: relays, data migrations, system jobs.
172    System,
173    /// Reads every row, writes none: an inspection agent.
174    Inspection,
175}
176
177/// The access policy of a SaaS event-sourced service: the PostgreSQL
178/// roles of the three [`AccessMode`]s and the name of the transaction-local
179/// setting that carries the tenant. Required to build a
180/// [`SaaSEventSourcedService`](crate::SaaSEventSourcedService); only the
181/// tenant id type `T` is left to the project.
182///
183/// The roles are `NOLOGIN` roles that own nothing. A connection reaches
184/// one of them with `SET LOCAL ROLE`, which PostgreSQL only allows for a
185/// role the connection's login is a **member** of: that membership, not a
186/// setting the application can freely `SET`, is the boundary in front of
187/// the system and inspection modes (see
188/// [`SaaSPGSchema::rls_memberships`](crate::SaaSPGSchema::rls_memberships)).
189///
190/// ```
191/// use edomata_saas::{AccessMode, RlsAccessPolicy, TenantId};
192///
193/// let policy: RlsAccessPolicy<TenantId> = RlsAccessPolicy::with_prefix("shop").unwrap();
194/// assert_eq!(policy.tenant_role(), "shop_tenant");
195/// assert_eq!(policy.tenant_setting(), "shop.tenant_id");
196///
197/// let access = policy.tenant_access(&TenantId::new("acme")).unwrap();
198/// assert_eq!(access.mode(), AccessMode::Tenant);
199/// assert_eq!(access.tenant(), Some("acme"));
200/// assert_eq!(policy.scoped_id(&TenantId::new("acme"), "order-1").unwrap(), "acme/order-1");
201///
202/// // A tenant that contains the separator is refused.
203/// assert!(policy.tenant_access(&TenantId::new("acme/evil")).is_err());
204/// ```
205pub struct RlsAccessPolicy<T = TenantId> {
206    tenant_role: String,
207    system_role: String,
208    inspection_role: String,
209    tenant_setting: String,
210    tenant_type: PhantomData<fn(&T)>,
211}

247impl<T> RlsAccessPolicy<T> {
248    /// A policy from explicit role names and tenant setting
249    /// (`namespace.name`, e.g. `app.tenant_id`).
250    ///
251    /// # Errors
252    ///
253    /// [`RlsPolicyError`] when a name is not a lowercase identifier, the
254    /// setting is not `namespace.name`, or two roles share a name.
255    pub fn new(
256        tenant_role: impl Into<String>,
257        system_role: impl Into<String>,
258        inspection_role: impl Into<String>,
259        tenant_setting: impl Into<String>,
260    ) -> Result<Self, RlsPolicyError> {
261        let policy = Self {
262            tenant_role: tenant_role.into(),
263            system_role: system_role.into(),
264            inspection_role: inspection_role.into(),
265            tenant_setting: tenant_setting.into(),
266            tenant_type: PhantomData,
267        };
268        for role in [
269            &policy.tenant_role,
270            &policy.system_role,
271            &policy.inspection_role,
272        ] {
273            check_identifier(role)?;
274        }
275        match policy.tenant_setting.split_once('.') {
276            Some((ns, name)) if is_identifier(ns) && is_identifier(name) => {}
277            _ => {
278                return Err(RlsPolicyError::InvalidSetting(policy.tenant_setting));
279            }
280        }
281        for (a, b) in [
282            (&policy.tenant_role, &policy.system_role),
283            (&policy.tenant_role, &policy.inspection_role),
284            (&policy.system_role, &policy.inspection_role),
285        ] {
286            if a == b {
287                return Err(RlsPolicyError::DuplicateRole(a.clone()));
288            }
289        }
290        Ok(policy)
291    }
292
293    /// Roles `{prefix}_tenant`, `{prefix}_system` and
294    /// `{prefix}_inspection`, and setting `{prefix}.tenant_id`.
295    ///
296    /// # Errors
297    ///
298    /// [`RlsPolicyError`] when `prefix` makes an invalid name.
299    pub fn with_prefix(prefix: &str) -> Result<Self, RlsPolicyError> {
300        check_identifier(prefix)?;
301        Self::new(
302            format!("{prefix}_tenant"),
303            format!("{prefix}_system"),
304            format!("{prefix}_inspection"),
305            format!("{prefix}.tenant_id"),
306        )
307    }
308
309    /// The same roles and setting, for another tenant id type.
310    pub fn retyped<U>(&self) -> RlsAccessPolicy<U> {
311        RlsAccessPolicy {
312            tenant_role: self.tenant_role.clone(),
313            system_role: self.system_role.clone(),
314            inspection_role: self.inspection_role.clone(),
315            tenant_setting: self.tenant_setting.clone(),
316            tenant_type: PhantomData,
317        }
318    }
319
320    /// The role of [`AccessMode::Tenant`].
321    pub fn tenant_role(&self) -> &str {
322        &self.tenant_role
323    }
324
325    /// The role of [`AccessMode::System`].
326    pub fn system_role(&self) -> &str {
327        &self.system_role
328    }
329
330    /// The role of [`AccessMode::Inspection`].
331    pub fn inspection_role(&self) -> &str {
332        &self.inspection_role
333    }
334
335    /// The role of `mode`.
336    pub fn role(&self, mode: AccessMode) -> &str {
337        match mode {
338            AccessMode::Tenant => &self.tenant_role,
339            AccessMode::System => &self.system_role,
340            AccessMode::Inspection => &self.inspection_role,
341        }
342    }
343
344    /// The transaction-local setting that carries the tenant.
345    pub fn tenant_setting(&self) -> &str {
346        &self.tenant_setting
347    }
348
349    /// The system mode (full access). Only usable on a connection whose
350    /// login is a member of the system role.
351    pub fn system_access(&self) -> Access {
352        self.access(AccessMode::System, None)
353    }
354
355    /// The inspection mode (read-only across tenants). Only usable on a
356    /// connection whose login is a member of the inspection role.
357    pub fn inspection_access(&self) -> Access {
358        self.access(AccessMode::Inspection, None)
359    }
360
361    fn access(&self, mode: AccessMode, tenant: Option<String>) -> Access {
362        Access {
363            mode,
364            role: self.role(mode).to_owned(),
365            tenant_setting: self.tenant_setting.clone(),
366            tenant,
367        }
368    }
369}

371impl<T: TenantKey> RlsAccessPolicy<T> {
372    /// The tenant mode for `tenant`.
373    ///
374    /// # Errors
375    ///
376    /// [`TenantKeyError`] when the key is empty or contains `/` or NUL.
377    pub fn tenant_access(&self, tenant: &T) -> Result<Access, TenantKeyError> {
378        let key = tenant.tenant_key();
379        validate_tenant_key(&key)?;
380        Ok(self.access(AccessMode::Tenant, Some(key)))
381    }
382
383    /// The tenant-scoped id `{tenant}/{id}` of a stream or a command: what
384    /// the generated `tenant_id` column is derived from.
385    ///
386    /// # Errors
387    ///
388    /// [`TenantKeyError`] when the tenant key is invalid.
389    pub fn scoped_id(&self, tenant: &T, id: &str) -> Result<String, TenantKeyError> {
390        let key = tenant.tenant_key();
391        validate_tenant_key(&key)?;
392        Ok(format!("{key}{TENANT_SEPARATOR}{id}"))
393    }
394}

396/// One way of reaching the database, produced by an [`RlsAccessPolicy`]
397/// (so a tenant is always validated): the role to `SET LOCAL ROLE` to, and
398/// the tenant to inject. The values cannot be built any other way.
399#[derive(Clone, Debug, PartialEq, Eq)]
400pub struct Access {
401    mode: AccessMode,
402    role: String,
403    tenant_setting: String,
404    tenant: Option<String>,
405}
406
407impl Access {
408    /// The mode.
409    pub fn mode(&self) -> AccessMode {
410        self.mode
411    }
412
413    /// The PostgreSQL role of the mode.
414    pub fn role(&self) -> &str {
415        &self.role
416    }
417
418    /// The tenant (tenant mode only).
419    pub fn tenant(&self) -> Option<&str> {
420        self.tenant.as_deref()
421    }
422
423    /// The name of the transaction-local tenant setting.
424    pub fn tenant_setting(&self) -> &str {
425        &self.tenant_setting
426    }
427
428    /// The value set in the tenant setting: the tenant, or the empty
429    /// string (which denies every row) for the system and inspection
430    /// modes.
431    pub fn tenant_setting_value(&self) -> &str {
432        self.tenant.as_deref().unwrap_or("")
433    }
434
435    /// The statement that switches the transaction to the role of the
436    /// mode. `SET LOCAL` ends with the transaction; PostgreSQL refuses it
437    /// unless the login is a member of the role. The role is a validated
438    /// identifier, safe to splice.
439    pub fn set_role_statement(&self) -> String {
440        format!("SET LOCAL ROLE {}", self.role)
441    }
