11/// A builder slot that has not been filled yet (see
12/// [`SaaSEventSourcedServiceBuilder`]).

16/// A builder slot that has been filled with a value.
17#[derive(Clone, Copy, Debug)]
18pub struct Set<V>(V);

20/// An event-sourced SaaS service: holds the guarded [`SaaSDomainDsl`], the
21/// mandatory [`RlsAccessPolicy`], and exposes the domain marker needed by
22/// `Backend::builder`. Mirrors Scala's `SaaSEventSourcedService`, where only
23/// the guarded `SaaS` DSL is reachable.
24///
25/// **It cannot be built without an access policy**: the only constructor is
26/// [`builder`](Self::builder), a typestate builder whose `build` exists once
27/// the authorization policy, the rejection mapping *and* the access policy
28/// are set. The tenant id type `T` (default [`TenantId`]) is the one thing
29/// left to the project.
30///
31/// ```
32/// use edomata_saas::*;
33///
34/// type Service = SaaSEventSourcedService<CallerIdentity, String, String, String, String, String>;
35///
36/// let service: Service = SaaSEventSourcedService::builder()
37///     .auth_policy(PermissivePolicy)
38///     .rejection(|reason| reason)
39///     .access_policy(RlsAccessPolicy::with_prefix("shop").unwrap())
40///     .build();
41/// assert_eq!(service.access_policy().tenant_role(), "shop_tenant");
42/// ```
43///
44/// Leaving the access policy out is a compile error:
45///
46/// ```compile_fail
47/// use edomata_saas::*;
48///
49/// type Service = SaaSEventSourcedService<CallerIdentity, String, String, String, String, String>;
50///
51/// // error[E0599]: no method named `build` found: `access_policy` was never called
52/// let service: Service = SaaSEventSourcedService::builder()
53///     .auth_policy(PermissivePolicy)
54///     .rejection(|reason| reason)
55///     .build();
56/// ```
57///
58/// There is no `new` that skips it either:
59///
60/// ```compile_fail
61/// use edomata_saas::*;
62///
63/// type Service = SaaSEventSourcedService<CallerIdentity, String, String, String, String, String>;
64///
65/// // error[E0599]: no function or associated item named `new` found
66/// let service: Service = SaaSEventSourcedService::new(PermissivePolicy, |reason: String| reason);
67/// ```
68pub struct SaaSEventSourcedService<Auth, C, A, E, R, N, T = TenantId> {
69    saas: SaaSDomainDsl<Auth, C, A, E, R, N>,
70    access: RlsAccessPolicy<T>,
71}

73/// The typestate builder of [`SaaSEventSourcedService`]: `build` is only
74/// available when the authorization policy, the rejection mapping and the
75/// access policy have all been set (their slots are [`Set`], not
76/// [`Missing`]).
77#[allow(clippy::type_complexity)] // the PhantomData names the service's type parameters
78pub struct SaaSEventSourcedServiceBuilder<Auth, C, A, E, R, N, T, P, F, X> {
79    auth_policy: P,
80    rejection: F,
81    access: X,
82    types: PhantomData<fn() -> (Auth, C, A, E, R, N, T)>,
83}

85impl<Auth, C, A, E, R, N, T> SaaSEventSourcedService<Auth, C, A, E, R, N, T> {
86    /// Starts building the service: set the
87    /// [`auth_policy`](SaaSEventSourcedServiceBuilder::auth_policy), the
88    /// [`rejection`](SaaSEventSourcedServiceBuilder::rejection) mapping and
89    /// the [`access_policy`](SaaSEventSourcedServiceBuilder::access_policy),
90    /// then `build`.
91    pub fn builder()
92    -> SaaSEventSourcedServiceBuilder<Auth, C, A, E, R, N, T, Missing, Missing, Missing> {
93        SaaSEventSourcedServiceBuilder {
94            auth_policy: Missing,
95            rejection: Missing,
96            access: Missing,
97            types: PhantomData,
98        }
99    }
100}

102impl<Auth, C, A, E, R, N, T, P, F, X>
103    SaaSEventSourcedServiceBuilder<Auth, C, A, E, R, N, T, P, F, X>
104{
105    /// The authorization policy (who may do what: [`AuthPolicy`]).
106    pub fn auth_policy<P2>(
107        self,
108        policy: P2,
109    ) -> SaaSEventSourcedServiceBuilder<Auth, C, A, E, R, N, T, Set<P2>, F, X>
110    where
111        P2: AuthPolicy<Auth> + 'static,
112    {
113        SaaSEventSourcedServiceBuilder {
114            auth_policy: Set(policy),
115            rejection: self.rejection,
116            access: self.access,
117            types: PhantomData,
118        }
119    }
120
121    /// The function turning guard messages into domain rejections.
122    pub fn rejection<F2>(
123        self,
124        mk_rejection: F2,
125    ) -> SaaSEventSourcedServiceBuilder<Auth, C, A, E, R, N, T, P, Set<F2>, X>
126    where
127        F2: Fn(String) -> R + Send + Sync + 'static,
128    {
129        SaaSEventSourcedServiceBuilder {
130            auth_policy: self.auth_policy,
131            rejection: Set(mk_rejection),
132            access: self.access,
133            types: PhantomData,
134        }
135    }
136
137    /// The mandatory Row-Level Security access policy.
138    #[allow(clippy::type_complexity)] // the typestate is the point
139    pub fn access_policy(
140        self,
141        policy: RlsAccessPolicy<T>,
142    ) -> SaaSEventSourcedServiceBuilder<Auth, C, A, E, R, N, T, P, F, Set<RlsAccessPolicy<T>>> {
143        SaaSEventSourcedServiceBuilder {
144            auth_policy: self.auth_policy,
145            rejection: self.rejection,
146            access: Set(policy),
147            types: PhantomData,
148        }
149    }
150}

152impl<Auth, C, A, E, R, N, T, P, F>
153    SaaSEventSourcedServiceBuilder<Auth, C, A, E, R, N, T, Set<P>, Set<F>, Set<RlsAccessPolicy<T>>>
154where
155    Auth: Clone + Send + Sync + 'static,
156    C: Clone + Send + Sync + 'static,

158    E: Clone + Send + Sync + 'static,
159    R: Clone + Send + Sync + 'static,
160    N: Clone + Send + Sync + 'static,
161    P: AuthPolicy<Auth> + 'static,
162    F: Fn(String) -> R + Send + Sync + 'static,
163{
164    /// Builds the service. Only callable once all three parts are set.
165    pub fn build(self) -> SaaSEventSourcedService<Auth, C, A, E, R, N, T> {
166        SaaSEventSourcedService {
167            saas: SaaSDomainDsl::new(self.auth_policy.0, self.rejection.0),
168            access: self.access.0,
169        }
170    }
171}
