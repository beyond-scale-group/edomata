3use crate::PGNaming;
4
5/// Default payload column type.
6pub const DEFAULT_PAYLOAD_TYPE: &str = "jsonb";
7
8/// Generates plain SQL DDL for the Edomata tables, for use with migration
9/// tools (Flyway, Liquibase...) instead of the drivers' automatic setup.
10///
11/// The output is byte-for-byte identical to Scala's `PGSchema`; the golden
12/// tests under `rust/tests/golden/` assert it. Every payload column defaults
13/// to `jsonb`. Payload type parameters are spliced into the DDL as is
14/// (`"json"`, `"jsonb"` or `"bytea"` are the intended values).
15///
16/// ```
17/// use edomata_postgres::{PGNaming, PGSchema};
18///
19/// let ddl = PGSchema::cqrs(&PGNaming::prefixed_str("auth").unwrap());
20/// assert_eq!(ddl.len(), 3); // states, outbox, commands
21/// assert!(ddl.iter().all(|s| s.ends_with(';')));
22///
23/// // Schema mode starts with `CREATE SCHEMA`; payload types can be chosen.
24/// let ddl = PGSchema::eventsourcing_with(&PGNaming::schema_str("auth").unwrap(), "bytea", "jsonb", "json");
25/// assert_eq!(ddl[0], r#"CREATE SCHEMA IF NOT EXISTS "auth";"#);
26/// assert!(ddl.iter().any(|s| s.contains(r#""auth".journal"#) && s.contains("payload bytea")));
27/// ```
28#[derive(Clone, Copy, Debug, Default)]
29pub struct PGSchema;
30
31impl PGSchema {
32    /// DDL for event-sourcing tables (journal, outbox, commands, snapshots,
33    /// migrations) with `jsonb` payloads.
34    pub fn eventsourcing(naming: &PGNaming) -> Vec<String> {
35        Self::eventsourcing_with(
36            naming,
37            DEFAULT_PAYLOAD_TYPE,
38            DEFAULT_PAYLOAD_TYPE,
39            DEFAULT_PAYLOAD_TYPE,
40        )
41    }
42
43    /// DDL for event-sourcing tables with explicit payload column types.
44    pub fn eventsourcing_with(
45        naming: &PGNaming,
46        event_type: &str,
47        notification_type: &str,
48        snapshot_type: &str,
49    ) -> Vec<String> {
50        let mut out = ddl::schema_statement(naming);
51        out.extend(ddl::journal_statements(naming, event_type));
52        out.extend(ddl::outbox_statements(naming, notification_type));
53        out.extend(ddl::commands_statements(naming));
54        out.extend(ddl::snapshots_statements(naming, snapshot_type));
55        out.extend(ddl::migrations_statements(naming));
56        out
57    }
58
59    /// DDL for the optional `relay_checkpoints` table used by
60    /// `edomata-broker`'s `JournalRelay`. Not included in
61    /// [`eventsourcing`](Self::eventsourcing) / [`cqrs`](Self::cqrs), whose
62    /// output stays byte-identical to Scala's; append it when journal
63    /// streaming is enabled.
64    pub fn relay_checkpoints(naming: &PGNaming) -> Vec<String> {
65        ddl::relay_checkpoints_statements(naming)
66    }
67
68    /// DDL for CQRS tables (states, outbox, commands) with `jsonb` payloads.
69    pub fn cqrs(naming: &PGNaming) -> Vec<String> {
70        Self::cqrs_with(naming, DEFAULT_PAYLOAD_TYPE, DEFAULT_PAYLOAD_TYPE)
71    }
72
73    /// DDL for CQRS tables with explicit payload column types.
74    pub fn cqrs_with(naming: &PGNaming, state_type: &str, notification_type: &str) -> Vec<String> {
75        let mut out = ddl::schema_statement(naming);
76        out.extend(ddl::states_statements(naming, state_type));
77        out.extend(ddl::outbox_statements(naming, notification_type));
78        out.extend(ddl::commands_statements(naming));
79        out
80    }
81}

83/// The individual DDL statements. Drivers reuse them for automatic setup so
84/// that the tables they create match the generated migrations exactly.
85pub mod ddl {
86    use crate::PGNaming;
87
88    /// `CREATE SCHEMA` in schema mode, nothing in prefix mode.
89    pub fn schema_statement(naming: &PGNaming) -> Vec<String> {
90        if naming.needs_schema_setup() {
91            vec![format!(
92                "CREATE SCHEMA IF NOT EXISTS \"{}\";",
93                naming.namespace()
94            )]
95        } else {
96            Vec::new()
97        }
98    }
99
100    /// The journal table and its indexes.
101    pub fn journal_statements(naming: &PGNaming, payload_type: &str) -> Vec<String> {
102        let t = naming.table("journal");
103        let pk = naming.constraint("journal_pk");
104        let un = naming.constraint("journal_un");
105        let seqnr_idx = naming.index("journal_seqnr_idx");
106        let stream_idx = naming.index("journal_stream_idx");
107        vec![
108            format!(
109                "CREATE TABLE IF NOT EXISTS {t} (\n  id uuid NOT NULL,\n  \"time\" timestamptz NOT NULL,\n  seqnr bigserial NOT NULL,\n  \"version\" int8 NOT NULL,\n  stream text NOT NULL,\n  payload {payload_type} NOT NULL,\n  CONSTRAINT {pk} PRIMARY KEY (id),\n  CONSTRAINT {un} UNIQUE (stream, version)\n);"
110            ),
111            format!("CREATE INDEX IF NOT EXISTS {seqnr_idx} ON {t} USING btree (seqnr);"),
112            format!(
113                "CREATE INDEX IF NOT EXISTS {stream_idx} ON {t} USING btree (stream, version);"
114            ),
115        ]
116    }
117
118    /// The outbox table.
119    pub fn outbox_statements(naming: &PGNaming, payload_type: &str) -> Vec<String> {
120        let t = naming.table("outbox");
121        let pk = naming.constraint("outbox_pk");
122        vec![format!(
123            "CREATE TABLE IF NOT EXISTS {t} (\n  seqnr bigserial NOT NULL,\n  stream text NOT NULL,\n  correlation text NULL,\n  causation text NULL,\n  payload {payload_type} NOT NULL,\n  created timestamptz NOT NULL,\n  published timestamptz NULL,\n  CONSTRAINT {pk} PRIMARY KEY (seqnr)\n);"
124        )]
125    }
126
127    /// The commands table.
128    pub fn commands_statements(naming: &PGNaming) -> Vec<String> {
129        let t = naming.table("commands");
130        let pk = naming.constraint("commands_pk");
131        vec![format!(
132            "CREATE TABLE IF NOT EXISTS {t} (\n  id text NOT NULL,\n  \"time\" timestamptz NOT NULL,\n  address text NOT NULL,\n  CONSTRAINT {pk} PRIMARY KEY (id)\n);"
133        )]
134    }
135
136    /// The snapshots table.
137    pub fn snapshots_statements(naming: &PGNaming, payload_type: &str) -> Vec<String> {
138        let t = naming.table("snapshots");
139        let pk = naming.constraint("snapshots_pk");
140        vec![format!(
141            "CREATE TABLE IF NOT EXISTS {t} (\n  id text NOT NULL,\n  \"version\" int8 NOT NULL,\n  state {payload_type} NOT NULL,\n  CONSTRAINT {pk} PRIMARY KEY (id)\n);"
142        )]
143    }
144
145    /// The migrations tracking table.
146    pub fn migrations_statements(naming: &PGNaming) -> Vec<String> {
147        let t = naming.table("migrations");
148        let pk = naming.constraint("migrations_pk");
149        vec![format!(
150            "CREATE TABLE IF NOT EXISTS {t} (\n  \"version\" text NOT NULL,\n  description text NOT NULL,\n  applied_at timestamptz NOT NULL DEFAULT now(),\n  CONSTRAINT {pk} PRIMARY KEY (\"version\")\n);"
151        )]
152    }
153
154    /// The relay checkpoints table of `edomata-broker`'s `JournalRelay`
155    /// (opt-in: not part of [`super::PGSchema::eventsourcing`], which stays
156    /// identical to Scala's output).
157    pub fn relay_checkpoints_statements(naming: &PGNaming) -> Vec<String> {
158        let t = naming.table("relay_checkpoints");
159        let pk = naming.constraint("relay_checkpoints_pk");
160        vec![format!(
161            "CREATE TABLE IF NOT EXISTS {t} (\n  relay text NOT NULL,\n  seqnr int8 NOT NULL,\n  updated_at timestamptz NOT NULL DEFAULT now(),\n  CONSTRAINT {pk} PRIMARY KEY (relay)\n);"
162        )]
163    }
164
165    /// The CQRS states table.
166    pub fn states_statements(naming: &PGNaming, payload_type: &str) -> Vec<String> {
167        let t = naming.table("states");
168        let pk = naming.constraint("states_pk");
169        vec![format!(
170            "CREATE TABLE IF NOT EXISTS {t} (\n  id text NOT NULL,\n  \"version\" int8 NOT NULL,\n  state {payload_type} NOT NULL,\n  CONSTRAINT {pk} PRIMARY KEY (id)\n);"
171        )]
172    }
