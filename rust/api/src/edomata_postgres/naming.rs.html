1//! Table naming strategies.
2
3use crate::{PGNamespace, PGNamespaceError};
4
5/// How the tables of one aggregate are named.
6///
7/// - [`PGNaming::Schema`] (the default in Scala) creates a dedicated
8///   PostgreSQL schema per aggregate, e.g. `"auth".journal`.
9/// - [`PGNaming::Prefixed`] uses a table-name prefix inside the current
10///   schema, e.g. `auth_journal`, which suits single-schema projects and
11///   Flyway migrations in `public`. Constraint and index names are prefixed
12///   too, to avoid collisions between aggregates.
13///
14/// ```
15/// use edomata_postgres::{PGNaming, PGNamespace};
16///
17/// let ns = PGNamespace::try_from("auth").unwrap();
18/// let schema = PGNaming::schema(ns.clone());
19/// assert_eq!(schema.table("journal"), "\"auth\".journal");
20/// assert_eq!(schema.constraint("journal_pk"), "journal_pk");
21/// assert!(schema.needs_schema_setup());
22///
23/// let prefixed = PGNaming::prefixed(ns);
24/// assert_eq!(prefixed.table("journal"), "auth_journal");
25/// assert_eq!(prefixed.index("journal_seqnr_idx"), "auth_journal_seqnr_idx");
26/// assert!(!prefixed.needs_schema_setup());
27/// ```
28#[derive(Clone, Debug, PartialEq, Eq, Hash)]
29pub enum PGNaming {
30    /// Schema-based naming: `"namespace".table_name`.
31    Schema(PGNamespace),
32    /// Prefix-based naming: `namespace_table_name`.
33    Prefixed(PGNamespace),
34}
35
36impl PGNaming {
37    /// Schema-based naming from a validated namespace.
38    pub fn schema(ns: PGNamespace) -> Self {
39        PGNaming::Schema(ns)
40    }
41
42    /// Prefix-based naming from a validated namespace.
43    pub fn prefixed(ns: PGNamespace) -> Self {
44        PGNaming::Prefixed(ns)
45    }
46
47    /// Schema-based naming, validating the namespace.
48    ///
49    /// # Errors
50    ///
51    /// Returns a [`PGNamespaceError`] if `ns` is not a valid
52    /// [`PGNamespace`].
53    pub fn schema_str(ns: &str) -> Result<Self, PGNamespaceError> {
54        PGNamespace::from_string(ns).map(PGNaming::Schema)
55    }
56
57    /// Prefix-based naming, validating the namespace.
58    ///
59    /// # Errors
60    ///
61    /// Returns a [`PGNamespaceError`] if `ns` is not a valid
62    /// [`PGNamespace`].
63    ///
64    /// ```
65    /// use edomata_postgres::PGNaming;
66    ///
67    /// assert_eq!(PGNaming::prefixed_str("orders").unwrap().table("outbox"), "orders_outbox");
68    /// assert!(PGNaming::prefixed_str("not valid").is_err());
69    /// ```
70    pub fn prefixed_str(ns: &str) -> Result<Self, PGNamespaceError> {
71        PGNamespace::from_string(ns).map(PGNaming::Prefixed)
72    }
73
74    /// The underlying namespace.
75    pub fn namespace(&self) -> &PGNamespace {
76        match self {
77            PGNaming::Schema(ns) | PGNaming::Prefixed(ns) => ns,
78        }
79    }
80
81    /// Whether a `CREATE SCHEMA` statement is needed during setup.
82    pub fn needs_schema_setup(&self) -> bool {
83        matches!(self, PGNaming::Schema(_))
84    }
85
86    /// A fully-qualified (schema mode) or prefixed (prefix mode) table
87    /// reference suitable for embedding in SQL. The schema name is quoted,
88    /// so it keeps its case; `name` is spliced as is.
89    pub fn table(&self, name: &str) -> String {
90        match self {
91            PGNaming::Schema(ns) => format!("\"{ns}\".{name}"),
92            PGNaming::Prefixed(ns) => format!("{ns}_{name}"),
93        }
94    }
95
96    /// A constraint name, prefixed in prefix mode.
97    pub fn constraint(&self, name: &str) -> String {
98        match self {
99            PGNaming::Schema(_) => name.to_owned(),
100            PGNaming::Prefixed(ns) => format!("{ns}_{name}"),
101        }
102    }
103
104    /// An index name, prefixed in prefix mode.
105    pub fn index(&self, name: &str) -> String {
106        match self {
107            PGNaming::Schema(_) => name.to_owned(),
108            PGNaming::Prefixed(ns) => format!("{ns}_{name}"),
109        }
110    }
111}
