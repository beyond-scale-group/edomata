1//! Samples of the "Distributing events with Kafka / RabbitMQ" chapter.
2//! Each publisher is behind the feature of the same name (`kafka`,
3//! `rabbitmq`), like in `edomata-examples`, so that the default build pulls
4//! no broker client; `--all-features` (clippy and tests in CI) compiles
5//! them. They need a broker, so they are compiled but not run.
6
7#[cfg(any(feature = "kafka", feature = "rabbitmq"))]
8use std::sync::Arc;
9#[cfg(any(feature = "kafka", feature = "rabbitmq"))]
10use std::time::Duration;
11
12#[cfg(any(feature = "kafka", feature = "rabbitmq"))]
13use crate::eventsourcing::{Account, Event, Notification, Rejection};
14#[cfg(any(feature = "kafka", feature = "rabbitmq"))]
15use edomata_backend::eventsourcing::Backend;
16#[cfg(any(feature = "kafka", feature = "rabbitmq"))]
17use edomata_broker::{CancellationToken, MessageEncoder, OutboxRelay, RelayConfig};
18
19#[cfg(any(feature = "kafka", feature = "rabbitmq"))]
20type AccountBackend = Backend<Account, Event, Rejection, Notification>;
21
22/// Relays the outbox to Kafka, as the leader among the replicas.
23#[cfg(feature = "kafka")]
24pub async fn kafka_relay(
25    backend: &AccountBackend,
26    pool: edomata_sqlx::PgPool,
27    bootstrap: &str,
28    cancel: CancellationToken,
29) -> Result<(), Box<dyn std::error::Error>> {
30    use edomata_broker::postgres::{LeaderLock, listen};
31    use edomata_kafka::KafkaPublisher;
32
33    // ANCHOR: kafka
34    // Idempotent producer (`enable.idempotence=true`, `acks=all`), a fixed
35    // topic (the default is one topic per source), the stream id as the
36    // partition key, the ids and metadata as headers.
37    let publisher = KafkaPublisher::builder(bootstrap)
38        .with_fixed_topic("accounts-notifications")
39        .with_config("compression.type", "lz4")
40        .build()?;
41    let relay = OutboxRelay::new(
42        Arc::clone(backend.outbox()),
43        Arc::new(publisher),
44        MessageEncoder::<Notification>::serde(),
45        RelayConfig::new("kafka_accounts").with_poll_interval(Duration::from_secs(10)),
46    )
47    .wake_on(backend.updates().outbox()) // writes of this process
48    .wake_on(listen(pool.clone(), "accounts_outbox")); // writes of other processes
49    relay
50        .run_as_leader(LeaderLock::new(pool, "kafka_accounts"), cancel)
51        .await?;
52    // ANCHOR_END: kafka
53    Ok(())
54}

56/// Relays the outbox to a RabbitMQ topic exchange.
57#[cfg(feature = "rabbitmq")]
58pub async fn rabbitmq_relay(
59    backend: &AccountBackend,
60    url: &str,
61    cancel: CancellationToken,
62) -> Result<(), Box<dyn std::error::Error>> {
63    use edomata_rabbitmq::RabbitMqPublisher;
64    use edomata_rabbitmq::lapin::ExchangeKind;
65
66    // ANCHOR: rabbitmq
67    // Publisher confirms, persistent messages, `message_id` set; a durable
68    // topic exchange with the stream id as routing key.
69    let publisher = RabbitMqPublisher::connect(url)
70        .await?
71        .with_fixed_exchange("accounts-notifications");
72    publisher
73        .declare_exchange("accounts-notifications", ExchangeKind::Topic)
74        .await?;
75    let relay = OutboxRelay::new(
76        Arc::clone(backend.outbox()),
77        Arc::new(publisher),
78        MessageEncoder::<Notification>::serde(),
79        RelayConfig::new("rabbitmq_accounts").with_poll_interval(Duration::from_secs(10)),
80    )
81    .wake_on(backend.updates().outbox());
82    relay.run(cancel).await?;
83    // ANCHOR_END: rabbitmq
84    Ok(())
85}
