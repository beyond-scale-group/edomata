3use edomata_backend::{BackendError, CommandResult};

5// ANCHOR: classify
6/// How an application maps the outcome of a command, for instance to HTTP
7/// status codes.
8pub fn status_code<R: std::fmt::Debug>(result: &CommandResult<R>) -> u16 {
9    match result {
10        // Accepted, indecisive, or a redundant (already handled) command.
11        Ok(Ok(())) => 200,
12        // A business rejection: expected, not an error.
13        Ok(Err(_reasons)) => 422,
14        // Concurrent writers kept winning for every attempt of the retry
15        // policy: retrying later with the same command id is safe.
16        Err(BackendError::MaxRetryExceeded) => 409,
17        // Only raised by direct storage calls: compiled services retry it
18        // and report `MaxRetryExceeded` instead.
19        Err(BackendError::VersionConflict) => 409,
20        // A codec or invariant failure: needs a fix, not a retry.
21        Err(BackendError::PersistenceError(_)) => 500,
22        // A driver error (connection, SQL...), wrapped with its source.
23        Err(BackendError::UnknownError(_)) => 503,
24    }
25}

27/// The `sqlx` error behind an `UnknownError`, for logs and alerts.
28pub fn sqlx_cause(error: &BackendError) -> Option<&sqlx::Error> {
29    match error {
30        BackendError::UnknownError(source) => source.downcast_ref::<sqlx::Error>(),
31        _ => None,
32    }
33}

38    use super::*;
39    use crate::eventsourcing::Event;

41    #[test]
42    fn outcomes_map_to_status_codes() {
43        assert_eq!(status_code::<()>(&Ok(Ok(()))), 200);
44        assert_eq!(status_code::<()>(&Err(BackendError::MaxRetryExceeded)), 409);
45        let wrapped = BackendError::unknown(sqlx::Error::PoolTimedOut);
46        assert!(matches!(
47            sqlx_cause(&wrapped),
48            Some(sqlx::Error::PoolTimedOut)
49        ));
50        assert_eq!(status_code::<()>(&Err(wrapped)), 503);
51    }

53    #[test]
54    fn codec_errors_become_persistence_errors() {
55        // ANCHOR: codec_error
56        use edomata_backend::{Codec, CodecError};
57        use edomata_serde::SerdeCodec;
58
59        // A payload written by another version of the code (or by Scala with
60        // another JSON shape) that `Event` cannot read.
61        let codec = SerdeCodec::<Event>::jsonb();
62        let error = codec.decode(br#"{"Opened":{}}"#).unwrap_err();
63        assert!(matches!(error, CodecError::Decode(_)));
64        // Reading the journal reports it as a `PersistenceError`.
65        let backend_error = BackendError::from(error);
66        assert!(matches!(
67            backend_error,
68            BackendError::PersistenceError(ref message) if message.starts_with("decoding failed")
69        ));
70        // `Opened` is a unit variant, serialized as a bare string.
71        assert_eq!(codec.encode(&Event::Opened).unwrap(), br#""Opened""#);
72        // ANCHOR_END: codec_error
73    }
