3use edomata_postgres::{EventMigration, PGNaming};
4use serde::{Deserialize, Serialize};
5
6// ANCHOR: versions
7/// Old event type, kept so that the compiler checks the migration is total.
8pub mod v1 {
9    use super::*;
10
11    #[derive(Clone, Debug, PartialEq, Eq, Serialize, Deserialize)]
12    pub enum Event {
13        Created { name: String },
14        PriceUpdated { price: i64 },
15    }
16}
17
18/// New event type.
19pub mod v2 {
20    use super::*;
21
22    #[derive(Clone, Debug, PartialEq, Eq, Serialize, Deserialize)]
23    pub enum Event {
24        Created { name: String },
25        PriceUpdated { price: i64, currency: String },
26    }
27}
28// ANCHOR_END: versions
29
30// ANCHOR: migration
31/// Migration 001: `PriceUpdated` gains a `currency`, defaulting to "USD".
32/// `transform` is a `match` over `v1::Event`: forgetting a variant does not compile.
33pub fn v1_to_v2() -> EventMigration {
34    EventMigration::typed(
35        "001",
36        "Add currency to PriceUpdated",
37        |raw| serde_json::from_str::<v1::Event>(raw).map_err(|e| e.to_string()),
38        |event| match event {
39            v1::Event::Created { name } => v2::Event::Created { name },
40            v1::Event::PriceUpdated { price } => v2::Event::PriceUpdated {
41                price,
42                currency: "USD".to_string(),
43            },
44        },
45        |event| serde_json::to_string(event).expect("events serialize"),
46    )
47}
48// ANCHOR_END: migration
49
50// ANCHOR: chaining
51pub fn all_migrations() -> Vec<EventMigration> {
52    vec![v1_to_v2(), v2_to_v3()]
53}
54
55pub fn v2_to_v3() -> EventMigration {
56    // Operating on the JSON text directly (no compile-time exhaustivity).
57    EventMigration::new("002", "Add description to Created", |raw| {
58        let mut value: serde_json::Value = serde_json::from_str(raw).map_err(|e| e.to_string())?;
59        if let Some(created) = value.get_mut("Created") {
60            created["description"] = serde_json::Value::String(String::new());
61        }
62        Ok(value.to_string())
63    })
64}
65
66/// Two migrations composed into one.
67pub fn v1_to_v3() -> EventMigration {
68    v1_to_v2().and_then(v2_to_v3())
69}
70// ANCHOR_END: chaining

72// ANCHOR: running
73/// Runs the migrations once the tables exist and before any command is
74/// handled: idempotent, safe on every start-up.
75pub async fn run_migrations(
76    pool: edomata_sqlx::PgPool,
77) -> Result<(), edomata_backend::BackendError> {
78    use edomata_backend::BackendError;
79    use edomata_postgres::PGSchema;
80    use edomata_sqlx::{SqlxDriver, SqlxMigrations};
81
82    let naming = PGNaming::prefixed_str("products").map_err(BackendError::persistence)?;
83    // 1. The journal and snapshots tables must exist: here from `PGSchema`,
84    //    as Flyway would create them.
85    for statement in PGSchema::eventsourcing(&naming) {
86        sqlx::query(&statement)
87            .execute(&pool)
88            .await
89            .map_err(BackendError::unknown)?;
90    }
91    // 2. The pending migrations rewrite the journal.
92    let result = SqlxMigrations::run(&naming, &pool, &all_migrations()).await?;
93    println!(
94        "Applied: {:?}, skipped: {:?}",
95        result.applied, result.skipped
96    );
97    // 3. Then the driver (no DDL needed any more) and the backend, with the
98    //    latest event codec only.
99    let _driver = SqlxDriver::new_with(naming, pool, true).await?;
100    Ok(())
101}

104#[cfg(test)]
105mod tests {
106    use super::*;
107
108    #[test]
109    fn migrations_rewrite_payloads() {
110        // ANCHOR: migration_tests
111        let migrated = v1_to_v2()
112            .run(r#"{"PriceUpdated":{"price":1999}}"#)
113            .unwrap();
114        assert_eq!(
115            migrated,
116            r#"{"PriceUpdated":{"price":1999,"currency":"USD"}}"#
117        );
118        let created = v1_to_v3().run(r#"{"Created":{"name":"Widget"}}"#).unwrap();
119        assert_eq!(created, r#"{"Created":{"description":"","name":"Widget"}}"#);
120        assert!(v1_to_v2().run("not json").is_err());
121        // ANCHOR_END: migration_tests
122    }
