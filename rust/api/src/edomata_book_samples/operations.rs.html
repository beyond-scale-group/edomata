22// ANCHOR: flyway_file
23/// The content of a Flyway migration (`V1__create_accounts_tables.sql`):
24/// every statement of `PGSchema`, each ending with a semicolon.
25pub fn flyway_migration(naming: &PGNaming) -> String {
26    PGSchema::eventsourcing(naming)
27        .iter()
28        .map(|statement| format!("{}\n", statement.trim_end()))
29        .collect()
30}

33// ANCHOR: startup
34/// Start-up order of a writer process whose tables come from Flyway.
35pub async fn start_writer(pool: PgPool) -> Result<AccountBackend, BackendError> {
36    let naming = PGNaming::prefixed_str("accounts").map_err(BackendError::persistence)?;
37    // 1. The driver executes no DDL: Flyway created the tables.
38    let driver = SqlxDriver::new_with(naming.clone(), pool.clone(), true)
39        .await?
40        // Wake relays running in other processes (see below).
41        .with_outbox_notify_channel("accounts_outbox");
42    // 2. Pending event migrations, on the existing journal, before any
43    //    command is handled.
44    SqlxMigrations::run(&naming, &pool, &crate::migrations::all_migrations()).await?;
45    // 3. The backend.
46    Backend::builder(AccountModel, AccountModel.dsl::<Command, Notification>())
47        .driver(driver)
48        .persisted_snapshot(SqlxCodec::<Account>::jsonb())
49        .build_default()
50        .await
51}
52// ANCHOR_END: startup

54// ANCHOR: relay_process
55/// A dedicated relay process: every replica runs this function, one of them
56/// wins the advisory lock and publishes, the others stand by.
57pub async fn run_outbox_relay(
58    pool: PgPool,
59    publisher: Arc<dyn Publisher>,
60    cancel: CancellationToken,
61) -> Result<(), RelayError> {
62    let naming = PGNaming::prefixed_str("accounts").map_err(BackendError::persistence)?;
63    // A reader-only backend: `skip_setup`, since the writer owns the schema.
64    let driver = SqlxDriver::new_with(naming, pool.clone(), true).await?;
65    let backend: AccountBackend =
66        Backend::builder(AccountModel, AccountModel.dsl::<Command, Notification>())
67            .driver(driver)
68            .build_default()
69            .await?;
70
71    let relay = OutboxRelay::new(
72        Arc::clone(backend.outbox()),
73        publisher,
74        MessageEncoder::<Notification>::serde(),
75        RelayConfig::new("accounts")
76            .with_batch_size(200)
77            .with_poll_interval(Duration::from_secs(10))
78            .with_retry(RetryPolicy {
79                initial_delay: Duration::from_millis(200),
80                max_delay: Duration::from_secs(30),
81                max_retries: Some(20), // then stop with an error, and restart
82            }),
83    )
84    // Woken by the writers' `NOTIFY accounts_outbox`; polling is the fallback.
85    .wake_on(listen(pool.clone(), "accounts_outbox"));
86
87    // Export the counters; they are also emitted as `tracing` events.
88    let metrics = relay.metrics();
89    let reporter = cancel.clone();
90    tokio::spawn(async move {
91        while !reporter.is_cancelled() {
92            tokio::time::sleep(Duration::from_secs(30)).await;
93            let m = metrics.snapshot();
94            tracing::info!(
95                published = m.published,
96                retried = m.retried,
97                failed = m.failed,
98                lag = m.lag,
99                "outbox relay"
100            );
101        }
102    });
103
104    // Use the relay source as the lock key, so replicas share one lock.
105    relay
106        .run_as_leader(LeaderLock::new(pool, "accounts"), cancel)
107        .await
108}

111// ANCHOR: journal_relay
112/// Streaming the journal from a checkpoint stored in PostgreSQL.
113pub async fn run_journal_relay(
114    backend: &AccountBackend,
115    pool: PgPool,
116    publisher: Arc<dyn Publisher>,
117    cancel: CancellationToken,
118) -> Result<(), RelayError> {
119    let naming = PGNaming::prefixed_str("accounts").map_err(BackendError::persistence)?;
120    // `relay_checkpoints` table; or `PGSchema::relay_checkpoints` in Flyway.
121    let checkpoints = PgCheckpointStore::new(pool.clone(), naming);
122    checkpoints.setup().await?;
123    let relay = JournalRelay::new(
124        Arc::clone(backend.journal()),
125        Arc::new(checkpoints),
126        publisher,
127        MessageEncoder::<Event>::serde(),
128        RelayConfig::new("accounts"),
129    )
130    .wake_on(backend.updates().journal());
131    relay
132        .run_as_leader(LeaderLock::with_key(pool, "accounts:journal"), cancel)
133        .await
134}

137// ANCHOR: dedup
138/// Consumer side: at-least-once delivery means duplicates. Record the
139/// message id (the `edomata-id` header, or `message_id` with RabbitMQ) in
140/// the same transaction as the effect, and skip ids already seen.
141pub async fn handle_once(
142    pool: &PgPool,
143    message_id: &str,
144    payload: &[u8],
145) -> Result<bool, sqlx::Error> {
146    let mut tx = pool.begin().await?;
147    let inserted =
148        sqlx::query("INSERT INTO processed_messages (id) VALUES ($1) ON CONFLICT (id) DO NOTHING")
149            .bind(message_id)
150            .execute(&mut *tx)
151            .await?
152            .rows_affected();
153    if inserted == 0 {
154        return Ok(false); // a redelivery: already handled
155    }
156    sqlx::query("INSERT INTO account_feed (message_id, payload) VALUES ($1, $2)")
157        .bind(message_id)
158        .bind(payload)
159        .execute(&mut *tx)
160        .await?;
161    tx.commit().await?;
162    Ok(true)
163}

166// ANCHOR: rls
167/// Multi-tenant tables with Row-Level Security, for Flyway.
168pub fn saas_ddl() -> Result<Vec<String>, edomata_postgres::PGNamespaceError> {
169    use edomata_saas::{RlsConfig, SaaSPGSchema};
170    let naming = PGNaming::prefixed_str("todos")?;
171    // Enables RLS on the states and outbox tables, adds a policy comparing
172    // `tenant_id` with the `app.tenant_id` setting, and grants the tables
173    // to the `app_reader` role.
174    Ok(SaaSPGSchema::cqrs_with(
175        &naming,
176        "jsonb",
177        "jsonb",
178        Some(&RlsConfig::new("app_reader", "app.tenant_id")),
179    ))
180}

182/// A read through the RLS-restricted role: the policy only lets the rows
183/// of the tenant set in this transaction through.
184pub async fn count_tenant_states(reader_pool: &PgPool, tenant: &str) -> Result<i64, sqlx::Error> {
185    let mut tx = reader_pool.begin().await?;
186    // `true`: the setting is local to the transaction.
187    sqlx::query("SELECT set_config('app.tenant_id', $1, true)")
188        .bind(tenant)
189        .execute(&mut *tx)
190        .await?;
191    let count = sqlx::query_scalar("SELECT count(*) FROM todos_states")
192        .fetch_one(&mut *tx)
193        .await?;
194    tx.commit().await?;
195    Ok(count)
196}

199#[cfg(test)]
200mod tests {
201    use super::*;
202
203    #[test]
204    fn flyway_file_holds_every_statement() {
205        let naming = PGNaming::prefixed_str("accounts").unwrap();
206        let sql = flyway_migration(&naming);
207        assert_eq!(
208            sql.matches("CREATE TABLE IF NOT EXISTS").count(),
209            5,
210            "journal, outbox, commands, snapshots, migrations"
211        );
212        assert!(!sql.contains("CREATE SCHEMA"), "prefixed mode");
213        assert!(sql.lines().filter(|l| l.ends_with(';')).count() >= 5);
214    }
215
216    #[test]
217    fn rls_ddl_enables_row_level_security() {
218        let ddl = saas_ddl().unwrap();
219        assert!(ddl.iter().any(|s| s.contains("ENABLE ROW LEVEL SECURITY")));
220        assert!(
221            ddl.iter()
222                .any(|s| s.contains("current_setting('app.tenant_id')"))
223        );
224    }
