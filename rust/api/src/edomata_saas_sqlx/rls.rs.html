19/// The single entry point to the event-sourced tables of a SaaS service
20/// under Row-Level Security.
21///
22/// Every transaction that touches them starts in [`begin`](Self::begin),
23/// which, inside the transaction:
24///
25/// 1. switches to the role of the access (`SET LOCAL ROLE`; PostgreSQL
26///    refuses it unless the connection's login is a **member** of that
27///    role, which is what keeps the system and inspection modes out of
28///    reach of a login that was not granted them), then
29/// 2. sets the tenant with `set_config(setting, value, true)`:
30///    **transaction-local**, so it ends with the transaction and a pooled
31///    connection never carries a tenant into its next use. A missing or
32///    empty tenant matches no row.
33///
34/// Nothing outside this type sets the role or the tenant, and nothing in
35/// the policies trusts a setting the application can change freely: the
36/// tenant setting only selects rows *within* the tenant role.
37///
38/// ```no_run
39/// # use edomata_saas::{PGNaming, RlsAccessPolicy, TenantId};
40/// # use edomata_saas_sqlx::RlsGateway;
41/// # async fn example(pool: sqlx::PgPool) -> Result<(), Box<dyn std::error::Error>> {
42/// let policy: RlsAccessPolicy<TenantId> = RlsAccessPolicy::with_prefix("shop")?;
43/// let gateway = RlsGateway::new(pool, policy.clone());
44///
45/// // Raw SQL, as tenant "acme": the transaction is committed on `Ok`.
46/// let access = policy.tenant_access(&TenantId::new("acme"))?;
47/// let visible: i64 = gateway
48///     .run(&access, |conn| {
49///         Box::pin(async move {
50///             sqlx::query_scalar("select count(*) from shop_journal")
51///                 .fetch_one(conn)
52///                 .await
53///                 .map_err(edomata_backend::BackendError::unknown)
54///         })
55///     })
56///     .await?;
57///
58/// // A backend: every transaction of its driver runs as the tenant role,
59/// // for the access of the surrounding `with_access`.
60/// let driver = gateway.driver(PGNaming::prefixed_str("shop")?).await?;
61/// # let _ = (visible, driver); Ok(()) }
62/// ```
63pub struct RlsGateway<T = TenantId> {
64    pool: PgPool,
65    policy: RlsAccessPolicy<T>,
66}

85impl<T> RlsGateway<T> {
86    /// A gateway over `pool`. The pool's login must be a member of the
87    /// roles of the modes it will use
88    /// ([`SaaSPGSchema::rls_memberships`](edomata_saas::SaaSPGSchema::rls_memberships)).
89    pub fn new(pool: PgPool, policy: RlsAccessPolicy<T>) -> Self {
90        Self { pool, policy }
91    }
92
93    /// The access policy.
94    pub fn policy(&self) -> &RlsAccessPolicy<T> {
95        &self.policy
96    }
97
98    /// The pool.
99    pub fn pool(&self) -> &PgPool {
100        &self.pool
101    }
102
103    /// Opens a transaction, switches to the role of `access` and injects
104    /// the tenant (see the [type](Self) documentation). The transaction
105    /// rolls back when dropped; commit it to keep its writes.
106    ///
107    /// # Errors
108    ///
109    /// [`BackendError::PersistenceError`] when `access` does not come from
110    /// this gateway's policy; [`BackendError::UnknownError`] when PostgreSQL
111    /// refuses (for example, the login is not a member of the role).
112    pub async fn begin(
113        &self,
114        access: &Access,
115    ) -> Result<Transaction<'static, Postgres>, BackendError> {
116        begin_access(&self.pool, &self.policy, access).await
117    }
118
119    /// Runs `f` in a transaction opened by [`begin`](Self::begin) and
120    /// closes it: committed when `f` returns `Ok`, rolled back otherwise.
121    ///
122    /// # Errors
123    ///
124    /// The errors of [`begin`](Self::begin), of `f`, and of the commit.
125    pub async fn run<F, V>(&self, access: &Access, f: F) -> Result<V, BackendError>
126    where
127        F: for<'c> FnOnce(&'c mut PgConnection) -> BoxFuture<'c, Result<V, BackendError>>,
128    {
129        let mut tx = self.begin(access).await?;
130        match f(&mut tx).await {
131            Ok(value) => {
132                tx.commit().await.map_err(map_sqlx)?;
133                Ok(value)
134            }
135            Err(e) => {
136                // Dropping would roll back too; be explicit.
137                let _ = tx.rollback().await;
138                Err(e)
139            }
140        }
141    }
142
143    /// Runs `future` with `access` as the ambient access of every driver
144    /// built by [`driver`](Self::driver) (and of
145    /// [`tx_scope`](Self::tx_scope)): the web tier calls it once per
146    /// request, with the authenticated tenant, around `service.handle(..)`.
147    /// The access is a task-local value: it does not follow
148    /// `tokio::spawn`, and a driver used outside any `with_access` fails
149    /// instead of running unscoped.
150    pub async fn with_access<Fut: Future>(&self, access: &Access, future: Fut) -> Fut::Output {
151        CURRENT.scope(access.clone(), future).await
152    }
153
154    /// The [`TxScope`] reading the ambient access set by
155    /// [`with_access`](Self::with_access).
156    pub fn tx_scope(&self) -> Arc<dyn TxScope> {
157        Arc::new(GatewayScope {
158            policy: self.policy.retyped(),
159            fixed: None,
160        })
161    }
162
163    /// A [`TxScope`] that always uses `access`, whatever the ambient
164    /// access: for the background writes of a persisted snapshot store
165    /// ([`SqlxDriver::with_snapshot_write_scope`]), which have no caller.
166    /// Using the system mode there needs the login to be a member of the
167    /// system role.
168    pub fn fixed_scope(&self, access: Access) -> Arc<dyn TxScope> {
169        Arc::new(GatewayScope {
170            policy: self.policy.retyped(),
171            fixed: Some(access),
172        })
173    }
174
175    /// An event-sourcing driver whose every transaction (journal, outbox,
176    /// commands, snapshots; reads included) goes through
177    /// [`begin`](Self::begin) for the ambient access of
178    /// [`with_access`](Self::with_access). No DDL is run: the tables
179    /// belong to the migration role ([`SaaSPGSchema::eventsourcing_rls`](edomata_saas::SaaSPGSchema::eventsourcing_rls)).
180    ///
181    /// # Errors
182    ///
183    /// Never in practice (no DDL is executed); the signature follows
184    /// [`SqlxDriver::new_with`].
185    pub async fn driver(&self, naming: PGNaming) -> Result<SqlxDriver, BackendError> {
186        Ok(SqlxDriver::new_with(naming, self.pool.clone(), true)
187            .await?
188            .with_tx_scope(self.tx_scope()))
189    }
190}

192async fn begin_access<T>(
193    pool: &PgPool,
194    policy: &RlsAccessPolicy<T>,
195    access: &Access,
196) -> Result<Transaction<'static, Postgres>, BackendError> {
197    if access.tenant_setting() != policy.tenant_setting()
198        || [
199            policy.tenant_role(),
200            policy.system_role(),
201            policy.inspection_role(),
202        ]
203        .iter()
204        .all(|role| *role != access.role())
205    {
206        return Err(BackendError::persistence(
207            "the access does not come from this gateway's access policy",
208        ));
209    }
210    let mut tx = pool.begin().await.map_err(map_sqlx)?;
211    sqlx::query(&access.set_role_statement())
212        .execute(&mut *tx)
213        .await
214        .map_err(map_sqlx)?;
215    // `true`: local to this transaction.
216    sqlx::query("select set_config($1, $2, true)")
217        .bind(access.tenant_setting())
218        .bind(access.tenant_setting_value())
219        .execute(&mut *tx)
220        .await
221        .map_err(map_sqlx)?;
222    Ok(tx)
223}

225#[derive(Debug)]
226struct GatewayScope {
227    policy: RlsAccessPolicy<()>,
228    fixed: Option<Access>,
229}
230
231#[async_trait]
232impl TxScope for GatewayScope {
233    async fn begin(&self, pool: &PgPool) -> Result<Transaction<'static, Postgres>, BackendError> {
234        let access = match &self.fixed {
235            Some(access) => access.clone(),
236            None => CURRENT.try_with(Access::clone).map_err(|_| {
237                BackendError::persistence(
238                    "no access context: run the call inside `RlsGateway::with_access`",
239                )
240            })?,
241        };
242        begin_access(pool, &self.policy, &access).await
243    }
244}
