1//! # Edomata SaaS sqlx driver
2//!
3//! A tenant-aware PostgreSQL CQRS driver, the port of `saas-skunk`. It is a
4//! drop-in replacement for `edomata_sqlx::SqlxCqrsDriver` whose `states`
5//! table carries `tenant_id` / `owner_id` columns and whose `outbox` table
6//! carries `tenant_id` (the DDL of
7//! `edomata_saas::SaaSPGSchema`), populated from the state through
8//! [`edomata_saas::TenantExtractor`], so that PostgreSQL Row-Level Security
9//! and tenant-scoped indexes can be used.
10//!
11//! ```no_run
12//! # use edomata_saas::*;
13//! # use edomata_saas_sqlx::{SaaSCodec, SaaSSqlxCqrsDriver};
14//! # use edomata_backend::cqrs::Backend;
15//! # use edomata_core::CqrsModel;
16//! # #[derive(Clone, serde::Serialize, serde::Deserialize)] struct Todo { title: String }
17//! # struct Todos;
18//! # impl CqrsModel for Todos { type State = CrudState<Todo>; type Rejection = String; fn initial(&self) -> CrudState<Todo> { CrudState::NonExistent } }
19//! # async fn example(pool: sqlx::PgPool) -> Result<(), Box<dyn std::error::Error>> {
20//! let service = SaaSCqrsService::<CallerIdentity, String, Todo, String, String>::new(PermissivePolicy, |m| m);
21//! let driver = SaaSSqlxCqrsDriver::new(PGNaming::prefixed_str("todos")?, pool).await?;
22//! let backend = Backend::builder(Todos, service.domain())
23//!     .driver(driver)
24//!     .build(SaaSCodec::jsonb_state(), SaaSCodec::jsonb_notification())
25//!     .await?;
26//! # let _ = backend; Ok(()) }
27//! ```
28//!
29//! ## Tables created by hand
30//!
31//! With `skip_setup` ([`SaaSSqlxCqrsDriver::new_with`]), generate the tables
32//! (and, optionally, the RLS policies) with
33//! [`SaaSPGSchema`](edomata_saas::SaaSPGSchema) and run them from a
34//! migration tool:
35//!
36//! ```no_run
37//! # use edomata_saas::{PGNaming, RlsConfig, SaaSPGSchema};
38//! # use edomata_saas_sqlx::SaaSSqlxCqrsDriver;
39//! # async fn example(pool: sqlx::PgPool) -> Result<(), Box<dyn std::error::Error>> {
40//! let naming = PGNaming::prefixed_str("todos")?;
41//! let rls = RlsConfig::new("app_user", "app.tenant_id");
42//! for statement in SaaSPGSchema::cqrs_with(&naming, "jsonb", "jsonb", Some(&rls)) {
43//!     println!("{statement}"); // into V1__todos.sql
44//! }
45//! let driver = SaaSSqlxCqrsDriver::new_with(naming, pool, true).await?;
46//! # let _ = driver; Ok(()) }
47//! ```
48//!
49//! ## Event-sourced services under Row-Level Security
50//!
51//! [`RlsGateway`] is the single entry point of the contract that
52//! `edomata_saas::SaaSEventSourcedService` requires: it switches the
53//! transaction to the tenant, system or inspection role (`SET LOCAL ROLE`,
54//! which needs role membership) and sets the tenant with a
55//! transaction-local `set_config`. `RlsGateway::driver` gives an
56//! `edomata_sqlx::SqlxDriver` that runs every transaction that way:
57//!
58//! ```no_run
59//! # use edomata_saas::{PGNaming, RlsAccessPolicy, TenantId};
60//! # use edomata_saas_sqlx::RlsGateway;
61//! # async fn example(pool: sqlx::PgPool) -> Result<(), Box<dyn std::error::Error>> {
62//! let policy: RlsAccessPolicy<TenantId> = RlsAccessPolicy::with_prefix("shop")?;
63//! let gateway = RlsGateway::new(pool, policy.clone());
64//! let driver = gateway.driver(PGNaming::prefixed_str("shop")?).await?;
65//! // per request: gateway.with_access(&policy.tenant_access(&tenant)?, handle(command)).await
66//! # let _ = driver; Ok(()) }
67//! ```
68//!
69//! The tables and policies come from
70//! `edomata_saas::SaaSPGSchema::eventsourcing_rls`.
71//!
72//! ## Where it fits
73//!
74//! It depends on `edomata-saas` (tenancy types, [`TenantExtractor`](edomata_saas::TenantExtractor)
75//! and the tenant-aware DDL), `edomata-backend` (the
76//! [`cqrs::StorageDriver`](edomata_backend::cqrs::StorageDriver) trait it
77//! implements), `edomata-sqlx` (shared query helpers, the outbox reader and
78//! [`SqlxHandler`]), `edomata-postgres` and `edomata-serde`. No library crate
79//! builds on it (the examples and the book samples use it); applications use it in place of `edomata-sqlx`'s
80//! CQRS driver.
81//!
82//! ## Feature flags
83//!
84//! This crate has no Cargo feature flags.
