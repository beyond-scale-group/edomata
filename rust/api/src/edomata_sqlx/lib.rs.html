1//! # Edomata sqlx driver
2//!
3//! The PostgreSQL storage driver of the Rust port, replacing both the Skunk
4//! and the Doobie drivers. It implements
5//! [`edomata_backend::eventsourcing::StorageDriver`] ([`SqlxDriver`]) and
6//! [`edomata_backend::cqrs::StorageDriver`] ([`SqlxCqrsDriver`]) on top of a
7//! [`sqlx::PgPool`], with the same tables, columns, constraints, index names
8//! and transaction semantics as the Scala drivers, so Rust and Scala
9//! services can share a database.
10//!
11//! - Payloads go through [`SqlxCodec`] (any [`edomata_backend::Codec`],
12//!   `SerdeCodec::jsonb()` by default) and are written in the column's
13//!   native wire format by `edomata_serde::pg`.
14//! - Table naming follows [`PGNaming`] (`Schema` or `Prefixed`); automatic
15//!   setup reuses the DDL of `edomata_postgres` and can be disabled with
16//!   `skip_setup`.
17//! - Optimistic concurrency and command idempotency rely on the
18//!   `(stream, version)` and command id unique constraints: a violation is
19//!   reported as `BackendError::VersionConflict`.
20//! - [`SqlxMigrations`] runs [`EventMigration`]s, tracked in the
21//!   `migrations` table.
22//!
23//! ```no_run
24//! # use edomata_core::*;
25//! # use edomata_backend::eventsourcing::Backend;
26//! # use edomata_sqlx::{PGNaming, SqlxDriver};
27//! # struct Counter;
28//! # impl DomainModel for Counter {
29//! #     type State = i32; type Event = i32; type Rejection = String;
30//! #     fn initial(&self) -> i32 { 0 }
31//! #     fn transition(&self, e: &i32, s: i32) -> Result<i32, NonEmpty<String>> { Ok(s + e) }
32//! # }
33//! # async fn example() -> Result<(), Box<dyn std::error::Error>> {
34//! let pool = sqlx::PgPool::connect("postgres://postgres:postgres@localhost/postgres").await?;
35//! let driver = SqlxDriver::new(PGNaming::prefixed_str("counters")?, pool).await?;
36//! let dsl = Counter.dsl::<i32, String>();
37//! let backend = Backend::builder(Counter, dsl)
38//!     .driver(driver)
39//!     .persisted_snapshot(Default::default())
40//!     .build_default() // jsonb serde codecs
41//!     .await?;
42//! let service = backend.compile(dsl.router(move |by| dsl.accept(by)));
43//! # let _ = service;
44//! # Ok(()) }
45//! ```
46//!
47//! ## Tables created by hand (Flyway, Liquibase...)
48//!
49//! [`SqlxDriver::new_with`] and [`SqlxCqrsDriver::new_with`] take the
50//! `skip_setup` flag of Scala's `from(naming, ..., skipSetup)`: with `true`,
51//! the driver never runs DDL and expects the tables that
52//! [`PGSchema`] describes to exist.
53//!
54//! ```no_run
55//! # use edomata_sqlx::{PGNaming, PGSchema, SqlxDriver};
56//! # async fn example(pool: sqlx::PgPool) -> Result<(), Box<dyn std::error::Error>> {
57//! let naming = PGNaming::prefixed_str("counters")?;
58//! // Paste these statements into a migration script such as `V1__counters.sql`.
59//! for statement in PGSchema::eventsourcing(&naming) {
60//!     println!("{statement}");
61//! }
62//! let driver = SqlxDriver::new_with(naming, pool, true).await?; // no DDL
63//! assert!(!driver.auto_setup());
64//! # Ok(()) }
65//! ```
66//!
67//! ## Where it fits
68//!
69//! It depends on `edomata-core` (programs), `edomata-backend` (the
70//! [`StorageDriver`](edomata_backend::eventsourcing::StorageDriver) traits
71//! it implements and the command handling built on them), `edomata-postgres`
72//! (naming and DDL) and `edomata-serde` (payload wire formats). It is used
73//! by `edomata-saas-sqlx` (through [`shared`]) and `edomata-simple`, is
74//! the storage of `edomata-e2e`, and backs the tests and examples of
75//! `edomata-broker` and the Kafka / RabbitMQ publishers.
76//!
77//! ## Feature flags
78//!
79//! This crate has no Cargo feature flags.
