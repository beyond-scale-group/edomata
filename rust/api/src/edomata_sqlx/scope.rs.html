1//! Transaction scopes: a hook that lets an application prepare every
2//! transaction the event-sourcing driver opens (for example `SET LOCAL
3//! ROLE` and a transaction-local tenant setting for Row-Level Security).
4
5use std::sync::Arc;
6
7use async_trait::async_trait;
8use edomata_backend::BackendError;
9use sqlx::pool::PoolConnection;
10use sqlx::{PgConnection, PgPool, Postgres, Transaction};
11
12use crate::error::map_sqlx;
13
14/// Prepares the transactions opened by [`SqlxDriver`](crate::SqlxDriver)
15/// once [`SqlxDriver::with_tx_scope`](crate::SqlxDriver::with_tx_scope) is
16/// set.
17///
18/// With a scope, the driver never uses a bare pooled connection: every
19/// statement (reads included) runs inside a transaction returned by
20/// [`begin`](Self::begin), which is committed (or rolled back on error) by
21/// the driver. An implementation opens the transaction and applies its
22/// transaction-local settings before returning it, and must fail (never
23/// fall back to an unprepared transaction) when it cannot.
24///
25/// Without a scope, the driver behaves exactly as before. The SaaS
26/// Row-Level Security gateway of `edomata-saas-sqlx` is the intended
27/// implementation.
28///
29/// ```
30/// use async_trait::async_trait;
31/// use edomata_backend::BackendError;
32/// use edomata_sqlx::TxScope;
33/// use sqlx::{PgPool, Postgres, Transaction};
34///
35/// #[derive(Debug)]
36/// struct ReadCommitted;
37///
38/// #[async_trait]
39/// impl TxScope for ReadCommitted {
40///     async fn begin(&self, pool: &PgPool) -> Result<Transaction<'static, Postgres>, BackendError> {
41///         let mut tx = pool.begin().await.map_err(BackendError::unknown)?;
42///         sqlx::query("set local transaction isolation level read committed")
43///             .execute(&mut *tx)
44///             .await
45///             .map_err(BackendError::unknown)?;
46///         Ok(tx)
47///     }
48/// }
49/// ```
50#[async_trait]
51pub trait TxScope: std::fmt::Debug + Send + Sync + 'static {
52    /// Opens a transaction on `pool` and prepares it.
53    ///
54    /// # Errors
55    ///
56    /// A [`BackendError`] when the transaction cannot be opened or
57    /// prepared (for example, when no access context is available).
58    async fn begin(&self, pool: &PgPool) -> Result<Transaction<'static, Postgres>, BackendError>;
59}
60
61/// A pool and an optional scope: the only way the driver reaches the
62/// database.
63#[derive(Clone, Debug)]
64pub(crate) struct Db {
65    pool: PgPool,
66    scope: Option<Arc<dyn TxScope>>,
67}
68
69impl Db {
70    pub fn new(pool: PgPool, scope: Option<Arc<dyn TxScope>>) -> Self {
71        Self { pool, scope }
72    }
73
74    pub fn pool(&self) -> &PgPool {
75        &self.pool
76    }
77
78    pub fn is_scoped(&self) -> bool {
79        self.scope.is_some()
80    }
81
82    /// A transaction (prepared by the scope, if any).
83    pub async fn begin(&self) -> Result<Transaction<'static, Postgres>, BackendError> {
84        match &self.scope {
85            None => self.pool.begin().await.map_err(map_sqlx),
86            Some(scope) => scope.begin(&self.pool).await,
87        }
88    }
89
90    /// A connection for reads: a pooled connection without a scope, a
91    /// prepared transaction with one.
92    pub async fn reader(&self) -> Result<Reader, BackendError> {
93        match &self.scope {
94            None => Ok(Reader::Conn(self.pool.acquire().await.map_err(map_sqlx)?)),
95            Some(scope) => Ok(Reader::Tx(scope.begin(&self.pool).await?)),
96        }
97    }
98}
99
100/// A connection for reads, see [`Db::reader`].
101pub(crate) enum Reader {
102    Conn(PoolConnection<Postgres>),
103    Tx(Transaction<'static, Postgres>),
104}
105
106impl Reader {
107    pub fn conn(&mut self) -> &mut PgConnection {
108        match self {
109            Reader::Conn(c) => c,
110            Reader::Tx(t) => t,
111        }
112    }
113
114    /// Ends the read (commits a scoped transaction).
115    pub async fn finish(self) -> Result<(), BackendError> {
116        match self {
117            Reader::Conn(_) => Ok(()),
118            Reader::Tx(t) => t.commit().await.map_err(map_sqlx),
119        }
120    }
121}
