3use edomata_backend::BackendError;
4
5/// PostgreSQL SQLSTATE of `unique_violation`.
6const UNIQUE_VIOLATION: &str = "23505";
7
8/// Whether a sqlx error is a PostgreSQL unique-constraint violation.
9pub fn is_unique_violation(error: &sqlx::Error) -> bool {
10    match error {
11        sqlx::Error::Database(db) => db.code().as_deref() == Some(UNIQUE_VIOLATION),
12        _ => false,
13    }
14}
15
16/// Maps a sqlx error raised while writing an aggregate: a unique violation
17/// (duplicate `(stream, version)` or command id) is a
18/// [`BackendError::VersionConflict`], as in the Scala drivers; anything else
19/// is wrapped by [`map_sqlx`].
20pub fn map_write(error: sqlx::Error) -> BackendError {
21    if is_unique_violation(&error) {
22        BackendError::VersionConflict
23    } else {
24        map_sqlx(error)
25    }
26}
27
28/// Wraps any sqlx error in [`BackendError::UnknownError`].
29pub fn map_sqlx(error: sqlx::Error) -> BackendError {
30    BackendError::unknown(error)
31}
32
33/// Checks that exactly `expected` rows were affected, like the Scala
34/// `assertInserted` helper.
35///
36/// # Errors
37///
38/// [`BackendError::PersistenceError`] when `affected != expected`.
39///
40/// ```
41/// use edomata_sqlx::shared::assert_inserted;
42///
43/// assert!(assert_inserted(1, 1).is_ok());
44/// assert!(assert_inserted(0, 1).is_err());
45/// ```
46pub fn assert_inserted(affected: u64, expected: u64) -> Result<(), BackendError> {
47    if affected == expected {
48        Ok(())
49    } else {
50        Err(BackendError::persistence(format!(
51            "expected to insert exactly {expected}, but inserted {affected}"
52        )))
53    }
54}
