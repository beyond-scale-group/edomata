1//! # Edomata Kafka
2//!
3//! A [`Publisher`] for `edomata-broker` relays backed by `rdkafka`:
4//!
5//! - idempotent producer (`enable.idempotence=true`, `acks=all`): librdkafka's
6//!   own retries never duplicate or reorder records (a batch retried by the
7//!   relay after a lost acknowledgment is redelivered, as at-least-once
8//!   requires);
9//! - the stream id is the partition key, which keeps per-stream ordering;
10//! - the topic is chosen by a function of the message (default: one topic
11//!   per relay source, i.e. per aggregate namespace);
12//! - the message id and metadata travel as Kafka headers (see
13//!   `edomata_broker::headers`).
14//!
15//! `publish` returns only after every message of the batch was
16//! acknowledged by the brokers; the relay marks outbox items as sent after
17//! that, so delivery is at-least-once and consumers deduplicate on the
18//! `edomata-id` header.
19//!
20//! Errors are classified for the relay: oversized or invalid messages,
21//! invalid topics and topic authorization failures are
22//! [`PublishError::Permanent`]; everything else (broker unavailable,
23//! timeouts, queue full...) is [`PublishError::Transient`] and retried.
24//!
25//! ## Where it fits
26//!
27//! `edomata-kafka` depends only on `edomata-core` and `edomata-broker`: it
28//! is one [`Publisher`] implementation for the broker-agnostic
29//! [`OutboxRelay`](edomata_broker::OutboxRelay) and
30//! [`JournalRelay`](edomata_broker::JournalRelay). Applications that do not
31//! use Kafka don't depend on it and don't build librdkafka. It has no Scala
32//! counterpart. The [`rdkafka`] crate is re-exported for its configuration
33//! and error types.
34//!
35//! ## Example
36//!
37//! An outbox relay publishing to one Kafka topic:
38//!
39//! ```no_run
40//! use std::sync::Arc;
41//!
42//! use edomata_backend::eventsourcing::Backend;
43//! use edomata_broker::postgres::LeaderLock;
44//! use edomata_broker::{CancellationToken, MessageEncoder, OutboxRelay, RelayConfig};
45//! use edomata_core::*;
46//! use edomata_kafka::KafkaPublisher;
47//! use edomata_sqlx::SqlxDriver;
48//!
49//! struct Counter;
50//! impl DomainModel for Counter {
51//!     type State = i32; type Event = i32; type Rejection = String;
52//!     fn initial(&self) -> i32 { 0 }
53//!     fn transition(&self, e: &i32, s: i32) -> Result<i32, NonEmpty<String>> { Ok(s + e) }
54//! }
55//!
56//! # async fn example(pool: sqlx::PgPool) -> Result<(), Box<dyn std::error::Error>> {
57//! let backend = Backend::builder(Counter, Counter.dsl::<i32, String>())
58//!     .driver(SqlxDriver::for_namespace("counters", pool.clone()).await?)
59//!     .build_default()
60//!     .await?;
61//!
62//! let publisher = KafkaPublisher::builder("localhost:9092")
63//!     .with_fixed_topic("counters-notifications")
64//!     .with_config("compression.type", "lz4")
65//!     .build()?;
66//!
67//! let relay = OutboxRelay::new(
68//!     Arc::clone(backend.outbox()),
69//!     Arc::new(publisher),
70//!     MessageEncoder::<String>::serde(),
71//!     RelayConfig::new("counters"),
72//! )
73//! .wake_on(backend.updates().outbox());
74//!
75//! relay
76//!     .run_as_leader(LeaderLock::new(pool, "counters"), CancellationToken::new())
77//!     .await?;
78//! # Ok(()) }
79//! ```
80//!
81//! ## Feature flags
82//!
83//! This crate has no Cargo feature flags of its own. librdkafka is built
84//! with the `rdkafka` features chosen by the workspace (`tokio`, `libz`);
85//! enable further ones (for example `ssl` or `gssapi`) on your own
86//! `rdkafka` dependency, which Cargo unifies with this one.

111/// Publishes broker messages to Kafka.
112///
113/// Each [`BrokerMessage`] becomes one record: the topic comes from the
114/// topic function ([`default_topic`] unless set on the builder), the key is
115/// the stream id, the payload is the encoded payload and every
116/// [`BrokerMessage::headers`] entry is a Kafka header. `publish` enqueues
117/// the whole batch, then waits for every delivery report; it fails with the
118/// first failed delivery, after which the relay republishes the batch.
119///
120/// Build it with [`KafkaPublisher::builder`]; creating the producer does not
121/// contact the brokers, so a wrong address shows up as transient publish
122/// failures.
123pub struct KafkaPublisher {
124    producer: FutureProducer,
125    topic: Arc<TopicFn>,
126    send_timeout: Duration,
127}

137/// Builds a [`KafkaPublisher`].
138///
139/// ```
140/// use std::time::Duration;
141/// use edomata_broker::{BrokerMessage, MessageKind};
142/// use edomata_kafka::KafkaPublisher;
143///
144/// // Building the producer does not connect: this runs without a broker.
145/// let publisher = KafkaPublisher::builder("localhost:9092")
146///     .with_topic(|m: &BrokerMessage| format!("{}.{}", m.source, m.kind))
147///     .with_send_timeout(Duration::from_secs(10))
148///     .with_config("linger.ms", "5")
149///     .build()
150///     .unwrap();
151///
152/// # let message = BrokerMessage {
153/// #     id: BrokerMessage::outbox_id("accounts", 1), source: "accounts".into(),
154/// #     kind: MessageKind::Notification, stream_id: "acc-1".into(), seq_nr: 1,
155/// #     time: chrono::DateTime::UNIX_EPOCH, content_type: "application/json".into(),
156/// #     payload: b"{}".to_vec(), correlation: None, causation: None,
157/// #     extra_headers: Default::default(),
158/// # };
159/// assert_eq!(publisher.topic_for(&message), "accounts.notification");
160/// ```
161pub struct KafkaPublisherBuilder {
162    config: ClientConfig,
163    topic: Arc<TopicFn>,
164    send_timeout: Duration,
165}

175impl KafkaPublisher {
176    /// A builder with the idempotent-producer settings for the given
177    /// `bootstrap.servers` (a comma-separated `host:port` list):
178    /// `enable.idempotence=true`, `acks=all`, `message.timeout.ms=30000`, the
179    /// [`default_topic`] and a 30-second send timeout.
180    pub fn builder(bootstrap_servers: &str) -> KafkaPublisherBuilder {
181        let mut config = ClientConfig::new();
182        config
183            .set("bootstrap.servers", bootstrap_servers)
184            .set("enable.idempotence", "true")
185            .set("acks", "all")
186            .set("message.timeout.ms", "30000");
187        KafkaPublisherBuilder {
188            config,
189            topic: Arc::new(default_topic),
190            send_timeout: Duration::from_secs(30),
191        }
192    }
193
194    /// The topic `message` goes to, as chosen by the topic function.
195    pub fn topic_for(&self, message: &BrokerMessage) -> String {
196        (self.topic)(message)
197    }
198
199    /// The underlying producer, for metadata queries or custom sends.
200    pub fn producer(&self) -> &FutureProducer {
201        &self.producer
202    }
203
204    /// Waits for every in-flight message to be delivered, for at most the
205    /// send timeout. Call it before shutting down; `publish` already waits
206    /// for its own batch.
207    ///
208    /// # Errors
209    ///
210    /// The [`KafkaError`] of librdkafka, for example when the timeout
211    /// expires with messages still queued.
212    pub fn flush(&self) -> Result<(), KafkaError> {
213        self.producer.flush(Timeout::After(self.send_timeout))
214    }
215}

217/// The default topic: the relay source with every character outside
218/// `[A-Za-z0-9._-]` replaced by `-` (Kafka's topic name alphabet), that is
219/// one topic per relay source, see [`sanitize_topic`].
220pub fn default_topic(message: &BrokerMessage) -> String {
221    sanitize_topic(&message.source)
222}
223
224/// Makes a string a valid Kafka topic name: every character outside
225/// `[A-Za-z0-9._-]` becomes `-`, and an empty name becomes `"edomata"`.
226/// It does not enforce Kafka's 249-character limit.
227///
228/// ```
229/// use edomata_kafka::sanitize_topic;
230///
231/// assert_eq!(sanitize_topic("accounts"), "accounts");
232/// assert_eq!(sanitize_topic("\"billing\".invoices"), "-billing-.invoices");
233/// assert_eq!(sanitize_topic(""), "edomata");
234/// ```
235pub fn sanitize_topic(name: &str) -> String {
236    let topic: String = name
237        .chars()
238        .map(|c| {
239            if c.is_ascii_alphanumeric() || matches!(c, '.' | '_' | '-') {
240                c
241            } else {
242                '-'
243            }
244        })
245        .collect();
246    if topic.is_empty() {
247        "edomata".to_string()
248    } else {
249        topic
250    }
251}

253impl KafkaPublisherBuilder {
254    /// Sets any librdkafka configuration property (see librdkafka's
255    /// `CONFIGURATION.md`), overriding the defaults. Setting
256    /// `enable.idempotence` to `false` or `acks` below `all` weakens the
257    /// ordering and durability guarantees described in the crate docs.
258    pub fn with_config(mut self, key: &str, value: &str) -> Self {
259        self.config.set(key, value);
260        self
261    }
262
263    /// Chooses the topic per message. The function is called once per
264    /// message; keep it deterministic so that a stream always goes to the
265    /// same topic (per-stream ordering holds only within one topic).
266    pub fn with_topic<F>(mut self, topic: F) -> Self
267    where
268        F: Fn(&BrokerMessage) -> String + Send + Sync + 'static,
269    {
270        self.topic = Arc::new(topic);
271        self
272    }
273
274    /// Sends every message to one topic; streams stay ordered through the
275    /// partition key.
276    pub fn with_fixed_topic(self, topic: impl Into<String>) -> Self {
277        let topic = topic.into();
278        self.with_topic(move |_| topic.clone())
279    }
280
281    /// How long to wait for a delivery acknowledgment (and for
282    /// [`KafkaPublisher::flush`]). A timeout is a transient failure. Keep it
283    /// at least as long as `message.timeout.ms`.
284    pub fn with_send_timeout(mut self, timeout: Duration) -> Self {
285        self.send_timeout = timeout;
286        self
287    }
288
289    /// Creates the producer.
290    ///
291    /// # Errors
292    ///
293    /// The [`KafkaError`] of librdkafka when the configuration is invalid.
294    pub fn build(self) -> Result<KafkaPublisher, KafkaError> {
295        let producer: FutureProducer = self.config.create()?;
296        Ok(KafkaPublisher {
297            producer,
298            topic: self.topic,
299            send_timeout: self.send_timeout,
300        })
301    }
302}

304fn headers_of(message: &BrokerMessage) -> OwnedHeaders {
305    message
306        .headers()
307        .iter()
308        .fold(OwnedHeaders::new(), |headers, (key, value)| {
309            headers.insert(Header {
310                key,
311                value: Some(value.as_str()),
312            })
313        })
314}

316fn classify(error: KafkaError) -> PublishError {
317    let permanent = matches!(
318        &error,
319        KafkaError::MessageProduction(
320            RDKafkaErrorCode::MessageSizeTooLarge
321                | RDKafkaErrorCode::InvalidMessage
322                | RDKafkaErrorCode::InvalidMessageSize
323                | RDKafkaErrorCode::InvalidTopic
324                | RDKafkaErrorCode::TopicAuthorizationFailed
325        )
326    );
327    if permanent {
328        PublishError::Permanent(Box::new(error))
329    } else {
330        PublishError::Transient(Box::new(error))
331    }
332}

334#[async_trait]
335impl Publisher for KafkaPublisher {
336    async fn publish(&self, batch: &NonEmpty<BrokerMessage>) -> Result<(), PublishError> {
337        // Enqueue every message, then await the acknowledgments in order:
338        // the idempotent producer preserves per-partition ordering, and the
339        // stream id is the partition key.
340        let topics: Vec<String> = batch.iter().map(|m| (self.topic)(m)).collect();
341        let mut deliveries = Vec::with_capacity(batch.len());
342        for (message, topic) in batch.iter().zip(&topics) {
343            let record = FutureRecord::to(topic)
344                .key(message.stream_id.as_str())
345                .payload(message.payload.as_slice())
346                .headers(headers_of(message));
347            deliveries.push(
348                self.producer
349                    .send(record, Timeout::After(self.send_timeout)),
350            );
351        }
352        for delivery in deliveries {
353            let delivered = delivery.await.map_err(|(e, _)| classify(e))?;
354            tracing::trace!(
355                partition = delivered.partition,
356                offset = delivered.offset,
357                "kafka delivery acknowledged"
358            );
359        }
360        Ok(())
361    }
