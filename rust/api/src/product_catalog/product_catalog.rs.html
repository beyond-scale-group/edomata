1//! Port of `examples/src/main/scala/ProductCatalogExample.scala`: a
2//! multi-tenant product catalogue with a custom authentication context (API
3//! key scopes), typed rejections, a lifecycle state machine, an admin bypass
4//! and the tenant-aware SaaS driver (`tenant_id` / `owner_id` columns).
5//!
6//! ```sh
7//! cargo run -p edomata-examples --bin product_catalog

24/// Custom authentication context based on API key scopes (not
25/// `CallerIdentity`): any type works as long as an `AuthPolicy` exists.
26#[derive(Clone, Debug, PartialEq, Eq)]
27struct ApiKeyContext {
28    #[allow(dead_code)]
29    api_key: String,
30    tenant_id: TenantId,
31    owner_id: UserId,
32    scopes: BTreeSet<String>,
33}

35#[derive(Clone, Copy, Debug, PartialEq, Eq, Serialize, Deserialize)]
36enum ProductStatus {
37    Draft,
38    Published,
39    Archived,
40}
41
42impl ProductStatus {
43    fn name(self) -> &'static str {
44        match self {
45            ProductStatus::Draft => "Draft",
46            ProductStatus::Published => "Published",
47            ProductStatus::Archived => "Archived",
48        }
49    }
50}

83#[derive(Clone, Debug, PartialEq, Eq, Serialize, Deserialize)]
84#[serde(rename_all_fields = "camelCase")]
85enum ProductNotification {
86    Created { name: String, price_cents: i64 },
87    DetailsUpdated { name: String },
88    PriceUpdated { price_cents: i64, currency: String },
89    Published {},
90    Archived {},
91    Deleted {},
92}

94/// Typed rejections: guard errors are mapped by `mk_rejection`.
95#[derive(Clone, Debug, PartialEq, Eq, Serialize, Deserialize)]
96enum ProductRejection {
97    NotFound,
98    InvalidTransition { from: String, to: String },
99    InvalidPrice { reason: String },
100    Unauthorized { reason: String },
101}

107struct ProductModel;
108
109impl CqrsModel for ProductModel {
110    type State = CrudState<Product>;
111    type Rejection = ProductRejection;
112
113    fn initial(&self) -> CrudState<Product> {
114        CrudState::NonExistent
115    }
116}
117
118/// Scope-based authorisation: `catalog:read`, `catalog:write`, `catalog:admin`.
119struct ScopePolicy;
120
121impl AuthPolicy<ApiKeyContext> for ScopePolicy {
122    fn tenant_id(&self, auth: &ApiKeyContext) -> TenantId {
123        auth.tenant_id.clone()
124    }
125
126    fn authorize(&self, auth: &ApiKeyContext, action: CrudAction) -> Result<(), String> {
127        let required = match action {
128            CrudAction::Create | CrudAction::Update => "catalog:write",
129            CrudAction::Read => "catalog:read",
130            CrudAction::Delete => "catalog:admin",
131        };
132        if auth.scopes.contains(required) {
133            Ok(())
134        } else {
135            Err(format!("Missing scope: {required}"))
136        }
137    }
138}
139
140/// Maps guard error strings to typed rejections.
141fn mk_rejection(msg: String) -> ProductRejection {
142    if msg.contains("Tenant mismatch") || msg.contains("Entity not found") {
143        ProductRejection::NotFound
144    } else {
145        ProductRejection::Unauthorized { reason: msg }
146    }
147}

149type Service =
150    SaaSCqrsService<ApiKeyContext, ProductCommand, Product, ProductRejection, ProductNotification>;

153type ProductApp =
154    SaaSCqrsApp<ApiKeyContext, ProductCommand, Product, ProductRejection, ProductNotification, ()>;

156// ---------------------------------------------------------------------------
157// 3. SAAS SERVICE (write side: guards are automatic)
158// ---------------------------------------------------------------------------
159
160fn price_check(d: &Dsl, price: i64) -> ProductApp {
161    if price <= 0 {
162        d.reject(ProductRejection::InvalidPrice {
163            reason: "Price must be positive".into(),
164        })
165    } else {
166        d.unit()
167    }
168}
169
170/// Applies `f` to an active product, rejecting with `NotFound` otherwise.
171fn on_active<F>(d: &Dsl, f: F) -> ProductApp
172where
173    F: Fn(&Dsl, TenantId, UserId, Product) -> ProductApp + Send + Sync + 'static,
174{
175    let d2 = d.clone();
176    d.entity_state().and_then(move |state| match state {
177        CrudState::Active {
178            tenant_id,
179            owner_id,
180            data,
181        } => f(&d2, tenant_id, owner_id, data),
182        _ => d2.reject(ProductRejection::NotFound),
183    })
184}
185
186fn set_active(
187    d: &Dsl,
188    tid: TenantId,
189    oid: UserId,
190    p: Product,
191    n: ProductNotification,
192) -> ProductApp {
193    d.set(CrudState::active(tid, oid, p)).then(d.publish([n]))
194}
195
196fn product_service(service: &Service) -> ProductApp {
197    let d = service.saas().clone();
198    d.clone().guarded_router(move |cmd| match cmd {
199        ProductCommand::Create {
200            name,
201            description,
202            price_cents,
203            currency,
204        } => {
205            let d2 = d.clone();
206            let n = name.clone();
207            let logic = price_check(&d, price_cents)
208                .then(d.auth().and_then(move |c| {
209                    d2.set(CrudState::active(
210                        c.tenant_id,
211                        c.owner_id,
212                        Product {
213                            name: name.clone(),
214                            description: description.clone(),
215                            price_cents,
216                            currency: currency.clone(),
217                            status: ProductStatus::Draft,
218                        },
219                    ))
220                }))
221                .then(d.publish([ProductNotification::Created {
222                    name: n,
223                    price_cents,
224                }]));
225            (CrudAction::Create, logic)
226        }
227        ProductCommand::UpdateDetails { name, description } => (
228            CrudAction::Update,
229            on_active(&d, move |d, tid, oid, p| {
230                let p = Product {
231                    name: name.clone(),
232                    description: description.clone(),
233                    ..p
234                };
235                set_active(
236                    d,
237                    tid,
238                    oid,
239                    p,
240                    ProductNotification::DetailsUpdated { name: name.clone() },
241                )
242            }),
243        ),
244        ProductCommand::UpdatePrice {
245            price_cents,
246            currency,
247        } => (
248            CrudAction::Update,
249            price_check(&d, price_cents).then(on_active(&d, move |d, tid, oid, p| {
250                let p = Product {
251                    price_cents,
252                    currency: currency.clone(),
253                    ..p
254                };
255                set_active(
256                    d,
257                    tid,
258                    oid,
259                    p,
260                    ProductNotification::PriceUpdated {
261                        price_cents,
262                        currency: currency.clone(),
263                    },
264                )
265            })),
266        ),
267        ProductCommand::Publish => (
268            CrudAction::Update,
269            on_active(&d, |d, tid, oid, p| match p.status {
270                ProductStatus::Draft | ProductStatus::Archived => set_active(
271                    d,
272                    tid,
273                    oid,
274                    Product {
275                        status: ProductStatus::Published,
276                        ..p
277                    },
278                    ProductNotification::Published {},
279                ),
280                other => d.reject(ProductRejection::InvalidTransition {
281                    from: other.name().into(),
282                    to: "Published".into(),
283                }),
284            }),
285        ),
286        ProductCommand::Archive => (
287            CrudAction::Update,
288            on_active(&d, |d, tid, oid, p| match p.status {
289                ProductStatus::Published => set_active(
290                    d,
291                    tid,
292                    oid,
293                    Product {
294                        status: ProductStatus::Archived,
295                        ..p
296                    },
297                    ProductNotification::Archived {},
298                ),
299                other => d.reject(ProductRejection::InvalidTransition {
300                    from: other.name().into(),
301                    to: "Archived".into(),
302                }),
303            }),
304        ),
305        ProductCommand::Delete => (
306            CrudAction::Delete,
307            on_active(&d, |d, tid, oid, p| match p.status {
308                ProductStatus::Published => d.reject(ProductRejection::InvalidTransition {
309                    from: "Published".into(),
310                    to: "Deleted".into(),
311                }),
312                _ => d
313                    .set(CrudState::deleted(tid, oid))
314                    .then(d.publish([ProductNotification::Deleted {}])),
315            }),
316        ),
317    })
318}

320// ---------------------------------------------------------------------------
321// 4. ADMIN SERVICE (write side: guards bypassed for the super-admin)
322// ---------------------------------------------------------------------------
323
324fn product_admin_service(service: &Service) -> ProductApp {
325    let d = service.saas().clone();
326    d.clone().unsafe_unguarded_router(move |cmd| match cmd {
327        ProductCommand::Delete => on_active(&d, |d, tid, oid, _| {
328            d.set(CrudState::deleted(tid, oid))
329                .then(d.publish([ProductNotification::Deleted {}]))
330        }),
331        _ => d.reject(ProductRejection::Unauthorized {
332            reason: "Admin service: Delete only".into(),
333        }),
334    })
335}

337// ---------------------------------------------------------------------------
338// 5. SQL READ PROJECTION (via the notification handler)
339//
340// As in the Scala example, notifications do not carry the product id, so the
341// projection uses placeholders; a real projection would put the id in the
342// notifications (see the `saas_todo` example).
343// ---------------------------------------------------------------------------
344
345const READ_MODEL_DDL: &str = "CREATE TABLE IF NOT EXISTS products_read (
346  id          text        NOT NULL PRIMARY KEY,
347  tenant_id   text        NOT NULL,
348  name        text        NOT NULL,
349  description text        NOT NULL,
350  price_cents bigint      NOT NULL,
351  currency    text        NOT NULL,
352  status      text        NOT NULL,
353  deleted     boolean     NOT NULL DEFAULT false,
354  created_at  timestamptz NOT NULL DEFAULT now(),
355  updated_at  timestamptz NOT NULL DEFAULT now()
356)";
357
358fn projection() -> SqlxHandler<ProductNotification> {
359    SqlxHandler::new(|ns, conn| {
360        Box::pin(async move {
361            for n in ns.iter() {
362                let query = match n {
363                    ProductNotification::Created { name, price_cents } => sqlx::query(
364                        "INSERT INTO products_read (id, tenant_id, name, description, price_cents, currency, status) VALUES ($1, $2, $3, '', $4, 'USD', 'Draft') ON CONFLICT (id) DO UPDATE SET name = EXCLUDED.name, price_cents = EXCLUDED.price_cents, status = 'Draft', deleted = false, updated_at = now()",
365                    )
366                    .bind("placeholder-id")
367                    .bind("placeholder-tenant")
368                    .bind(name)
369                    .bind(price_cents),
370                    ProductNotification::DetailsUpdated { name } => {
371                        sqlx::query("UPDATE products_read SET name = $1, updated_at = now() WHERE id = $2")
372                            .bind(name)
373                            .bind("placeholder-id")
374                    }
375                    ProductNotification::PriceUpdated {
376                        price_cents,
377                        currency,
378                    } => sqlx::query(
379                        "UPDATE products_read SET price_cents = $1, currency = $2, updated_at = now() WHERE id = $3",
380                    )
381                    .bind(price_cents)
382                    .bind(currency)
383                    .bind("placeholder-id"),
384                    ProductNotification::Published {} => {
385                        sqlx::query("UPDATE products_read SET status = 'Published', updated_at = now() WHERE id = $1")
386                            .bind("placeholder-id")
387                    }
388                    ProductNotification::Archived {} => {
389                        sqlx::query("UPDATE products_read SET status = 'Archived', updated_at = now() WHERE id = $1")
390                            .bind("placeholder-id")
391                    }
392                    ProductNotification::Deleted {} => {
393                        sqlx::query("UPDATE products_read SET deleted = true, updated_at = now() WHERE id = $1")
394                            .bind("placeholder-id")
395                    }
396                };
397                query
398                    .execute(&mut *conn)
399                    .await
400                    .map_err(BackendError::unknown)?;
401            }
402            Ok(())
403        })
404    })
405}

411fn api_key(key: &str, tenant: &str, owner: &str, scopes: &[&str]) -> ApiKeyContext {
412    ApiKeyContext {
413        api_key: key.to_string(),
414        tenant_id: TenantId::new(tenant),
415        owner_id: UserId::new(owner),
416        scopes: scopes.iter().map(|s| s.to_string()).collect(),
417    }
418}

420#[tokio::main]
421async fn main() -> Result<(), Box<dyn std::error::Error>> {
422    let naming = PGNaming::prefixed_str("catalog")?;
423    // The DDL for migration tools (tenant-aware tables, optionally with RLS).
424    println!("-- SaaSPGSchema::cqrs(catalog):");
425    for statement in SaaSPGSchema::cqrs(&naming) {
426        println!("{statement}");
427    }
428
429    let pool = connect().await?;
430    sqlx::query(READ_MODEL_DDL).execute(&pool).await?;
431    let service: Service = SaaSCqrsService::new(ScopePolicy, mk_rejection);
432    let driver = SaaSSqlxCqrsDriver::new(naming, pool).await?;
433    let backend = Backend::builder(ProductModel, service.domain())
434        .driver(driver.clone())
435        .with_event_handler(projection())
436        .build(SaaSCodec::jsonb_state(), SaaSCodec::jsonb_notification())
437        .await?;
438    let run = backend.compile(product_service(&service));
439    let _admin_run = backend.compile(product_admin_service(&service));
440
441    let tenant_a = api_key(
442        "key-acme-001",
443        "acme-corp",
444        "alice",
445        &["catalog:read", "catalog:write"],
446    );
447    let tenant_b = api_key(
448        "key-globex-001",
449        "globex",
450        "bob",
451        &["catalog:read", "catalog:write"],
452    );
453    let admin = api_key(
454        "key-admin-001",
455        "acme-corp",
456        "admin",
457        &["catalog:read", "catalog:write", "catalog:admin"],
458    );
459    let product = format!("product-{}", uuid::Uuid::new_v4());
460    let cmd = |caller: &ApiKeyContext, c: ProductCommand| {
461        command(&product, SaaSCommand::new(caller.clone(), c))
462    };
463
464    // Tenant A creates a product (starts as Draft), updates the price, publishes.
465    println!(
466        "Create: {:?}",
467        run(cmd(
468            &tenant_a,
469            ProductCommand::Create {
470                name: "Widget".into(),
471                description: "A fine widget".into(),
472                price_cents: 1999,
473                currency: "USD".into(),
474            },
475        ))
476        .await?
477    );
478    println!(
479        "Update price: {:?}",
480        run(cmd(
481            &tenant_a,
482            ProductCommand::UpdatePrice {
483                price_cents: 2499,
484                currency: "USD".into(),
485            },
486        ))
487        .await?
488    );
489    println!(
490        "Publish: {:?}",
491        run(cmd(&tenant_a, ProductCommand::Publish)).await?
492    );
493    // Tenant B tries to update Tenant A's product: REJECTED (tenant mismatch).
494    println!(
495        "Cross-tenant attempt: {:?}",
496        run(cmd(
497            &tenant_b,
498            ProductCommand::UpdateDetails {
499                name: "Hacked!".into(),
500                description: "pwned".into(),
501            },
502        ))
503        .await?
504    );
505    // Cannot delete a published product.
506    println!(
507        "Delete published: {:?}",
508        run(cmd(&admin, ProductCommand::Delete)).await?
509    );
510    // Archive first, then delete.
511    println!(
512        "Archive: {:?}",
513        run(cmd(&tenant_a, ProductCommand::Archive)).await?
514    );
515    println!(
516        "Delete archived: {:?}",
517        run(cmd(&admin, ProductCommand::Delete)).await?
518    );
519
520    // The tenant-aware driver fills tenant_id / owner_id, so states can be
521    // listed per tenant.
522    let lister = TenantStateLister::new(&driver, SaaSCodec::<CrudState<Product>>::jsonb_state());
523    println!(
524        "acme-corp has {} product state(s)",
525        lister
526            .list_by_tenant(&TenantId::new("acme-corp"))
527            .await?
528            .len()
529    );
530    println!("Done!");
531    Ok(())
