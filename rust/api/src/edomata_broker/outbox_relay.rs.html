17/// Publishes unpublished outbox items to a [`Publisher`] and marks them as
18/// sent only after the broker acknowledged them: the `OutboxConsumer` of
19/// `edomata-backend` wired to a broker.
20///
21/// The relay passes over the outbox once when started, then whenever one of
22/// its wake-up streams yields (the backend's in-process outbox signal, a
23/// PostgreSQL `LISTEN` stream from [`crate::postgres::listen`], ...) and at
24/// least every [`RelayConfig::poll_interval`]. Items are published in
25/// sequence-number order, in batches of [`RelayConfig::batch_size`];
26/// transient failures are retried with exponential backoff and nothing is
27/// marked as sent until the batch succeeded, so delivery is at-least-once.
28///
29/// Several replicas can run the same relay with
30/// [`run_as_leader`](Self::run_as_leader): a PostgreSQL advisory lock makes
31/// sure only one publishes at a time.
32///
33/// If the process stops after the broker acknowledged a batch but before
34/// the items were marked, the next pass publishes them again with the same
35/// [`BrokerMessage::id`]: consumers deduplicate on it.
36///
37/// ```
38/// use std::sync::Arc;
39///
40/// use edomata_backend::eventsourcing::Backend;
41/// use edomata_backend::inmemory::InMemoryDriver;
42/// use edomata_broker::{MessageEncoder, OutboxRelay, RecordingPublisher, RelayConfig, RetryPolicy};
43/// use edomata_core::*;
44///
45/// struct Counter;
46/// impl DomainModel for Counter {
47///     type State = i32; type Event = i32; type Rejection = String;
48///     fn initial(&self) -> i32 { 0 }
49///     fn transition(&self, e: &i32, s: i32) -> Result<i32, NonEmpty<String>> { Ok(s + e) }
50/// }
51///
52/// # #[tokio::main(flavor = "current_thread")]
53/// # async fn main() -> Result<(), Box<dyn std::error::Error>> {
54/// let dsl = Counter.dsl::<i32, String>();
55/// let backend = Backend::builder(Counter, dsl)
56///     .driver(InMemoryDriver::new())
57///     .build_default()
58///     .await?;
59/// let service = backend.compile(dsl.router(move |n: i32| dsl.accept(n).publish([n.to_string()])));
60/// for (id, n) in [("c1", 1), ("c2", 2)] {
61///     assert!(service(CommandMessage::new(id, chrono::Utc::now(), "counter-1", n)).await?.is_ok());
62/// }
63///
64/// let publisher = Arc::new(RecordingPublisher::new());
65/// let relay = OutboxRelay::new(
66///     Arc::clone(backend.outbox()),
67///     Arc::clone(&publisher) as _,
68///     MessageEncoder::serde(),
69///     // Give up at the first failure, to show it in this example.
70///     RelayConfig::new("counters").with_retry(RetryPolicy { max_retries: Some(0), ..RetryPolicy::default() }),
71/// );
72///
73/// // The broker is down: the error is reported and nothing is marked...
74/// publisher.fail_next(1);
75/// assert!(relay.relay_once().await.is_err());
76///
77/// // ...and the next pass publishes both items, in sequence order.
78/// assert_eq!(relay.relay_once().await?, 2);
79/// let payloads: Vec<String> = publisher.messages().iter().map(|m| m.payload_text()).collect();
80/// assert_eq!(payloads, [r#""1""#, r#""2""#]);
81/// # Ok(()) }
82/// ```
83pub struct OutboxRelay<N> {
84    reader: Arc<dyn OutboxReader<N>>,
85    publisher: Arc<dyn Publisher>,
86    encoder: MessageEncoder<N>,
87    config: RelayConfig,
88    metrics: Arc<RelayMetrics>,
89    wakeups: Mutex<Vec<BoxStream<'static, ()>>>,
90}

100impl<N: Payload> OutboxRelay<N> {
101    /// A relay from an outbox reader (typically
102    /// [`Backend::outbox`](edomata_backend::eventsourcing::Backend::outbox)) to
103    /// a publisher, encoding payloads with `encoder`. The relay does nothing
104    /// until [`relay_once`](Self::relay_once), [`run`](Self::run) or
105    /// [`run_as_leader`](Self::run_as_leader) is called.
106    pub fn new(
107        reader: Arc<dyn OutboxReader<N>>,
108        publisher: Arc<dyn Publisher>,
109        encoder: MessageEncoder<N>,
110        config: RelayConfig,
111    ) -> Self {
112        Self {
113            reader,
114            publisher,
115            encoder,
116            config,
117            metrics: Arc::new(RelayMetrics::new()),
118            wakeups: Mutex::new(Vec::new()),
119        }
120    }
121
122    /// Adds a wake-up stream: every element triggers a pass. Streams are
123    /// taken by the first [`run`](Self::run) / [`run_as_leader`](Self::run_as_leader)
124    /// call; when they end, the relay keeps polling every
125    /// [`RelayConfig::poll_interval`].
126    pub fn wake_on(self, wakeups: BoxStream<'static, ()>) -> Self {
127        self.wakeups.lock().unwrap().push(wakeups);
128        self
129    }
130
131    /// The relay's counters, shared with the running relay.
132    pub fn metrics(&self) -> Arc<RelayMetrics> {
133        Arc::clone(&self.metrics)
134    }
135
136    /// The configuration.
137    pub fn config(&self) -> &RelayConfig {
138        &self.config
139    }
140
141    /// The message published for an outbox item; its id is
142    /// [`BrokerMessage::outbox_id`] of the relay source and the item's
143    /// sequence number, and it carries the item's correlation and causation
144    /// ids.
145    ///
146    /// # Errors
147    ///
148    /// [`RelayError::Encode`] when the payload cannot be encoded.
149    pub fn message(&self, item: &OutboxItem<N>) -> Result<BrokerMessage, RelayError> {
150        Ok(BrokerMessage {
151            id: BrokerMessage::outbox_id(&self.config.source, item.seq_nr),
152            source: self.config.source.clone(),
153            kind: MessageKind::Notification,
154            stream_id: item.stream_id.clone(),
155            seq_nr: item.seq_nr,
156            time: item.time,
157            content_type: self.encoder.content_type().to_string(),
158            payload: self.encoder.encode(&item.data)?,
159            correlation: item.metadata.correlation.clone(),
160            causation: item.metadata.causation.clone(),
161            extra_headers: Default::default(),
162        })
163    }
164
165    /// One pass: publishes and marks every pending item, batch by batch.
166    /// Returns the number of items published. Does not take the leader lock
167    /// and ignores the wake-up streams.
168    ///
169    /// # Errors
170    ///
171    /// - [`RelayError::Backend`] when reading the outbox or marking items
172    ///   fails;
173    /// - [`RelayError::Encode`] when a payload cannot be encoded;
174    /// - [`RelayError::Publish`] on a permanent publish failure, or once the
175    ///   [`RetryPolicy`](crate::RetryPolicy) budget is exhausted.
176    ///
177    /// Batches published before the error stay marked; the failed batch and
178    /// the ones after it stay pending.
179    pub async fn relay_once(&self) -> Result<usize, RelayError> {
180        self.pass(&CancellationToken::new()).await
181    }
182
183    async fn pass(&self, cancel: &CancellationToken) -> Result<usize, RelayError> {
184        let mut batches = self.reader.read().try_chunks(self.config.batch_size);
185        let mut total = 0usize;
186        while let Some(batch) = batches.next().await {
187            let batch = batch.map_err(|e| e.1)?;
188            let Some(items) = NonEmpty::from_vec(batch) else {
189                continue;
190            };
191            let messages = items
192                .iter()
193                .map(|item| self.message(item))
194                .collect::<Result<Vec<_>, _>>()?;
195            let messages = NonEmpty::from_vec(messages).expect("same length as items");
196            publish_with_retry(
197                &self.publisher,
198                &messages,
199                &self.config,
200                &self.metrics,
201                cancel,
202            )
203            .await?;
204            self.reader.mark_all_as_sent(&items).await?;
205            self.metrics.add_published(items.len() as u64);
206            total += items.len();
207            tracing::debug!(source = %self.config.source, count = items.len(), "published outbox batch");
208        }
209        self.metrics.set_lag(total as u64);
210        self.metrics.add_pass();
211        Ok(total)
212    }
213
214    /// Runs the relay until `cancel` fires: a pass now, then one after
215    /// every wake-up and at least every [`RelayConfig::poll_interval`].
216    /// Returns `Ok(())` when cancelled.
217    ///
218    /// # Errors
219    ///
220    /// The first error of a pass (see [`relay_once`](Self::relay_once))
221    /// stops the relay and is returned; [`RelayError::Cancelled`] when
222    /// `cancel` fires while a batch waits to be retried. Restart the relay
223    /// (with a supervisor or a retry loop) to resume from the unmarked items.
224    pub async fn run(&self, cancel: CancellationToken) -> Result<(), RelayError> {
225        let wakeups = std::mem::take(&mut *self.wakeups.lock().unwrap());
226        let inner = cancel.clone();
227        run_loop(&self.config, wakeups, cancel, || self.pass(&inner)).await
228    }
229
230    /// Runs the relay while holding the leader lock of the source: a
231    /// stand-by replica keeps trying to acquire it every
232    /// [`RelayConfig::leader_retry_interval`]; when the lock is lost (the
233    /// database connection dropped) the relay stops publishing and goes back
234    /// to standing by. Returns `Ok(())` when `cancel` fires.
235    ///
236    /// Use the same [`LeaderLock`] key for every replica of one relay, for
237    /// example `LeaderLock::new(pool, &config.source)`. A new leader resumes
238    /// from the unmarked items, so a batch in flight when the lock was lost
239    /// can be delivered twice.
240    ///
241    /// # Errors
242    ///
243    /// [`RelayError::Leader`] when the lock cannot be queried, plus the
244    /// errors of [`run`](Self::run) while leading.
245    ///
246    /// ```no_run
247    /// # use edomata_broker::{CancellationToken, OutboxRelay, postgres::LeaderLock};
248    /// # async fn example(relay: OutboxRelay<String>, pool: sqlx::PgPool) -> Result<(), edomata_broker::RelayError> {
249    /// let cancel = CancellationToken::new();
250    /// let shutdown = cancel.clone();
251    /// tokio::spawn(async move {
252    ///     // On a shutdown signal (here: after an hour), stop gracefully.
253    ///     tokio::time::sleep(std::time::Duration::from_secs(3600)).await;
254    ///     shutdown.cancel();
255    /// });
256    /// relay.run_as_leader(LeaderLock::new(pool, "accounts"), cancel).await
257    /// # }
258    /// ```
259    pub async fn run_as_leader(
260        &self,
261        lock: LeaderLock,
262        cancel: CancellationToken,
263    ) -> Result<(), RelayError> {
264        let wakeups = std::mem::take(&mut *self.wakeups.lock().unwrap());
265        let mut wakeups = Some(wakeups);
266        loop {
267            if cancel.is_cancelled() {
268                return Ok(());
269            }
270            match lock.try_acquire().await? {
271                Some(mut guard) => {
272                    tracing::info!(source = %self.config.source, "became the outbox relay leader");
273                    let leader_cancel = cancel.child_token();
274                    let streams = wakeups.take().unwrap_or_default();
275                    let inner = leader_cancel.clone();
276                    let relay = run_loop(&self.config, streams, leader_cancel.clone(), || {
277                        self.pass(&inner)
278                    });
279                    let lost = guard.lost();
280                    let outcome = tokio::select! {
281                        r = relay => Some(r),
282                        _ = lost => None,
283                    };
284                    match outcome {
285                        Some(result) => {
286                            guard.release().await;
287                            return result;
288                        }
289                        None => {
290                            tracing::warn!(source = %self.config.source, "leader lock lost, standing by");
291                            leader_cancel.cancel();
292                            guard.release().await;
293                            // The wake-up streams were consumed by the leader loop;
294                            // the polling fallback drives the next term.
295                        }
296                    }
297                }
298                None => {
299                    tracing::debug!(source = %self.config.source, "another relay is the leader");
300                    tokio::select! {
301                        _ = cancel.cancelled() => return Ok(()),
302                        _ = tokio::time::sleep(self.config.leader_retry_interval) => {}
303                    }
304                }
305            }
306        }
307    }
