11/// A boxed, thread-safe error: the source of [`PublishError`] and of some
12/// [`RelayError`](crate::RelayError) variants.
13pub type BoxError = Box<dyn std::error::Error + Send + Sync + 'static>;
14
15/// Why a batch could not be published.
16///
17/// The distinction drives the relay: transient failures are retried with
18/// the configured [`RetryPolicy`](crate::RetryPolicy), permanent ones stop
19/// the relay with [`RelayError::Publish`](crate::RelayError::Publish). In
20/// both cases nothing of the batch is marked as sent. When unsure, report a
21/// failure as transient: a retry at worst redelivers messages, which
22/// at-least-once consumers tolerate.
23///
24/// ```
25/// use edomata_broker::PublishError;
26///
27/// assert!(PublishError::transient("connection reset").is_transient());
28/// assert!(!PublishError::permanent("unknown topic").is_transient());
29/// ```
30#[derive(Debug, thiserror::Error)]
31pub enum PublishError {
32    /// The broker is unavailable or the batch was not acknowledged; the
33    /// relay retries with exponential backoff.
34    #[error("transient publish failure: {0}")]
35    Transient(#[source] BoxError),
36    /// The batch can never be published (misconfiguration, rejected
37    /// message); the relay stops and reports it.
38    #[error("permanent publish failure: {0}")]
39    Permanent(#[source] BoxError),
40}
41
42impl PublishError {
43    /// A transient failure from a message.
44    pub fn transient(message: impl Into<String>) -> Self {
45        PublishError::Transient(message.into().into())
46    }
47
48    /// A permanent failure from a message.
49    pub fn permanent(message: impl Into<String>) -> Self {
50        PublishError::Permanent(message.into().into())
51    }
52
53    /// Whether the relay should retry.
54    pub fn is_transient(&self) -> bool {
55        matches!(self, PublishError::Transient(_))
56    }
57}

59/// Publishes batches of messages to a broker.
60///
61/// A publisher must return `Ok` only once the broker acknowledged **every**
62/// message of the batch: the relays mark outbox items as sent (or advance
63/// the journal checkpoint) only after `publish` succeeded, which is what
64/// makes delivery at-least-once. Messages of a batch are given in sequence
65/// order and must be published in that order per stream.
66///
67/// A publisher may deliver part of a batch and then fail: the relay
68/// republishes the whole batch, so consumers see duplicates, never gaps.
69/// Classify failures with [`PublishError`]. Implementations exist for
70/// Kafka (`edomata_kafka::KafkaPublisher`), RabbitMQ
71/// (`edomata_rabbitmq::RabbitMqPublisher`), and tests
72/// ([`RecordingPublisher`]); `Arc<P>` is a publisher when `P` is.
73///
74/// ```
75/// use edomata_broker::{BrokerMessage, PublishError, Publisher};
76/// use edomata_core::NonEmpty;
77///
78/// /// Writes every message to standard output, one line each.
79/// struct Stdout;
80///
81/// #[async_trait::async_trait]
82/// impl Publisher for Stdout {
83///     async fn publish(&self, batch: &NonEmpty<BrokerMessage>) -> Result<(), PublishError> {
84///         for message in batch.iter() {
85///             // A real broker call; map its errors to transient / permanent.
86///             println!("{} {} {}", message.id, message.stream_id, message.payload_text());
87///         }
88///         // Return only once every message was acknowledged.
89///         Ok(())
90///     }
91/// }
92/// ```
93#[async_trait]
94pub trait Publisher: Send + Sync {
95    /// Publishes a batch, in order, and waits for the broker's
96    /// acknowledgment of every message.
97    ///
98    /// # Errors
99    ///
100    /// [`PublishError::Transient`] when the batch may succeed later (the
101    /// relay retries it), [`PublishError::Permanent`] when it never will
102    /// (the relay stops).
103    async fn publish(&self, batch: &NonEmpty<BrokerMessage>) -> Result<(), PublishError>;
104}
105
106#[async_trait]
107impl<P: Publisher + ?Sized> Publisher for Arc<P> {
108    async fn publish(&self, batch: &NonEmpty<BrokerMessage>) -> Result<(), PublishError> {
109        (**self).publish(batch).await
110    }
111}

113/// An in-memory publisher that records every message it receives, with
114/// programmable failures. Meant for tests and examples.
115///
116/// ```
117/// use std::sync::Arc;
118/// use edomata_broker::{
119///     BrokerMessage, MessageKind, PublishError, Publisher, RecordingPublisher,
120/// };
121/// use edomata_core::NonEmpty;
122///
123/// # fn message(seq_nr: i64) -> BrokerMessage {
124/// #     BrokerMessage {
125/// #         id: BrokerMessage::outbox_id("s", seq_nr), source: "s".into(),
126/// #         kind: MessageKind::Notification, stream_id: "a".into(), seq_nr,
127/// #         time: chrono::DateTime::UNIX_EPOCH, content_type: "application/json".into(),
128/// #         payload: b"{}".to_vec(), correlation: None, causation: None,
129/// #         extra_headers: Default::default(),
130/// #     }
131/// # }
132/// # futures::executor::block_on(async {
133/// let publisher = RecordingPublisher::new();
134/// let batch = NonEmpty::new(message(1));
135///
136/// publisher.fail_next(1); // the broker is down once
137/// assert!(matches!(publisher.publish(&batch).await, Err(PublishError::Transient(_))));
138/// assert!(publisher.messages().is_empty());
139///
140/// publisher.publish(&batch).await.unwrap();
141/// assert_eq!(publisher.ids(), ["s:outbox:1"]);
142/// assert_eq!(publisher.calls(), 2);
143/// # });
144/// ```
145#[derive(Debug, Default)]
146pub struct RecordingPublisher {
147    messages: Mutex<Vec<BrokerMessage>>,
148    fail_before: AtomicUsize,
149    fail_after: AtomicUsize,
150    calls: AtomicUsize,
151}
152
153impl RecordingPublisher {
154    /// An empty publisher.
155    pub fn new() -> Self {
156        Self::default()
157    }
158
159    /// The next `n` calls fail transiently **before** recording anything
160    /// (the broker is down).
161    pub fn fail_next(&self, n: usize) {
162        self.fail_before.store(n, Ordering::SeqCst);
163    }
164
165    /// The next `n` calls record the batch and **then** fail transiently:
166    /// the broker received the messages but the relay never learnt it (a
167    /// crash between publishing and marking).
168    pub fn fail_after_publishing_next(&self, n: usize) {
169        self.fail_after.store(n, Ordering::SeqCst);
170    }
171
172    /// Every message received so far, in order.
173    pub fn messages(&self) -> Vec<BrokerMessage> {
174        self.messages.lock().unwrap().clone()
175    }
176
177    /// The ids of every message received so far, in order.
178    pub fn ids(&self) -> Vec<String> {
179        self.messages
180            .lock()
181            .unwrap()
182            .iter()
183            .map(|m| m.id.clone())
184            .collect()
185    }
186
187    /// Number of `publish` calls so far (failed ones included).
188    pub fn calls(&self) -> usize {
189        self.calls.load(Ordering::SeqCst)
190    }
191
192    /// Forgets every recorded message.
193    pub fn clear(&self) {
194        self.messages.lock().unwrap().clear();
195    }
196}
197
198#[async_trait]
199impl Publisher for RecordingPublisher {
200    async fn publish(&self, batch: &NonEmpty<BrokerMessage>) -> Result<(), PublishError> {
201        self.calls.fetch_add(1, Ordering::SeqCst);
202        if take_one(&self.fail_before) {
203            return Err(PublishError::transient("broker unavailable"));
204        }
205        self.messages.lock().unwrap().extend(batch.iter().cloned());
206        if take_one(&self.fail_after) {
207            return Err(PublishError::transient("acknowledgment lost"));
208        }
209        Ok(())
210    }
211}
212
213fn take_one(counter: &AtomicUsize) -> bool {
214    counter
215        .fetch_update(Ordering::SeqCst, Ordering::SeqCst, |n| n.checked_sub(1))
216        .is_ok()
217}
