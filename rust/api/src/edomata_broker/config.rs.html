5/// Exponential backoff for transient publish failures.
6///
7/// When [`Publisher::publish`](crate::Publisher::publish) fails with
8/// [`PublishError::Transient`](crate::PublishError::Transient), the relay
9/// waits [`delay_for`](Self::delay_for)`(retry)` and publishes the same
10/// batch again: the delay starts at `initial_delay`, doubles after every
11/// failure and is capped at `max_delay`. Once the budget is
12/// [`exhausted`](Self::exhausted), the relay stops with
13/// [`RelayError::Publish`](crate::RelayError::Publish). Nothing is marked as
14/// sent while a batch is being retried.
15///
16/// ```
17/// use std::time::Duration;
18/// use edomata_broker::RetryPolicy;
19///
20/// let policy = RetryPolicy {
21///     initial_delay: Duration::from_millis(100),
22///     max_delay: Duration::from_secs(1),
23///     max_retries: Some(5),
24/// };
25/// assert_eq!(policy.delay_for(1), Duration::from_millis(100));
26/// assert_eq!(policy.delay_for(2), Duration::from_millis(200));
27/// assert_eq!(policy.delay_for(5), Duration::from_secs(1)); // capped
28/// assert!(!policy.exhausted(5));
29/// assert!(policy.exhausted(6));
30///
31/// // The default retries forever, from 200 ms up to 30 s.
32/// assert_eq!(RetryPolicy::default().max_retries, None);
33/// assert!(!RetryPolicy::default().exhausted(u32::MAX));
34/// ```
35#[derive(Clone, Copy, Debug, PartialEq, Eq)]
36pub struct RetryPolicy {
37    /// Delay before the first retry; it doubles after every failure.
38    pub initial_delay: Duration,
39    /// Upper bound of the delay.
40    pub max_delay: Duration,
41    /// Number of retries before giving up (`None`: retry forever).
42    pub max_retries: Option<u32>,
43}
44
45impl Default for RetryPolicy {
46    fn default() -> Self {
47        Self {
48            initial_delay: Duration::from_millis(200),
49            max_delay: Duration::from_secs(30),
50            max_retries: None,
51        }
52    }
53}
54
55impl RetryPolicy {
56    /// The delay before retry number `retry` (1-based):
57    /// `initial_delay * 2^(retry - 1)`, capped at `max_delay`. The
58    /// computation saturates instead of overflowing; `0` is treated as `1`.
59    pub fn delay_for(&self, retry: u32) -> Duration {
60        let factor = 2u32.saturating_pow(retry.saturating_sub(1));
61        self.initial_delay
62            .saturating_mul(factor)
63            .min(self.max_delay)
64    }
65
66    /// Whether retry number `retry` (1-based) exceeds the budget, that is
67    /// `retry > max_retries`; always `false` when `max_retries` is `None`.
68    pub fn exhausted(&self, retry: u32) -> bool {
69        self.max_retries.is_some_and(|max| retry > max)
70    }
71}

73/// Settings shared by the outbox and journal relays.
74///
75/// ```
76/// use std::time::Duration;
77/// use edomata_broker::{RelayConfig, RetryPolicy};
78///
79/// let config = RelayConfig::new("accounts")
80///     .with_batch_size(500)
81///     .with_poll_interval(Duration::from_secs(30))
82///     .with_retry(RetryPolicy { max_retries: Some(10), ..RetryPolicy::default() });
83/// assert_eq!(config.source, "accounts");
84/// assert_eq!(config.batch_size, 500);
85///
86/// // A batch size of 0 would never make progress: it is raised to 1.
87/// assert_eq!(RelayConfig::new("accounts").with_batch_size(0).batch_size, 1);
88/// ```
89#[derive(Clone, Debug, PartialEq, Eq)]
90pub struct RelayConfig {
91    /// Name of the source (typically the aggregate namespace); part of every
92    /// message id and the default topic / exchange. Use it as the
93    /// [`LeaderLock`](crate::postgres::LeaderLock) source too so that replicas of one relay share a lock.
94    pub source: String,
95    /// Items published (and marked) per batch; at least 1 when set through
96    /// [`with_batch_size`](Self::with_batch_size). Each batch is one
97    /// [`Publisher::publish`](crate::Publisher::publish) call.
98    pub batch_size: usize,
99    /// Polling fallback: the relay re-reads the source at least this often
100    /// even without a wake-up signal.
101    pub poll_interval: Duration,
102    /// Backoff on transient publish failures.
103    pub retry: RetryPolicy,
104    /// How often a stand-by relay retries to become the leader.
105    pub leader_retry_interval: Duration,
106}
107
108impl RelayConfig {
109    /// Defaults: batches of 100, polling every 5 seconds, unlimited retries
110    /// from 200 ms up to 30 s, leader election retried every second.
111    pub fn new(source: impl Into<String>) -> Self {
112        Self {
113            source: source.into(),
114            batch_size: 100,
115            poll_interval: Duration::from_secs(5),
116            retry: RetryPolicy::default(),
117            leader_retry_interval: Duration::from_secs(1),
118        }
119    }
120
121    /// Sets the batch size (at least 1).
122    pub fn with_batch_size(mut self, batch_size: usize) -> Self {
123        self.batch_size = batch_size.max(1);
124        self
125    }
126
127    /// Sets the polling fallback interval.
128    pub fn with_poll_interval(mut self, interval: Duration) -> Self {
129        self.poll_interval = interval;
130        self
131    }
132
133    /// Sets the retry policy.
134    pub fn with_retry(mut self, retry: RetryPolicy) -> Self {
135        self.retry = retry;
136        self
137    }
138
139    /// Sets the leader election retry interval.
140    pub fn with_leader_retry_interval(mut self, interval: Duration) -> Self {
141        self.leader_retry_interval = interval;
142        self
143    }
144}
