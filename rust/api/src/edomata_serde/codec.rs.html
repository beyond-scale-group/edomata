1//! The serde_json codec.
2
3use std::marker::PhantomData;
4
5use edomata_backend::{Codec, CodecError, PayloadFormat};
6use serde::Serialize;
7use serde::de::DeserializeOwned;
8
9/// A [`Codec`] for any `T: Serialize + DeserializeOwned`, backed by
10/// `serde_json`.
11///
12/// The format only selects the PostgreSQL column type the bytes are stored
13/// in; the bytes are always compact JSON (`serde_json::to_writer`), so a
14/// `bytea` payload written by this codec is readable as JSON too.
15///
16/// `SerdeCodec` is a small `Copy` value (just the format): it is `Default` (`jsonb`) and
17/// can be passed to `BackendBuilder::build`.
18///
19/// # Errors
20///
21/// [`Codec::encode`] fails with [`CodecError::Encode`] when `serde_json`
22/// cannot serialize the value (for instance a map with non-string keys);
23/// [`Codec::decode`] fails with [`CodecError::Decode`] when the bytes are not
24/// valid JSON for `T`. Both carry `serde_json`'s message.
25///
26/// ```
27/// use edomata_serde::{Codec, CodecError, PayloadFormat, SerdeCodec};
28///
29/// // The same JSON bytes, whatever the column type.
30/// for codec in [SerdeCodec::<Vec<u32>>::jsonb(), SerdeCodec::json(), SerdeCodec::bytea()] {
31///     assert_eq!(codec.encode(&vec![1, 2]).unwrap(), b"[1,2]");
32/// }
33/// assert_eq!(SerdeCodec::<u32>::bytea().format(), PayloadFormat::Bytea);
34/// assert_eq!(SerdeCodec::<u32>::default().format(), PayloadFormat::Jsonb);
35///
36/// let err = SerdeCodec::<u32>::jsonb().decode(b"\"not a number\"").unwrap_err();
37/// assert!(matches!(err, CodecError::Decode(_)));
38/// ```
39pub struct SerdeCodec<T> {
40    format: PayloadFormat,
41    _marker: PhantomData<fn() -> T>,
42}
43
44impl<T> Clone for SerdeCodec<T> {
45    fn clone(&self) -> Self {
46        *self
47    }
48}
49
50impl<T> Copy for SerdeCodec<T> {}

52impl<T> std::fmt::Debug for SerdeCodec<T> {
53    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
54        f.debug_struct("SerdeCodec")
55            .field("format", &self.format)
56            .finish()
57    }
58}
59
60impl<T> Default for SerdeCodec<T> {
61    /// The `jsonb` codec.
62    fn default() -> Self {
63        Self::jsonb()
64    }
65}
66
67impl<T> SerdeCodec<T> {
68    /// A codec storing JSON in a `jsonb` column (the default).
69    pub const fn jsonb() -> Self {
70        Self::with_format(PayloadFormat::Jsonb)
71    }
72
73    /// A codec storing JSON in a `json` column.
74    pub const fn json() -> Self {
75        Self::with_format(PayloadFormat::Json)
76    }
77
78    /// A codec storing JSON bytes in a `bytea` column.
79    pub const fn bytea() -> Self {
80        Self::with_format(PayloadFormat::Bytea)
81    }
82
83    /// A codec for the given column type.
84    pub const fn with_format(format: PayloadFormat) -> Self {
85        Self {
86            format,
87            _marker: PhantomData,
88        }
89    }
90}
91
92impl<T> Codec<T> for SerdeCodec<T>
93where
94    T: Serialize + DeserializeOwned + Send + Sync,
95{
96    fn format(&self) -> PayloadFormat {
97        self.format
98    }
99
100    fn encode(&self, value: &T) -> Result<Vec<u8>, CodecError> {
101        let mut out = Vec::new();
102        serde_json::to_writer(&mut out, value).map_err(|e| CodecError::Encode(e.to_string()))?;
103        Ok(out)
104    }
105
106    fn decode(&self, bytes: &[u8]) -> Result<T, CodecError> {
107        serde_json::from_slice(bytes).map_err(|e| CodecError::Decode(e.to_string()))
108    }
109}
