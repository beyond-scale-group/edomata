19#[derive(Clone, Debug, PartialEq, Eq, Serialize, Deserialize)]
20enum Event {
21    Opened,
22    Received(i32),
23    Closed,
24}
25
26#[derive(Clone, Debug, PartialEq, Eq, Serialize, Deserialize)]
27enum Rejection {
28    Unknown,
29}
30
31#[derive(Clone, Debug, PartialEq, Eq, Serialize, Deserialize)]
32enum Counter {
33    Empty,
34    Open(i32),
35    Closed,
36}
37
38impl Counter {
39    /// Receives `i`: opens the counter first when needed, refuses when
40    /// closed.
41    fn receive(&self, i: i32) -> Decision<Rejection, Event, Counter> {
42        let decision = match self {
43            Counter::Empty => Decision::accept(nonempty![Event::Opened, Event::Received(i)]),
44            Counter::Open(_) => Decision::accept(Event::Received(i)),
45            Counter::Closed => Decision::reject(Rejection::Unknown),
46        };
47        CounterModel.perform(self.clone(), decision)
48    }
49}
50
51struct CounterModel;
52
53impl DomainModel for CounterModel {
54    type State = Counter;
55    type Event = Event;
56    type Rejection = Rejection;
57
58    fn initial(&self) -> Counter {
59        Counter::Empty
60    }
61
62    // The Scala example leaves the state untouched on every event; the
63    // counter here actually counts, which is what a reader expects.
64    fn transition(&self, event: &Event, state: Counter) -> Result<Counter, NonEmpty<Rejection>> {
65        Ok(match (event, state) {
66            (Event::Opened, _) => Counter::Open(0),
67            (Event::Received(i), Counter::Open(n)) => Counter::Open(n + i),
68            (Event::Received(i), _) => Counter::Open(*i),
69            (Event::Closed, _) => Counter::Closed,
70        })
71    }
72}
73
74#[derive(Clone, Debug, PartialEq, Eq, Serialize, Deserialize)]
75enum Updates {
76    Updated,
77    Closed,
78}
79
80type CounterApp = App<String, Counter, Event, Rejection, Updates, ()>;
81
82/// `Application.app` in Scala: a router over string commands.
83fn app() -> CounterApp {
84    let dsl = CounterModel.dsl::<String, Updates>();
85    dsl.router(move |cmd| match cmd.as_str() {
86        "" => dsl.read().map(|ctx| ctx.command.derive_meta()).void(),
87        "receive" => dsl
88            .state()
89            .and_then(move |s| dsl.decide(s.receive(2)))
90            .and_then(move |ns| {
91                dsl.eval(move || {
92                    let ns = ns.clone();
93                    async move { println!("new state: {ns:?}") }
94                })
95            })
96            .and_then(move |()| dsl.publish([Updates::Updated])),
97        _ => dsl.reject(Rejection::Unknown),
98    })
99}
100
101/// `CounterService` in Scala: the same service written against the model's
102/// decisions directly.
103fn counter_service() -> CounterApp {
104    let dsl = CounterModel.dsl::<String, Updates>();
105    dsl.router(move |cmd| match cmd.as_str() {
106        "" => dsl
107            .state()
108            .and_then(move |s| dsl.decide(s.receive(2).void())),
109        _ => dsl.reject(Rejection::Unknown),
110    })
111}

113/// `SyntaxExample` in Scala: the `edomata_core::syntax` helpers.
114fn syntax_example() {
115    let l1: Decision<Rejection, Event, ()> = Err::<(), _>(Rejection::Unknown).to_decision();
116    let l2: Decision<Rejection, Event, ()> = Err::<Event, _>(Rejection::Unknown).to_accepted();
117    let l4: Decision<Rejection, Event, ()> =
118        Ok::<_, NonEmpty<Rejection>>(Event::Closed).to_accepted_nec();
119    let l5: Decision<Rejection, Event, ()> = Some(Event::Closed).to_accepted();
120    let l6: Decision<Rejection, Event, ()> = Some(Rejection::Unknown).to_rejected();
121    let l7: Decision<Rejection, Event, ()> = None::<Event>.to_accepted_or(Rejection::Unknown);
122    let l8: Decision<Rejection, Event, ()> = Event::Closed.accept();
123    let l9: Decision<Rejection, Event, ()> = Rejection::Unknown.reject();
124    let l10: Decision<Rejection, Event, i32> = 1.into_decision();
125    println!("syntax: {l1:?} {l2:?} {l4:?} {l5:?} {l6:?} {l7:?} {l8:?} {l9:?} {l10:?}");
126}

128#[tokio::main]
129async fn main() -> Result<(), Box<dyn std::error::Error>> {
130    // `Counter.Empty.perform(Decision.accept(Event.Opened))` in Scala.
131    let ns = CounterModel.perform(Counter::Empty, Decision::accept(Event::Opened));
132    println!("performed: {ns:?}");
133    syntax_example();
134
135    let pool = connect().await?;
136    let driver = SqlxDriver::for_namespace("counter", pool).await?;
137    let backend = Backend::builder(CounterModel, CounterModel.dsl::<String, Updates>())
138        .driver(driver)
139        .in_mem_snapshot(200)
140        .with_retry_config(RetryConfig {
141            initial_delay: Duration::from_secs(2),
142            ..RetryConfig::default()
143        })
144        .build_default()
145        .await?;
146
147    let service = backend.compile(app());
148    let result = service(command("a", "receive".to_string())).await?;
149    println!("receive: {result:?}");
150    let result = backend.compile(counter_service())(command("a", String::new())).await?;
151    println!("counter service: {result:?}");
152    let state = backend.repository().get("a").await?;
153    println!("state of a: {state:?}");
154    backend.close().await?;
155    Ok(())
156}
