1//! Read and write sides of a CQRS storage.

3use async_trait::async_trait;
4use edomata_core::NonEmpty;
5
6use super::{AggregateState, CommandState};
7use crate::{BackendError, CommandRef, SeqNr};
8
9/// Reads the current state of aggregates.
10#[async_trait]
11pub trait RepositoryReader<S>: Send + Sync {
12    /// Current state of the aggregate; the model's initial state at version
13    /// 0 when it was never saved.
14    async fn get(&self, id: &str) -> Result<AggregateState<S>, BackendError>;
15}
16
17/// Loads aggregates for commands and saves the outcome of handling them.
18///
19/// `save` must be atomic: the state, the notifications and the command
20/// record are written in one transaction, and a concurrent save of the same
21/// aggregate (or a duplicate command id) fails with
22/// [`BackendError::VersionConflict`].
23#[async_trait]
24pub trait Repository<S, N>: RepositoryReader<S> {
25    /// Loads the aggregate targeted by `cmd`, or reports the command as
26    /// redundant when it was already handled.
27    async fn load(&self, cmd: CommandRef<'_>) -> Result<CommandState<S>, BackendError>;
28
29    /// Saves the new state (expected to replace version `version`),
30    /// publishes the notifications and records the command, atomically.
31    async fn save(
32        &self,
33        cmd: CommandRef<'_>,
34        version: SeqNr,
35        new_state: S,
36        notifications: Vec<N>,
37    ) -> Result<(), BackendError>;
38
39    /// Publishes notifications without touching the aggregate.
40    async fn notify(
41        &self,
42        cmd: CommandRef<'_>,
43        notifications: NonEmpty<N>,
44    ) -> Result<(), BackendError>;
45}
