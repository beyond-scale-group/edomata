10/// Retry policy applied by command handlers on
11/// [`BackendError::VersionConflict`].
12#[derive(Clone, Copy, Debug, PartialEq, Eq)]
13pub struct RetryConfig {
14    /// Maximum number of attempts (the first attempt included).
15    pub max_retry: u32,
16    /// Delay before the second attempt; it doubles after every failure and a
17    /// random jitter of up to 500 ms is added.
18    pub initial_delay: Duration,
19}
20
21impl Default for RetryConfig {
22    fn default() -> Self {
23        Self {
24            max_retry: 5,
25            initial_delay: Duration::from_secs(2),
26        }
27    }
28}

30/// Runs `f` up to `max` times while it fails with
31/// [`BackendError::VersionConflict`], waiting `wait` (doubling each time,
32/// plus a random jitter of up to 500 ms) between attempts. When the last
33/// attempt still conflicts, fails with [`BackendError::MaxRetryExceeded`].
34/// Any other error is returned as is.
35///
36/// Command handlers use it with their [`RetryConfig`]; call it directly to
37/// retry your own read-decide-write loops. Waiting uses `tokio::time`, so
38/// it needs a Tokio runtime with the time driver enabled.
39///
40/// # Errors
41///
42/// [`BackendError::MaxRetryExceeded`] after `max` conflicting attempts, or
43/// the first error that is not a [`BackendError::VersionConflict`].
44///
45/// ```
46/// use std::time::Duration;
47/// use edomata_backend::{retry, BackendError};
48///
49/// # tokio::runtime::Builder::new_current_thread().enable_time().start_paused(true).build().unwrap().block_on(async {
50/// // Conflicts twice, then succeeds on the third attempt.
51/// let mut attempts = 0;
52/// let result = retry(3, Duration::from_millis(10), || {
53///     attempts += 1;
54///     let outcome = if attempts < 3 { Err(BackendError::VersionConflict) } else { Ok(attempts) };
55///     async move { outcome }
56/// })
57/// .await;
58/// assert_eq!(result, Ok(3));
59///
60/// // Always conflicting: gives up after `max` attempts.
61/// let result: Result<(), _> = retry(2, Duration::ZERO, || async { Err(BackendError::VersionConflict) }).await;
62/// assert_eq!(result, Err(BackendError::MaxRetryExceeded));
63/// # });
64/// ```
65pub async fn retry<T, F, Fut>(max: u32, wait: Duration, mut f: F) -> Result<T, BackendError>
66where
67    F: FnMut() -> Fut,
68    Fut: Future<Output = Result<T, BackendError>>,
69{
70    let mut remaining = max;
71    let mut wait = wait;
72    loop {
73        match f().await {
74            Err(BackendError::VersionConflict) if remaining > 1 => {
75                let jitter = Duration::from_millis(rand::rng().random_range(0..500));
76                tokio::time::sleep(wait + jitter).await;
77                wait *= 2;
78                remaining -= 1;
79            }
80            Err(BackendError::VersionConflict) => return Err(BackendError::MaxRetryExceeded),
81            other => return other,
82        }
83    }
84}
85
86/// [`retry`] with a [`RetryConfig`].
87///
88/// # Errors
89///
90/// As [`retry`].
91pub async fn retry_with<T, F, Fut>(config: RetryConfig, f: F) -> Result<T, BackendError>
92where
93    F: FnMut() -> Fut,
94    Fut: Future<Output = Result<T, BackendError>>,
95{
96    retry(config.max_retry, config.initial_delay, f).await
