12/// Reads the current state and the history of an aggregate.
13#[async_trait]
14pub trait RepositoryReader<S, E, R>: Send + Sync {
15    /// Current state of the aggregate, folded from its snapshot and journal.
16    async fn get(&self, stream_id: &str) -> Result<AggregateState<S, E, R>, BackendError>;
17
18    /// Every state the aggregate went through, from the initial state to the
19    /// current one (or to the first conflict, included).
20    fn history(
21        &self,
22        stream_id: &str,
23    ) -> BoxStream<'_, Result<AggregateState<S, E, R>, BackendError>>;
24}
25
26/// The default [`RepositoryReader`]: folds the journal with the model,
27/// starting from the latest snapshot when there is one.
28pub struct JournalRepositoryReader<S, E, R> {
29    journal: Arc<dyn JournalReader<E>>,
30    snapshot: Arc<dyn SnapshotReader<S>>,
31    model: SharedModel<S, E, R>,
32}
33
34impl<S, E, R> JournalRepositoryReader<S, E, R> {
35    /// Builds a reader over a journal and a snapshot reader.
36    pub fn new(
37        journal: Arc<dyn JournalReader<E>>,
38        snapshot: Arc<dyn SnapshotReader<S>>,
39        model: SharedModel<S, E, R>,
40    ) -> Self {
41        Self {
42            journal,
43            snapshot,
44            model,
45        }
46    }
47}
48
49#[async_trait]
50impl<S: Payload, E: Payload, R: Payload> RepositoryReader<S, E, R>
51    for JournalRepositoryReader<S, E, R>
52{
53    async fn get(&self, stream_id: &str) -> Result<AggregateState<S, E, R>, BackendError> {
54        match self.snapshot.get(stream_id).await? {
55            Some(last) => {
56                let events = self.journal.read_stream_after(stream_id, last.version - 1);
57                last_state(scan_state(
58                    events,
59                    AggregateState::Valid(last),
60                    &*self.model,
61                ))
62                .await
63            }
64            None => last_state(self.history(stream_id)).await,
65        }
66    }
67
68    fn history(
69        &self,
70        stream_id: &str,
71    ) -> BoxStream<'_, Result<AggregateState<S, E, R>, BackendError>> {
72        let initial = AggregateState::Valid(ValidState::new(self.model.initial(), 0));
73        Box::pin(scan_state(
74            self.journal.read_stream(stream_id),
75            initial,
76            &*self.model,
77        ))
78    }
79}

81/// Emits `last`, then every state obtained by applying the events in order,
82/// stopping after the first conflict (which is emitted).
83pub(crate) fn scan_state<'a, S, E, R>(
84    events: impl Stream<Item = Result<EventMessage<E>, BackendError>> + Send + 'a,
85    last: AggregateState<S, E, R>,
86    model: &'a (dyn edomata_core::DomainModel<State = S, Event = E, Rejection = R> + Send + Sync),
87) -> impl Stream<Item = Result<AggregateState<S, E, R>, BackendError>> + Send + 'a
88where
89    S: Payload,
90    E: Payload,
91    R: Payload,
92{
93    let folded = events.scan(Some(last.clone()), move |current, event| {
94        let next = match (current.take(), event) {
95            (None, _) => None,
96            (Some(_), Err(e)) => Some(Err(e)),
97            (Some(AggregateState::Valid(ValidState { state, version })), Ok(ev)) => {
98                match model.transition(&ev.payload, state.clone()) {
99                    Ok(s) => {
100                        let next = AggregateState::valid(s, version + 1);
101                        *current = Some(next.clone());
102                        Some(Ok(next))
103                    }
104                    Err(errors) => Some(Ok(AggregateState::Conflicted {
105                        last: state,
106                        on_event: ev,
107                        errors,
108                    })),
109                }
110            }
111            (Some(conflicted @ AggregateState::Conflicted { .. }), Ok(_)) => {
112                // Scala keeps emitting the conflicted state but `takeWhile`
113                // stops right after it; we stop as well.
114                let _ = conflicted;
115                None
116            }
117        };
118        std::future::ready(next)
119    });
120    futures::stream::once(std::future::ready(Ok(last))).chain(folded)
121}
122
123async fn last_state<S, E, R>(
124    stream: impl Stream<Item = Result<AggregateState<S, E, R>, BackendError>> + Send,
125) -> Result<AggregateState<S, E, R>, BackendError> {
126    let mut stream = std::pin::pin!(stream);
127    let mut last = None;
128    while let Some(item) = stream.try_next().await? {
129        last = Some(item);
130    }
131    last.ok_or_else(|| BackendError::persistence("empty aggregate history"))
132}
