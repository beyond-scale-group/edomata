1//! In-process update signals.
2
3use std::sync::Arc;
4
5use futures::Stream;
6use futures::stream::BoxStream;
7use tokio::sync::Notify;
8
9/// A coalescing, in-process signal: the counterpart of Scala's
10/// `Queue.circularBuffer[F, Unit](1)`.
11///
12/// Notifying stores at most one pending signal, so a burst of notifications
13/// wakes the listener once; a signal raised before anyone listens is kept
14/// until it is consumed.
15#[derive(Clone, Debug, Default)]
16pub struct Signal {
17    notify: Arc<Notify>,
18}
19
20impl Signal {
21    /// Creates a signal.
22    pub fn new() -> Self {
23        Self::default()
24    }
25
26    /// Raises the signal.
27    pub fn notify(&self) {
28        self.notify.notify_one();
29    }
30
31    /// Waits for the next signal.
32    pub async fn wait(&self) {
33        self.notify.notified().await;
34    }
35
36    /// An endless stream that yields once per (coalesced) signal.
37    pub fn stream(&self) -> impl Stream<Item = ()> + Send + 'static {
38        futures::stream::unfold(Arc::clone(&self.notify), |notify| async move {
39            notify.notified().await;
40            Some(((), notify))
41        })
42    }
43
44    /// [`Signal::stream`], boxed.
45    pub fn boxed_stream(&self) -> BoxStream<'static, ()> {
46        Box::pin(self.stream())
47    }
48}
