8/// A domain model whose transition returns a plain `Result` with a vector
9/// of rejections. Mirrors Scala's `JDomainModel`; implement it directly or
10/// build one from closures with [`ClosureModel::new`] (`JDomainModel.create`).
11///
12/// ```
13/// use edomata_simple::SimpleDomainModel;
14///
15/// enum Event { Deposited(u64), Withdrawn(u64) }
16///
17/// struct Account;
18///
19/// impl SimpleDomainModel for Account {
20///     type State = u64;
21///     type Event = Event;
22///     type Rejection = String;
23///
24///     fn initial(&self) -> u64 { 0 }
25///
26///     fn transition(&self, event: &Event, balance: u64) -> Result<u64, Vec<String>> {
27///         match event {
28///             Event::Deposited(n) => Ok(balance + n),
29///             Event::Withdrawn(n) => balance.checked_sub(*n).ok_or_else(|| vec!["insufficient balance".to_string()]),
30///         }
31///     }
32/// }
33///
34/// assert_eq!(Account.transition(&Event::Deposited(10), 0), Ok(10));
35/// assert!(Account.transition(&Event::Withdrawn(10), 0).is_err());
36/// ```
37pub trait SimpleDomainModel: Send + Sync + 'static {
38    /// Aggregate state.
39    type State;
40    /// Domain event.
41    type Event;
42    /// Domain error.
43    type Rejection;
44
45    /// Initial (empty) state for this aggregate.
46    fn initial(&self) -> Self::State;
47
48    /// Given an event and the current state, the new state or the
49    /// rejections. As in Scala, an `Err` must carry at least one reason:
50    /// [`ModelAdapter`] (and so the backend) panics on an empty `Err`
51    /// (`IllegalArgumentException` in Scala).
52    fn transition(
53        &self,
54        event: &Self::Event,
55        state: Self::State,
56    ) -> Result<Self::State, Vec<Self::Rejection>>;
57
58    /// Adapts this model to the core [`DomainModel`] trait
59    /// (`JDomainModel.toModelTC`).
60    fn into_model(self) -> ModelAdapter<Self>
61    where
62        Self: Sized,
63    {
64        ModelAdapter(self)
65    }
66}
67
68type TransitionFn<S, E, R> = dyn Fn(&E, S) -> Result<S, Vec<R>> + Send + Sync;
69
70/// A [`SimpleDomainModel`] built from closures.
71///
72/// ```
73/// use edomata_simple::{ClosureModel, SimpleDomainModel};
74///
75/// let model = <ClosureModel<i64, i64, String>>::new(0, |event: &i64, balance| {
76///     let next = balance + event;
77///     if next < 0 { Err(vec!["insufficient balance".to_string()]) } else { Ok(next) }
78/// });
79/// assert_eq!(model.initial(), 0);
80/// assert_eq!(model.transition(&5, 10), Ok(15));
81/// assert_eq!(model.transition(&-20, 10), Err(vec!["insufficient balance".to_string()]));
82/// ```
83pub struct ClosureModel<S, E, R> {
84    initial: S,
85    transition: Arc<TransitionFn<S, E, R>>,
86}
87
88impl<S: Clone, E, R> Clone for ClosureModel<S, E, R> {
89    fn clone(&self) -> Self {
90        Self {
91            initial: self.initial.clone(),
92            transition: Arc::clone(&self.transition),
93        }
94    }
95}
96
97impl<S: std::fmt::Debug, E, R> std::fmt::Debug for ClosureModel<S, E, R> {
98    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
99        f.debug_struct("ClosureModel")
100            .field("initial", &self.initial)
101            .finish()
102    }
103}

105impl<S, E, R> ClosureModel<S, E, R>
106where
107    S: Clone + Send + Sync + 'static,
108    E: Send + Sync + 'static,
109    R: Send + Sync + 'static,
110{
111    /// Builds a model from an initial state and a transition closure
112    /// (`JDomainModel.create`).
113    pub fn new<F>(initial: S, transition: F) -> Self
114    where
115        F: Fn(&E, S) -> Result<S, Vec<R>> + Send + Sync + 'static,
116    {
117        Self {
118            initial,
119            transition: Arc::new(transition),
120        }
121    }
122}

124impl<S, E, R> SimpleDomainModel for ClosureModel<S, E, R>
125where
126    S: Clone + Send + Sync + 'static,
127    E: Send + Sync + 'static,
128    R: Send + Sync + 'static,
129{
130    type State = S;
131    type Event = E;
132    type Rejection = R;
133
134    fn initial(&self) -> S {
135        self.initial.clone()
136    }
137
138    fn transition(&self, event: &E, state: S) -> Result<S, Vec<R>> {
139        (self.transition)(event, state)
140    }
141}

143/// A [`SimpleDomainModel`] seen as a core [`DomainModel`]: rejection vectors
144/// become [`NonEmpty`] chains.
145///
146/// # Panics
147///
148/// Its `transition` panics if the wrapped model returns `Err` with an empty
149/// vector, a programming error (Scala throws `IllegalArgumentException`).
150#[derive(Clone, Debug)]
151pub struct ModelAdapter<M>(pub M);
152
153impl<M: SimpleDomainModel> DomainModel for ModelAdapter<M> {
154    type State = M::State;
155    type Event = M::Event;
156    type Rejection = M::Rejection;
157
158    fn initial(&self) -> M::State {
159        self.0.initial()
160    }
161
162    fn transition(
163        &self,
164        event: &M::Event,
165        state: M::State,
166    ) -> Result<M::State, NonEmpty<M::Rejection>> {
167        self.0.transition(event, state).map_err(|reasons| {
168            NonEmpty::from_vec(reasons).expect("Rejection must have at least one reason")
169        })
170    }
171}
