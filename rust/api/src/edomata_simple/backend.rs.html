21/// Outcome of handling a command: `Ok(Ok(()))` when accepted or already
22/// handled, `Ok(Err(reasons))` when rejected, `Err` on storage failure.
23pub type HandleResult<R> = Result<Result<(), Vec<R>>, SimpleError>;

42/// A compiled handler: a function from a command message to a
43/// [`HandleResult`].
44pub type SimpleService<C, R> =
45    Arc<dyn Fn(CommandMessage<C>) -> BoxFuture<'static, HandleResult<R>> + Send + Sync>;

47/// An event-sourced backend on PostgreSQL with a closure-based API.
48/// Mirrors Scala's `JBackend`; build one with [`SimpleBackend::builder`]
49/// (see the [crate-level example](crate)).
50///
51/// Commands are handled with optimistic concurrency, retried on version
52/// conflicts up to the builder's `max_retry`, and deduplicated by command
53/// id: a command id already handled is accepted again without effect.
54pub struct SimpleBackend<S, E, R, N> {
55    inner: Backend<S, E, R, N>,
56}
57

60        Self {

101    /// Compiles a handler into a reusable service.
102    ///
103    /// # Panics
104    ///
105    /// The service panics if the handler returns a `Rejected` decision
106    /// without reasons, a programming error (Scala throws
107    /// `IllegalArgumentException`).
108    pub fn compile<C: Payload>(
109        &self,
110        handler: &CommandHandler<C, S, E, R, N>,
111    ) -> SimpleService<C, R> {
112        let handler = handler.clone();
113        let app: Edomaton<RequestContext<C, S>, R, E, N, ()> =
114            Edomaton::new(move |ctx: RequestContext<C, S>| {
115                let run = handler.call(Context::from_request(ctx));
116                async move {
117                    let out = run.await;
118                    // As in Scala, a rejection without reasons is a programming
119                    // error (`IllegalArgumentException`).
120                    let decision = out
121                        .decision
122                        .into_decision()
123                        .expect("Rejected must have at least one reason");
124                    ResponseD::new(decision, out.notifications)
125                }
126            });
127        let service = self.inner.compile(app);
128        Arc::new(move |cmd| {
129            let fut = service(cmd);
130            Box::pin(async move {
131                match fut.await {
132                    Ok(Ok(())) => Ok(Ok(())),
133                    Ok(Err(reasons)) => Ok(Err(reasons.into_vec())),
134                    Err(e) => Err(SimpleError::Backend(e)),
135                }
136            })
137        })
138    }
139
140    /// Handles one command with `handler` (`JBackend.handle`). Compiles the
141    /// handler on each call; use [`compile`](Self::compile) to reuse it.
142    ///
143    /// # Errors
144    ///
145    /// [`SimpleError::Backend`] on storage failures, including
146    /// `BackendError::MaxRetryExceeded` after too many version conflicts.
147    /// A rejection is not an error: it is the inner `Err(reasons)`.
148    ///
149    /// # Panics
150    ///
151    /// See [`compile`](Self::compile).
152    pub async fn handle<C: Payload>(
153        &self,
154        handler: &CommandHandler<C, S, E, R, N>,
155        command: CommandMessage<C>,
156    ) -> HandleResult<R> {
157        self.compile(handler)(command).await
158    }

293/// Builder of a [`SimpleBackend`] (`JBackendBuilder`). Required settings:
294/// a namespace, a connection ([`pool`](Self::pool) or

297    model: M,
298    naming: Option<Result<PGNaming, SimpleError>>,
299    pool: Option<PgPool>,
300    database_url: Option<String>,
301    event_codec: Option<SqlxCodec<M::Event>>,
302    notification_codec: Option<SqlxCodec<N>>,
303    max_retry: u32,
304    in_mem_snapshot_size: usize,

306}

308impl<M: SimpleDomainModel, N> std::fmt::Debug for SimpleBackendBuilder<M, N> {
309    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
310        f.debug_struct("SimpleBackendBuilder")
311            .field(
312                "naming",
313                &self
314                    .naming
315                    .as_ref()
316                    .map(|n| n.as_ref().map(|n| format!("{n:?}"))),

318            .field("has_pool", &self.pool.is_some())
319            .field(
320                "database_url",
321                &self.database_url.as_deref().map(|_| "<set>"),
322            )
323            .field("has_event_codec", &self.event_codec.is_some())
324            .field("has_notification_codec", &self.notification_codec.is_some())
325            .field("max_retry", &self.max_retry)
326            .field("in_mem_snapshot_size", &self.in_mem_snapshot_size)
327            .field("skip_setup", &self.skip_setup)
328            .finish()
329    }

425    /// Builds the backend, opening the connection pool when needed and
426    /// creating the tables unless `skip_setup` is set.
427    ///
428    /// # Errors
429    ///
430    /// - [`SimpleError::MissingConfig`] if the namespace, a codec or the
431    ///   connection is missing;
432    /// - [`SimpleError::InvalidNamespace`] for an invalid namespace;
433    /// - [`SimpleError::Connection`] if the pool cannot connect;
434    /// - [`SimpleError::Backend`] if the DDL setup fails.
435    pub async fn build(self) -> Result<BuiltBackend<M, N>, SimpleError> {
436        let naming = self
437            .naming
438            .ok_or(SimpleError::MissingConfig("namespace"))??;
439        let event_codec = self
440            .event_codec
441            .ok_or(SimpleError::MissingConfig("eventCodec"))?;
442        let notification_codec = self
443            .notification_codec
444            .ok_or(SimpleError::MissingConfig("notificationCodec"))?;
445        let pool = match (self.pool, self.database_url) {
446            (Some(pool), _) => pool,
447            (None, Some(url)) => PgPoolOptions::new()
448                .connect(&url)
449                .await
450                .map_err(SimpleError::Connection)?,
451            (None, None) => return Err(SimpleError::MissingConfig("pool or databaseUrl")),
452        };
453        let driver = SqlxDriver::new_with(naming, pool, self.skip_setup).await?;
454        let inner = Backend::builder(ModelAdapter(self.model), DomainDsl::<(), _, _, _, N>::new())
455            .driver(driver)
456            .in_mem_snapshot(self.in_mem_snapshot_size)
457            .with_retry_config(RetryConfig {
458                max_retry: self.max_retry,
459                ..RetryConfig::default()
460            })
461            .build(event_codec, notification_codec)
462            .await?;
463        Ok(SimpleBackend { inner })
464    }

466    /// Builds the backend on `runtime` and wraps it for blocking use
467    /// (`JBackendBuilder.build(runtime)`). Must not be called from within
468    /// an asynchronous context (it blocks on `runtime`).
469    ///
470    /// # Errors
471    ///
472    /// Those of [`build`](Self::build).
473    pub fn build_blocking(
474        self,
475        runtime: SimpleRuntime,
476    ) -> Result<BuiltBlockingBackend<M, N>, SimpleError> {
477        let backend = runtime.block_on(self.build())?;
478        Ok(BlockingBackend::new(backend, runtime))
479    }
