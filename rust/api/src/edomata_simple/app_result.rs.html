1//! The output of a command handler, the counterpart of `JAppResult`.
2
3use crate::SimpleDecision;
4
5/// What a command handler returns: a decision (with a `()` result) and the
6/// notifications to publish. Mirrors Scala's `JAppResult`. Notifications
7/// are written to the outbox together with the events, and also when the
8/// decision is a rejection.
9///
10/// ```
11/// use edomata_simple::AppResult;
12///
13/// let result: AppResult<String, i64, String> = AppResult::accept([100]).and_publish(["deposited".to_string()]);
14/// assert_eq!(result.decision.events(), &[100]);
15/// assert_eq!(result.notifications, ["deposited".to_string()]);
16///
17/// // A rejection must carry at least one reason (see `SimpleBackend::compile`).
18/// let rejected: AppResult<String, i64, String> = AppResult::reject(["zero amount".to_string()]);
19/// assert!(rejected.decision.is_rejected());
20/// ```
21#[derive(Clone, Debug, PartialEq, Eq)]
22pub struct AppResult<R, E, N> {
23    /// The decision.
24    pub decision: SimpleDecision<R, E, ()>,
25    /// Notifications to publish, whatever the decision.
26    pub notifications: Vec<N>,
27}
28
29impl<R, E, N> AppResult<R, E, N> {
30    /// A result with a decision and no notification (`JAppResult.decide`).
31    pub fn decide(decision: SimpleDecision<R, E, ()>) -> Self {
32        Self {
33            decision,
34            notifications: Vec::new(),
35        }
36    }
37
38    /// A result with a decision and notifications
39    /// (`JAppResult.decideAndPublish`).
40    pub fn decide_and_publish<I: IntoIterator<Item = N>>(
41        decision: SimpleDecision<R, E, ()>,
42        notifications: I,
43    ) -> Self {
44        Self {
45            decision,
46            notifications: notifications.into_iter().collect(),
47        }
48    }
49
50    /// Accepts events (`JAppResult.accept`).
51    pub fn accept<I: IntoIterator<Item = E>>(events: I) -> Self {
52        Self::decide(SimpleDecision::<R, E, ()>::accept(events))
53    }
54
55    /// Rejects (`JAppResult.reject`).
56    pub fn reject<I: IntoIterator<Item = R>>(reasons: I) -> Self {
57        Self::decide(SimpleDecision::reject(reasons))
58    }
59
60    /// Publishes notifications without deciding anything
61    /// (`JAppResult.publish`).
62    pub fn publish<I: IntoIterator<Item = N>>(notifications: I) -> Self {
63        Self::decide_and_publish(SimpleDecision::<R, E, ()>::unit(), notifications)
64    }
65
66    /// Adds notifications to this result.
67    pub fn and_publish<I: IntoIterator<Item = N>>(mut self, notifications: I) -> Self {
68        self.notifications.extend(notifications);
69        self
70    }
71}
