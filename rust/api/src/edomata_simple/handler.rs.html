12/// What a handler sees: the command, its message and the current state.
13/// Mirrors Scala's `JRequestContext`.
14#[derive(Clone, Debug, PartialEq, Eq)]
15pub struct Context<C, S> {
16    /// The command payload.
17    pub command: C,
18    /// The full command message.
19    pub message: CommandMessage<C>,
20    /// The aggregate's current state.
21    pub state: S,
22}
23
24impl<C: Clone, S> Context<C, S> {
25    /// Builds a context from a core [`RequestContext`].
26    pub fn from_request(ctx: RequestContext<C, S>) -> Self {
27        Self {
28            command: ctx.command.payload.clone(),
29            message: ctx.command,
30            state: ctx.state,
31        }
32    }
33}
34
35impl<C, S> Context<C, S> {
36    /// The aggregate address.
37    pub fn address(&self) -> &str {
38        &self.message.address
39    }
40
41    /// The command message id.
42    pub fn message_id(&self) -> &str {
43        &self.message.id
44    }
45
46    /// When the command was issued.
47    pub fn time(&self) -> DateTime<Utc> {
48        self.message.time
49    }
50}

52type HandlerFn<C, S, E, R, N> =
53    dyn Fn(Context<C, S>) -> BoxFuture<'static, AppResult<R, E, N>> + Send + Sync;
54
55/// A command handler: a function from a [`Context`] to an [`AppResult`].
56/// Mirrors Scala's `JCommandHandler`, with a blocking
57/// ([`CommandHandler::new`]) and an asynchronous
58/// ([`CommandHandler::new_async`]) constructor.
59///
60/// ```
61/// use edomata_simple::{AppResult, CommandHandler, CommandMessage, Context};
62///
63/// let handler: CommandHandler<i64, i64, i64, String, String> = CommandHandler::new(|ctx| {
64///     if ctx.state + ctx.command < 0 {
65///         AppResult::reject(["insufficient balance".to_string()])
66///     } else {
67///         AppResult::accept([ctx.command])
68///     }
69/// });
70///
71/// let message = CommandMessage::new("cmd-1", chrono::DateTime::UNIX_EPOCH, "acc-1", -50);
72/// let ctx = Context { command: -50, message, state: 20 };
73/// let result = futures::executor::block_on(handler.call(ctx));
74/// assert_eq!(result.decision.reasons(), &["insufficient balance".to_string()]);
75/// ```
76pub struct CommandHandler<C, S, E, R, N> {
77    run: Arc<HandlerFn<C, S, E, R, N>>,
78}
79
80impl<C, S, E, R, N> Clone for CommandHandler<C, S, E, R, N> {
81    fn clone(&self) -> Self {
82        Self {
83            run: Arc::clone(&self.run),
84        }
85    }
86}
87
88impl<C, S, E, R, N> std::fmt::Debug for CommandHandler<C, S, E, R, N> {
89    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
90        f.write_str("CommandHandler")
91    }
92}

94impl<C, S, E, R, N> CommandHandler<C, S, E, R, N>
95where
96    C: Send + 'static,
97    S: Send + 'static,
98    E: Send + 'static,
99    R: Send + 'static,
100    N: Send + 'static,
101{
102    /// A handler from a plain function (`JCommandHandler.create`).
103    pub fn new<F>(f: F) -> Self
104    where
105        F: Fn(Context<C, S>) -> AppResult<R, E, N> + Send + Sync + 'static,
106    {
107        Self {
108            run: Arc::new(move |ctx| {
109                let out = f(ctx);
110                Box::pin(async move { out })
111            }),
112        }
113    }
114
115    /// A handler from an asynchronous function, e.g. one that calls an
116    /// external service before deciding. The future runs inside the
117    /// command's retry loop, so it may run more than once on version
118    /// conflicts.
119    pub fn new_async<F, Fut>(f: F) -> Self
120    where
121        F: Fn(Context<C, S>) -> Fut + Send + Sync + 'static,
122        Fut: Future<Output = AppResult<R, E, N>> + Send + 'static,
123    {
124        Self {
125            run: Arc::new(move |ctx| Box::pin(f(ctx))),
126        }
127    }
128
129    /// Runs the handler on a context, without touching any storage.
130    pub fn call(&self, ctx: Context<C, S>) -> BoxFuture<'static, AppResult<R, E, N>> {
131        (self.run)(ctx)
132    }
133}
