Enum CrudState

pub enum CrudState<A> {
    NonExistent,
    Active {
        tenant_id: TenantId,
        owner_id: UserId,
        data: A,
    },
    Deleted {
        tenant_id: TenantId,
        owner_id: UserId,
    },
}
Expand description

Lifecycle state of a tenant-owned entity: the state type of every SaaS aggregate. The tenant and owner are fixed when the entity is created and kept after deletion, so guards and tenant-aware storage can always attribute it.

use edomata_saas::CrudState;

let todo = CrudState::active("acme", "alice", "buy milk");
assert_eq!(todo.tenant_id().map(|t| t.value()), Some("acme"));
assert_eq!(todo.clone().map(str::len).data(), Some(&8));
assert!(!CrudState::<&str>::deleted("acme", "alice").is_active());
assert_eq!(CrudState::<&str>::default(), CrudState::NonExistent);
